# FastAPI + Agent 测试

使用 `pytest` + `httpx` 测试 FastAPI 端点，包括如何 mock Agent 避免真实 LLM 调用。

## 一、测试工具链

```bash
pip install pytest pytest-asyncio httpx
```

| 工具 | 作用 |
|------|------|
| `pytest` | 测试框架 |
| `pytest-asyncio` | 支持 async 测试函数 |
| `httpx.AsyncClient` | FastAPI 推荐的异步测试客户端 |
| `unittest.mock` | Mock Agent/LLM，避免真实调用 |

## 二、基础端点测试

```python
# test_api.py
import pytest
from httpx import AsyncClient, ASGITransport
from main import app  # 你的 FastAPI app

@pytest.fixture
async def client():
    """创建测试客户端（不启动真实服务器）"""
    transport = ASGITransport(app=app)
    async with AsyncClient(transport=transport, base_url="http://test") as c:
        yield c

@pytest.mark.asyncio
async def test_health(client):
    """测试健康检查端点"""
    response = await client.get("/health")
    assert response.status_code == 200
    assert response.json() == {"status": "ok"}

@pytest.mark.asyncio
async def test_chat_missing_message(client):
    """测试缺少必填字段 → 422"""
    response = await client.post("/chat", json={})
    assert response.status_code == 422  # Pydantic 校验失败

@pytest.mark.asyncio
async def test_chat_invalid_thread_id(client):
    """测试无效参数"""
    response = await client.post("/chat", json={"message": ""})
    # 根据业务逻辑，空消息可能返回 400
    assert response.status_code in (200, 400)
```

## 三、Mock Agent（避免真实 LLM 调用）

测试时不应该真的调用 LLM API（慢、花钱、不确定）。用 mock 替代：

```python
# test_chat.py
import pytest
from unittest.mock import AsyncMock, patch, MagicMock
from httpx import AsyncClient, ASGITransport
from main import app

@pytest.fixture
async def client():
    transport = ASGITransport(app=app)
    async with AsyncClient(transport=transport, base_url="http://test") as c:
        yield c

@pytest.mark.asyncio
async def test_chat_success(client):
    """Mock Agent，测试 /chat 端点的请求/响应格式"""
    # 构造 mock 返回值
    mock_message = MagicMock()
    mock_message.content = "你好！我是 AI 助手。"
    
    mock_result = {"messages": [MagicMock(), mock_message]}
    
    # Patch agent.ainvoke
    with patch("main.agent.ainvoke", new_callable=AsyncMock) as mock_invoke:
        mock_invoke.return_value = mock_result
        
        response = await client.post("/chat", json={
            "message": "你好",
            "thread_id": "test_001",
        })
        
        assert response.status_code == 200
        data = response.json()
        assert data["reply"] == "你好！我是 AI 助手。"
        assert data["thread_id"] == "test_001"
        
        # 验证 agent.ainvoke 被调用了一次
        mock_invoke.assert_called_once()

@pytest.mark.asyncio
async def test_chat_agent_error(client):
    """Agent 抛异常时，API 应返回 500"""
    with patch("main.agent.ainvoke", new_callable=AsyncMock) as mock_invoke:
        mock_invoke.side_effect = Exception("LLM API 超时")
        
        response = await client.post("/chat", json={
            "message": "你好",
            "thread_id": "test_002",
        })
        
        assert response.status_code == 500
```

## 四、测试流式响应（SSE）

```python
@pytest.mark.asyncio
async def test_stream_response(client):
    """测试 SSE 流式端点"""
    # Mock astream_events
    async def mock_events(*args, **kwargs):
        events = [
            {"event": "on_chat_model_stream", "data": {"chunk": MagicMock(content="你")}},
            {"event": "on_chat_model_stream", "data": {"chunk": MagicMock(content="好")}},
            {"event": "on_chat_model_stream", "data": {"chunk": MagicMock(content="！")}},
        ]
        for e in events:
            yield e
    
    with patch("main.agent.astream_events", return_value=mock_events()):
        response = await client.post("/chat/stream", json={
            "message": "你好",
            "thread_id": "test_stream",
        })
        
        assert response.status_code == 200
        assert "text/event-stream" in response.headers["content-type"]
        
        # 解析 SSE 数据
        body = response.text
        assert '"content": "你"' in body or '"content":"你"' in body
        assert '"type": "done"' in body or '"type":"done"' in body
```

## 五、测试文件上传

```python
@pytest.mark.asyncio
async def test_upload_document(client):
    """测试文档上传端点"""
    # 构造测试文件
    file_content = b"# Test Document\nThis is a test."
    
    response = await client.post(
        "/documents/upload",
        files={"file": ("test.md", file_content, "text/markdown")},
    )
    
    assert response.status_code == 200
    data = response.json()
    assert data["status"] == "processing"
    assert "task_id" in data

@pytest.mark.asyncio
async def test_upload_invalid_type(client):

    )
    assert response.status_code == 400
```

## 六、测试 WebSocket

```python
@pytest.mark.asyncio
async def test_websocket_chat(client):
    """测试 WebSocket 对话"""
    # httpx 不直接支持 WebSocket，用 starlette 的 TestClient
    from starlette.testclient import TestClient
    
    with TestClient(app) as sync_client:
        with sync_client.websocket_connect("/chat/test_ws") as ws:
            # 发送消息
            ws.send_json({"content": "你好"})
            
            # 接收回复（可能多条）
            messages = []
            while True:
                data = ws.receive_json()
                messages.append(data)
                if data.get("type") == "done":
                    break
            
            # 验证收到了 token 和 done
            types = [m["type"] for m in messages]
            assert "done" in types
```

## 七、集成测试（真实 Agent，标记慢测试）

```python
# test_integration.py
import pytest

@pytest.mark.asyncio
@pytest.mark.slow  # 标记为慢测试，CI 中可跳过
async def test_real_agent_chat(client):
    """集成测试：真实调用 LLM（慢、花钱）"""
    response = await client.post("/chat", json={
        "message": "1+1等于几？只回答数字",
        "thread_id": "integration_test",
    })
    
    assert response.status_code == 200
    reply = response.json()["reply"]
    assert "2" in reply
```

```bash
# 运行所有测试（跳过慢测试）

# 运行全部（包括集成测试）
pytest -v
```

## 九、常见陷阱

### 1. 测试中真实调用 LLM

```python
# 错误：每个测试都调 LLM → 慢、贵、不稳定
async def test_chat(client):
    response = await client.post("/chat", json={"message": "你好"})
    # 每次运行结果不同，断言不好写

# 正确：mock Agent，只测 HTTP 层逻辑
with patch("main.agent.ainvoke", new_callable=AsyncMock) as mock:
    mock.return_value = {"messages": [MagicMock(content="固定回复")]}
    ...
```

### 2. 异步测试忘记标记

```python
# 错误：async 测试没有 @pytest.mark.asyncio → 被跳过
async def test_something(client):
    ...

# 正确：加标记
@pytest.mark.asyncio
async def test_something(client):
    ...
```
