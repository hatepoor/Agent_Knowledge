# 文档加载器

> ⚠️ **`langchain-community` 已弃用**：该包已被官方 sunset，不再积极维护。新项目应直接使用 Python 库（`pypdf`、`beautifulsoup4`、`csv` 等）加载数据，然后手动包装为 `Document` 对象。`Document` 数据模型本身（`langchain_core.documents.Document`）不受影响。

> 文档加载的核心任务是从各种数据源加载内容并返回 `Document` 对象列表（`page_content` + `metadata`）。有两种方式实现：

## 推荐方式：直接用 Python 库（新）

不再依赖 `langchain_community` 的加载器类，直接用 Python 标准库或第三方库加载数据，然后手动构造 `Document` 对象。这是官方当前推荐的方式。

### PDF 加载

```python
import pypdf  # pip install pypdf
from langchain_core.documents import Document

def load_pdf(file_path: str) -> list[Document]:
    """加载 PDF，每页一个 Document"""
    reader = pypdf.PdfReader(file_path)
    return [
        Document(
            page_content=page.extract_text() or "",
            metadata={"source": file_path, "page": i},
        )
        for i, page in enumerate(reader.pages)
    ]

docs = load_pdf("./docs/whitepaper.pdf")
print(f"Loaded {len(docs)} pages")
for doc in docs[:3]:
    print(f"Page {doc.metadata['page']}: {doc.page_content[:80]}...")
```

**替代方案对比**：

| 库 | 特点 | 安装 |
|---|------|------|
| `pypdf` | 简单直接，每页提取文本 | `pip install pypdf` |
| `pdfplumber` | 表格提取效果更好 | `pip install pdfplumber` |
| `pymupdf` (fitz) | 速度最快，支持图片提取 | `pip install pymupdf` |
| `unstructured` | 自动识别标题/段落/表格 | `pip install "unstructured[pdf]"` |

### 网页加载

```python
import requests
import bs4  # pip install beautifulsoup4
from langchain_core.documents import Document

def load_web(url: str) -> list[Document]:
    """加载网页内容"""
    resp = requests.get(url, timeout=20, headers={
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
    })
    resp.raise_for_status()
    soup = bs4.BeautifulSoup(resp.text, "html.parser")
    return [Document(
        page_content=soup.get_text(separator="\n", strip=True),
        metadata={"source": url, "title": soup.title.string if soup.title else ""},
    )]

# 只提取特定元素
def load_web_selective(url: str) -> list[Document]:
    """只提取文章标题和内容"""
    resp = requests.get(url, timeout=20)
    soup = bs4.BeautifulSoup(resp.text, "html.parser", parse_only=bs4.SoupStrainer(
        class_=("post-title", "post-header", "post-content")
    ))
    return [Document(page_content=soup.get_text(), metadata={"source": url})]

docs = load_web("https://example.com/article")
```

### 文本文件加载

```python
from langchain_core.documents import Document
from pathlib import Path

def load_text(file_path: str, encoding: str = "utf-8") -> list[Document]:
    """加载文本文件"""
    content = Path(file_path).read_text(encoding=encoding)
    return [Document(
        page_content=content,
        metadata={"source": file_path},
    )]

# 自动检测编码（处理 GBK 等中文编码）
def load_text_auto(file_path: str) -> list[Document]:
    for enc in ("utf-8", "gbk", "gb2312", "latin-1"):
        try:
            content = Path(file_path).read_text(encoding=enc)
            return [Document(page_content=content, metadata={"source": file_path})]
        except UnicodeDecodeError:
            continue
    raise ValueError(f"无法解码文件: {file_path}")
```

### CSV 加载

```python
import csv
from langchain_core.documents import Document

def load_csv(file_path: str) -> list[Document]:
    """加载 CSV，每行一个 Document"""
    docs = []
    with open(file_path, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for i, row in enumerate(reader):
            # 将所有列拼接为 page_content
            content = "\n".join(f"{k}: {v}" for k, v in row.items())
            docs.append(Document(
                page_content=content,
                metadata={"source": file_path, "row": i},
            ))
    return docs

docs = load_csv("./data/users.csv")
print(docs[0].page_content)
# name: Alice
# age: 30
# city: Beijing
```

### JSON 加载

```python
import json
from langchain_core.documents import Document

def load_json(file_path: str, content_key: str = "content") -> list[Document]:
    """加载 JSON 数组，每个元素一个 Document"""
    with open(file_path, "r", encoding="utf-8") as f:
        data = json.load(f)
    return [
        Document(
            page_content=item.get(content_key, str(item)),
            metadata={"source": file_path, "index": i},
        )
        for i, item in enumerate(data)
    ]

# JSON Lines（每行一个 JSON 对象）
def load_jsonl(file_path: str, content_key: str = "content") -> list[Document]:
    docs = []
    with open(file_path, "r", encoding="utf-8") as f:
        for i, line in enumerate(f):
            item = json.loads(line)
            docs.append(Document(
                page_content=item.get(content_key, str(item)),
                metadata={"source": file_path, "line": i},
            ))
    return docs
```

### 批量加载整个目录

```python
from pathlib import Path
from langchain_core.documents import Document

def load_directory(
    dir_path: str,
    extensions: list[str] = None,
    recursive: bool = True,
) -> list[Document]:
    """批量加载目录下的文件"""
    base = Path(dir_path)
    pattern = "**/*" if recursive else "*"
    docs = []

    for file_path in base.glob(pattern):
        if not file_path.is_file():
            continue
        if extensions and file_path.suffix.lower() not in extensions:
            continue

        try:
            content = file_path.read_text(encoding="utf-8")
        except UnicodeDecodeError:
            content = file_path.read_text(encoding="gbk", errors="ignore")

        docs.append(Document(
            page_content=content,
            metadata={"source": str(file_path), "type": file_path.suffix},
        ))

    return docs

# 加载所有 .txt 和 .md 文件
docs = load_directory("./knowledge_base/", extensions=[".txt", ".md"])
print(f"Loaded {len(docs)} documents")
```

## 旧方式：langchain_community 加载器（已弃用）

> ⚠️ 以下内容仅供参考旧代码。新项目请使用上面的推荐方式。

```python
# ❌ 不推荐：会触发 DeprecationWarning
from langchain_community.document_loaders import TextLoader, PyPDFLoader, WebBaseLoader

loader = TextLoader("./docs/intro.txt")
docs = loader.load()
```

### 旧加载器速查表

| 加载器 | 数据源 | 新方式替代 |
|--------|--------|-----------|
| `TextLoader` | .txt 文件 | `Path.read_text()` |
| `PyPDFLoader` | .pdf 文件 | `pypdf.PdfReader` |
| `WebBaseLoader` | 网页 URL | `requests` + `beautifulsoup4` |
| `CSVLoader` | .csv 文件 | `csv.DictReader` |
| `JSONLoader` | .json 文件 | `json.load()` |
| `DirectoryLoader` | 整个目录 | `pathlib.Path.glob()` |
| `UnstructuredFileLoader` | 几乎所有格式 | `pip install unstructured`（独立包，不受影响） |
| `NotionDBLoader` | Notion 数据库 | Notion API |
| `GitLoader` | Git 仓库 | `GitPython` 库 |

### 旧方式通用接口

```python
# 旧加载器的通用接口（仅供参考）
docs = loader.load()           # 同步加载，返回 List[Document]
docs = await loader.aload()    # 异步加载
```

## 完整 RAG 预处理流程（新方式）

```python
from pathlib import Path
import pypdf
from langchain_core.documents import Document
from langchain_text_splitters import RecursiveCharacterTextSplitter

# 1. 加载文档（直接用 Python 库）
def load_all_docs(base_dir: str) -> list[Document]:
    docs = []
    base = Path(base_dir)

    # 加载 PDF
    for pdf_path in base.glob("**/*.pdf"):
        reader = pypdf.PdfReader(str(pdf_path))
        for i, page in enumerate(reader.pages):
            docs.append(Document(
                page_content=page.extract_text() or "",
                metadata={"source": str(pdf_path), "page": i, "type": "pdf"},
            ))

    # 加载文本文件
    for txt_path in base.glob("**/*.txt"):
        docs.append(Document(
            page_content=txt_path.read_text(encoding="utf-8"),
            metadata={"source": str(txt_path), "type": "txt"},
        ))

    return docs

raw_docs = load_all_docs("./knowledge_base/")
print(f"Loaded {len(raw_docs)} documents")

# 2. 分割文档
splitter = RecursiveCharacterTextSplitter(
    chunk_size=500,
    chunk_overlap=50,
    separators=["\n\n", "\n", "。", "，", " "],
)
split_docs = splitter.split_documents(raw_docs)
print(f"Split into {len(split_docs)} chunks")

# 3. 查看结果
for doc in split_docs[:3]:
    print(f"[{doc.metadata['source']}:{doc.metadata.get('page', '?')}]")
    print(f"  {doc.page_content[:80]}...")

# 4. 后续：嵌入 → 存入向量数据库
```

## 常见陷阱

### 陷阱 1：编码问题

```python
# ❌ 中文文件默认 UTF-8 读取 GBK 文件会报错
content = Path("./data.txt").read_text()  # UnicodeDecodeError

# ✅ 指定编码或自动检测
content = Path("./data.txt").read_text(encoding="gbk")

# 或写一个自动检测函数
def read_text_auto(file_path: str) -> str:
    for enc in ("utf-8", "gbk", "gb2312", "latin-1"):
        try:
            return Path(file_path).read_text(encoding=enc)
        except UnicodeDecodeError:
            continue
    raise ValueError(f"无法解码: {file_path}")
```

### 陷阱 2：大文件 OOM

```python
# ❌ 10GB 的日志文件直接 read_text() 会撑爆内存
content = Path("./huge_log.txt").read_text()  # MemoryError!

# ✅ 逐行读取
with open("./huge_log.txt", "r", encoding="utf-8") as f:
    for line in f:
        process_line(line)  # 每次只处理一行
```

### 陷阱 3：PDF 表格丢失

```python
# ❌ pypdf 提取表格时格式可能错乱
reader = pypdf.PdfReader("report.pdf")
text = reader.pages[0].extract_text()
# 表格内容变成纯文本，行列关系丢失

# ✅ 用 pdfplumber 保留表格结构
import pdfplumber
with pdfplumber.open("report.pdf") as pdf:
    page = pdf.pages[0]
    tables = page.extract_tables()  # 返回二维列表
    text = page.extract_text()
```

### 陷阱 4：网页加载被反爬拦截

```python
# ❌ 默认请求头可能被网站拦截
resp = requests.get("https://example.com")  # 返回 403

# ✅ 自定义请求头
resp = requests.get("https://example.com", headers={
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
    "Accept": "text/html,application/xhtml+xml",
    "Accept-Language": "zh-CN,zh;q=0.9",
}, timeout=30)
```

### 陷阱 5：metadata 中 source 路径不一致

```python
# 建议：统一使用相对路径或在加载后标准化
for doc in docs:
    doc.metadata["source"] = Path(doc.metadata["source"]).name  # 只保留文件名
```

## 自定义加载器

当需要复用的加载逻辑时，可以封装为函数：

```python
from langchain_core.documents import Document
import sqlite3

def load_from_database(
    db_path: str,
    query: str,
    content_column: str = "content",
) -> list[Document]:
    """从数据库加载文档"""
    conn = sqlite3.connect(db_path)
    cursor = conn.execute(query)
    docs = []
    for row in cursor:
        docs.append(Document(
            page_content=str(row[0]),
            metadata={"source": "database", "id": row[1][[| if len(row) > 1 else None},
        ))
    conn.close()
    return docs

# 使用
docs = load_from_database(
    db_path="./data.db",
    query="SELECT content, id FROM articles WHERE status='published'",
)
```
