# Checkpointer 检索流程示例

以下代码演示 SqliteSaver 的完整工作流程，注释中说明了每一步框架内部发生的事。

```python
import sqlite3  
import os  
from dotenv import load_dotenv  
from langchain_deepseek import ChatDeepSeek  
from langgraph.checkpoint.sqlite import SqliteSaver  
from langchain.agents import create_agent  

# ============================================================
# 初始化 Checkpointer
# ============================================================
# SqliteSaver 封装了一个 sqlite3 连接。
# 调用 setup() 时，它会在数据库中自动创建 checkpoints 表：
#
#   CREATE TABLE checkpoints (
#       thread_id     TEXT NOT NULL,
#       checkpoint_id TEXT NOT NULL,
#       parent_id     TEXT,
#       checkpoint    TEXT NOT NULL,   -- JSON 格式的状态快照
#       metadata      TEXT NOT NULL,   -- JSON 格式的元数据
#       created_at    TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
#       PRIMARY KEY (thread_id, checkpoint_id)
#   );
#
connection = sqlite3.connect("resources/test.db", check_same_thread=False)  
checkpointer = SqliteSaver(connection)  
checkpointer.setup()  

load_dotenv()  
dsapi = os.getenv("DEEPSEEK_API_KEY")  

llm = ChatDeepSeek(  
    model="deepseek-v4-flash",  
    api_key=dsapi,  
    temperature=1.0,  
)  

system_prompt = (  
    "你是一个通用助手。\n"  
    "回答要求：\n"  
    "- 用中文回答，语气简洁友好\n"  
    "- 只使用纯文本，禁止任何 Markdown 标记和特殊符号\n"  
    "- 回答简洁"  
)  

# ============================================================
# 注册 Checkpointer 到 Agent
# ============================================================
# 这一步将 checkpointer 注入 Agent 框架内部。
# 之后每次 agent.invoke()，框架会自动：
#   1. 调用 checkpointer.get(config)        ← 从数据库读历史
#   2. 拼接待发送给 LLM 的 messages
#   3. 调用 LLM
#   4. 调用 checkpointer.put(config, 新状态) ← 写回数据库
#
agent = create_agent(  
    model=llm,  
    system_prompt=system_prompt,  
    checkpointer=checkpointer,  
)  

config = {"configurable": {"thread_id": "thread_2"}}  
print("初始化完成")
from langchain_core.messages import HumanMessage  

def chat(user_input):  
    """
    发送消息并打印 AI 回复。
    
    每次调用时，agent.invoke() 内部发生：
    
    ┌──────────────────────────────────────────────────────┐
    │  agent.invoke() 开始                                  │
    │                                                      │
    │  第 1 步：checkpointer.get(config)                    │
    │     ↓                                                │
    │   SqliteSaver.get() 内部：                            │
    │     conn.execute("""                                 │
    │       SELECT checkpoint FROM checkpoints              │
    │       WHERE thread_id = ?                             │
    │       ORDER BY checkpoint_id DESC LIMIT 1             │
    │     """, (config["configurable"]["thread_id"],))      │
    │     ↓                                                │
    │   有历史 → 反序列化 JSON → 返回 AgentState            │
    │   无历史 → 返回 None                                  │
    │                                                      │
    │  第 2 步：拼接 messages                                │
    │     get() 返回的状态 + 本轮输入 → 发给 LLM             │
    │                                                      │
    │  第 3 步：LLM 返回响应                                 │
    │                                                      │
    │  第 4 步：checkpointer.put(config, 新状态)             │
    │     ↓                                                │
    │   SqliteSaver.put() 内部：                            │
    │     conn.execute("""                                 │
    │       INSERT INTO checkpoints                         │
    │       (thread_id, checkpoint_id, parent_id,           │
    │        checkpoint, metadata)                          │
    │       VALUES (?, ?, ?, ?, ?)                          │
    │     """, (                                            │
    │       thread_id,                                      │
    │       新 checkpoint_id,                                │
    │       上一个 checkpoint_id,  -- 用于时间旅行           │
    │       json.dumps(状态),                                │
    │       json.dumps(metadata),                           │
    │     ))                                                │
    │     connection.commit()                               │
    │                                                      │
    └──────────────────────────────────────────────────────┘
    """
    print(f"用户：{user_input}")  
    resp = agent.invoke({"messages": [HumanMessage(content=user_input)]}, config)  
    print(f"助手：{resp['messages'][-1].content}\n")  


# ============================================================
# 第一次调用：thread_2 无历史 → get 返回 None
# ============================================================
# get 执行 SQL：
#   SELECT checkpoint FROM checkpoints
#   WHERE thread_id = 'thread_2'
#   ORDER BY checkpoint_id DESC LIMIT 1
# → 结果为空，返回 None
#
# LLM 只收到当前这一条 HumanMessage，无历史。
#
# 调用结束后 put 执行 SQL：
#   INSERT INTO checkpoints (thread_id, checkpoint_id, ...)
#   VALUES ('thread_2', 'step_1', NULL, '{...}', '{"step": 1}')
#
chat("你好，我叫张飞行")


# ============================================================
# 第二次调用：同一 thread_id → get 读到历史
# ============================================================
# 假设后续再调 chat("我叫什么？")：
#
# get 执行 SQL：
#   SELECT checkpoint FROM checkpoints
#   WHERE thread_id = 'thread_2'
#   ORDER BY checkpoint_id DESC LIMIT 1
# → 返回上一轮的状态，包含：
#   HumanMessage("你好，我叫张飞行")
#   AIMessage("你好张飞行...")
#
# 框架把这条历史拼入 messages，LLM 看到完整上下文
# → 能回答"你叫张飞行"
#
# chat("我叫什么？")


# ============================================================
# 直接查看数据库
# ============================================================
# 用 sqlite3 命令行可以查看已存储的检查点：
#
#   $ sqlite3 resources/test.db
#   sqlite> SELECT thread_id, checkpoint_id, length(checkpoint)
#           FROM checkpoints;
#   thread_2 | step_1 | 241
#
# 每条记录就是一个检查点，thread_id 相同的记录构成一条对话历史链。
