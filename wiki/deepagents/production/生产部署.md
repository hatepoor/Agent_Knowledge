# DeepAgents 生产部署

将 Deep Agent 从原型推向生产环境的完整方案。

## 一、完整生产骨架

```python
# main.py — 生产就绪的 Deep Agent API
import os
import json
import sqlite3
from contextlib import asynccontextmanager
from pathlib import Path

from fastapi import FastAPI, Depends, Header
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from langchain_core.tools import tool

from deepagents import create_deep_agent
from deepagents.backends import CompositeBackend, StateBackend, StoreBackend
from deepagents.middleware import FilesystemPermission

from langchain_deepseek import ChatDeepSeek
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.store.memory import InMemoryStore

# ── 配置 ──
class Settings(BaseModel):
    model_name: str = "deepseek-v4-flash"
    api_key: str = ""
    db_path: str = "./data/sessions.db"
    skills_dir: str = "./skills"
    memory_file: str = "./AGENTS.md"

settings = Settings()

# ── 自定义工具（生产环境应从 tools/ 目录导入）──
@tool
def internet_search(query: str, max_results: int = 5) -> str:
    """搜索互联网。参数: query=搜索关键词"""
    return f"[搜索] {query} 的模拟结果"

@tool
def custom_tool(data: str) -> str:
    """自定义工具示例。参数: data=输入数据"""
    return f"处理结果: {data}"

@tool
def send_email(to: str, subject: str, body: str) -> str:
    """发送邮件。参数: to=收件人, subject=主题, body=正文"""
    return f"[邮件已发送] {to}: {subject}"

# ── 生命周期 ──
agent = None
checkpointer = None
store = None

@asynccontextmanager
async def lifespan(app: FastAPI):
    global agent, checkpointer, store

    # Checkpointer
    Path("./data").mkdir(exist_ok=True)
    conn = sqlite3.connect(settings.db_path, check_same_thread=False)
    checkpointer = SqliteSaver(conn)
    checkpointer.setup()

    # Store（长期记忆）
    store = InMemoryStore()

    # LLM
    main_llm = ChatDeepSeek(model=settings.model_name, api_key=settings.api_key)
    summary_llm = ChatDeepSeek(model=settings.model_name, api_key=settings.api_key, temperature=0)

    # Deep Agent
    agent = create_deep_agent(
        model=main_llm,
        tools=[internet_search, custom_tool, send_email],
        system_prompt="你是 ACME 技术助手。用中文回答，引用来源。",
        backend=CompositeBackend(
            default=StateBackend(),
            routes={"/memories/": StoreBackend()},  # store 已通过下方 store= 参数传入
        ),
        memory=[settings.memory_file],
        skills=["code_review", "report_writing"],
        permissions=[
            FilesystemPermission(path="/workspace", access="read_write"),
            FilesystemPermission(path="/memories", access="read_write"),
        ],
        interrupt_on={"send_email": True},
        checkpointer=checkpointer,
        store=store,
    )
    yield
    conn.close()

app = FastAPI(lifespan=lifespan)
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

# ── API ──
class ChatRequest(BaseModel):
    message: str
    thread_id: str = "default"

@app.post("/chat")
async def chat(req: ChatRequest):
    result = await agent.ainvoke(
        {"messages": [{"role": "user", "content": req.message}]},
        config={"configurable": {"thread_id": req.thread_id}},
    )
    return {"reply": result["messages"][-1].content}

@app.post("/chat/stream")
async def chat_stream(req: ChatRequest):
    async def generator():
        async for event in agent.astream_events(
            {"messages": [{"role": "user", "content": req.message}]},
            config={"configurable": {"thread_id": req.thread_id}},
            version="v3",
        ):
            yield f"data: {json.dumps({'event': event['event']})}\n\n"
    return StreamingResponse(generator(), media_type="text/event-stream")

@app.get("/health")
async def health():
    return {"status": "ok"}
```

启动：

```bash
python -m uvicorn main:app --host 0.0.0.0 --port 8000 --workers 2
```

## 二、生产 Checklist

| 维度 | 检查项 |
|------|--------|
| ✅ 持久化 | SqliteSaver（开发）/ PostgresSaver（生产） |
| ✅ 长期记忆 | StoreBackend 跨线程共享 |
| ✅ 中间件 | HITL 审批敏感操作 |
| ✅ 沙箱 | 执行代码用 Sandbox backend |
| ✅ 日志 | structlog JSON 格式 |
| ✅ 监控 | Prometheus metrics |
| ✅ 认证 | Bearer Token / OAuth2 |
| ✅ CORS | 限制前端域名 |
| ✅ 超时 | Agent 层 120s + HTTP 层 130s |
