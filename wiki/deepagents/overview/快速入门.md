# DeepAgents 快速入门

从安装到第一个 Deep Agent 运行。

## 一、安装

```bash
# uv（推荐）
uv add deepagents

# pip
pip install deepagents
```

## 二、最小 Agent

```python
from deepagents import create_deep_agent

# 第1步：创建 Agent（只需要一个 model，其余全默认）
agent = create_deep_agent(model="claude-sonnet-4-6")

# 第2步：调用
result = agent.invoke({
    "messages": [{"role": "user", "content": "写一篇关于 LangGraph 的研究报告并保存到 /report.md"}]
})

# 第3步：查看结果
print(result["messages"][-1].content)
```

**这个 Agent 已经自带了**：文件系统（ls/read/write/edit）、任务规划（write_todos）、子代理（task）、上下文压缩。

## 三、带自定义工具的 Agent

```python
from deepagents import create_deep_agent
from tavily import TavilyClient
import os

tavily_client = TavilyClient(api_key=os.environ["TAVILY_API_KEY"])

def internet_search(query: str, max_results: int = 5):
    """搜索互联网"""
    return tavily_client.search(query, max_results=max_results)

agent = create_deep_agent(
    model="deepseek-v4-flash",
    tools=[internet_search],          # 自定义工具 → 和内置工具合并
    system_prompt="你是研究员。先分解任务，再搜索，最后写报告。",
)

result = agent.invoke({
    "messages": [{"role": "user", "content": "研究 LangChain 的最新进展并写报告"}]
})
```

## 四、流式调用

```python
for event in agent.stream_events(
    {"messages": [{"role": "user", "content": "写一个 Python 脚本计算 1 到 100 的和"}]},
    version="v3",
):
    kind = event.get("event")
    if kind == "on_chat_model_stream":
        print(event["data"]["chunk"].content, end="", flush=True)
    elif kind == "on_tool_start":
        print(f"\n[工具] {event['name']}")
```

## 五、带文件权限的 Agent

```python
from deepagents import create_deep_agent
from deepagents.middleware import FilesystemPermission

agent = create_deep_agent(
    model="deepseek-v4-flash",
    permissions=[
        FilesystemPermission(path="/reports", access="read_write"),
        FilesystemPermission(path="/tmp", access="read_write"),
        # 其他路径默认不可访问
    ],
)
```

## 六、带 HITL 的 Agent

```python
agent = create_deep_agent(
    model="deepseek-v4-flash",
    interrupt_on={
        "edit_file": True,    # 编辑文件前需审批
        "write_file": True,   # 写文件前需审批
        "execute": True,      # 执行命令前需审批
    },
)
```

## 七、带沙箱的 Agent

```python
from deepagents import create_deep_agent
from deepagents.backends.langsmith import LangSmithSandbox

agent = create_deep_agent(
    model="deepseek-v4-flash",
    backend=LangSmithSandbox(sandbox=sandbox_instance),  # 沙箱后端
    # Agent 自动获得 execute 工具，在沙箱中执行 shell 命令
)
```
