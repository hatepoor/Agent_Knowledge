# 测试模式

为 LangGraph 应用编写测试是确保质量的关键环节。本教程涵盖从单元测试到集成测试的完整测试策略，包括如何 Mock LLM 和工具、如何使用 `InMemorySaver` 构建测试夹具、以及如何测试人机协作工作流。

## 测试策略概览

```
测试金字塔：
         ╱  E2E 测试  ╲          ← 少量：完整流程测试
        ╱  集成测试      ╲        ← 中量：图级别测试
       ╱  单元测试          ╲     ← 大量：节点级别测试
      ╱══════════════════════╲
```

| 测试层级 | 测试对象 | 速度 | 依赖 |
|----------|----------|------|------|
| 单元测试 | 单个节点函数 | 快 | Mock 外部依赖 |
| 集成测试 | 完整的图执行 | 中 | 内存 Checkpointer |
| E2E 测试 | 真实 API 调用 | 慢 | 真实 LLM 和工具 |

## 一、单元测试：测试单个节点

### 1.1 基本节点测试

```python
# test_nodes.py
import pytest
from typing import TypedDict

# 假设这是你的节点代码
class State(TypedDict):
    question: str
    answer: str
    word_count: int

def count_words_node(state: State) -> dict:
    """统计字数的节点"""
    text = state["question"]
    count = len(text.split())
    return {"word_count": count}

def format_answer_node(state: State) -> dict:
    """格式化回答的节点"""
    answer = state.get("answer", "")
    if not answer:
        return {"answer": "暂无回答"}
    return {"answer": f"【回答】{answer}"}

# ===== 测试代码 =====

class TestCountWordsNode:
    """字数统计节点测试"""

    def test_normal_text(self):
        """测试正常文本"""
        state = {"question": "hello world foo bar", "answer": "", "word_count": 0}
        result = count_words_node(state)
        assert result["word_count"] == 4

    def test_empty_text(self):
        """测试空文本"""
        state = {"question": "", "answer": "", "word_count": 0}
        result = count_words_node(state)
        assert result["word_count"] == 0

    def test_single_word(self):
        """测试单个词"""
        state = {"question": "hello", "answer": "", "word_count": 0}
        result = count_words_node(state)
        assert result["word_count"] == 1

    def test_chinese_text(self):
        """测试中文文本"""
        state = {"question": "你好 世界 测试", "answer": "", "word_count": 0}
        result = count_words_node(state)
        assert result["word_count"] == 3

    def test_extra_whitespace(self):
        """测试多余空格"""
        state = {"question": "  hello   world  ", "answer": "", "word_count": 0}
        result = count_words_node(state)
        assert result["word_count"] == 2

        """测试空回答"""
        state = {"question": "", "answer": "", "word_count": 0}
        result = format_answer_node(state)
        assert result["answer"] == "暂无回答"

    def test_missing_answer_key(self):
```

```python
# test_with_mock_llm.py
import pytest
from unittest.mock import patch, MagicMock
from langchain_core.messages import AIMessage, HumanMessage
from typing import TypedDict

class State(TypedDict):
    messages: list
    answer: str

def llm_node(state: State) -> dict:
    """调用 LLM 的节点"""
    from langchain_openai import ChatOpenAI
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
    response = llm.invoke(state["messages"])
    return {"answer": response.content}

class TestLLMNode:
    """LLM 节点测试"""

    @patch("test_with_mock_llm.ChatOpenAI")
    def test_llm_response(self, mock_chat_class):
        """测试 LLM 响应（Mock）"""
        # 设置 Mock 返回值
        mock_llm = MagicMock()
        mock_llm.invoke.return_value = AIMessage(content="这是 Mock 回答")
        mock_chat_class.return_value = mock_llm

        # 执行节点
        state = {
            "messages": [HumanMessage(content="你好")],
            "answer": ""
        }
        result = llm_node(state)

        # 验证结果
        assert result["answer"] == "这是 Mock 回答"
        mock_llm.invoke.assert_called_once()

    @patch("test_with_mock_llm.ChatOpenAI")
    def test_llm_error_handling(self, mock_chat_class):
        """测试 LLM 错误处理（Mock）"""
        mock_llm = MagicMock()
        mock_llm.invoke.side_effect = Exception("API 错误")
        mock_chat_class.return_value = mock_llm

        state = {
            "messages": [HumanMessage(content="你好")],
            "answer": ""
        }
```

```python
# test_graph_integration.py
import pytest
from typing import TypedDict
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver

class State(TypedDict):
    question: str
    answer: str
    step: str

def ask_node(state: State) -> dict:
    """提问节点"""
    return {"answer": f"回答: {state['question']}", "step": "asked"}

def process_node(state: State) -> dict:
    """处理节点"""
    return {"answer": f"已处理: {state['answer']}", "step": "processed"}

def build_test_graph():
    """构建测试用图"""
    builder = StateGraph(State)
    builder.add_node("ask", ask_node)
    builder.add_node("process", process_node)
    builder.add_edge(START, "ask")
    builder.add_edge("ask", "process")
    builder.add_edge("process", END)
    return builder.compile()

class TestGraphIntegration:
    """图集成测试"""

    def test_full_execution(self):
        """测试完整执行流程"""
        graph = build_test_graph()
        result = graph.invoke({"question": "什么是 AI?", "answer": "", "step": ""})

        assert result["step"] == "processed"
        assert "已处理" in result["answer"]
        assert "回答" in result["answer"]

    def test_stream_execution(self):
        """测试流式执行"""
        graph = build_test_graph()
        events = list(graph.stream(
            {"question": "你好", "answer": "", "step": ""},
            stream_mode="updates"
        ))

        # 验证执行顺序
        assert len(events) == 2
        assert "ask" in events[0]
        assert "process" in events[1]

    def test_state_progression(self):
        """测试状态变化"""
        graph = build_test_graph()

        events = list(graph.stream(
            {"question": "测试", "answer": "", "step": ""},
            stream_mode="values"
        ))

        # 验证状态逐步变化
        assert events[-1]["step"] == "processed"
        assert "测试" in events[-1]["answer"]
```

```python
# test_routing.py
from typing import TypedDict
from langgraph.graph import StateGraph, START, END

class State(TypedDict):
    input_text: str
    category: str
    result: str

def classify_node(state: State) -> dict:
    """分类节点"""
    text = state["input_text"]
    if "?" in text:
        return {"category": "question"}
    elif len(text) > 20:
        return {"category": "long_text"}
    else:
        return {"category": "short_text"}

def handle_question(state: State) -> dict:
    return {"result": "这是一个问题"}

def handle_long(state: State) -> dict:
    return {"result": "这是长文本"}

def handle_short(state: State) -> dict:
    return {"result": "这是短文本"}

def route_by_category(state: State) -> str:
    return state["category"]

def build_routing_graph():
    builder = StateGraph(State)
    builder.add_node("classify", classify_node)
    builder.add_node("question", handle_question)
    builder.add_node("long", handle_long)
    builder.add_node("short", handle_short)

    builder.add_edge(START, "classify")
    builder.add_conditional_edges("classify", route_by_category, {
        "question": "question",
        "long_text": "long",
        "short_text": "short",
    })
    builder.add_edge("question", END)
    builder.add_edge("long", END)
    builder.add_edge("short", END)

    return builder.compile()

class TestRouting:
    """路由测试"""

    def test_question_routing(self):
        """测试问题路由"""
        graph = build_routing_graph()
        result = graph.invoke({"input_text": "什么是 AI?", "category": "", "result": ""})
        assert result["category"] == "question"
        assert result["result"] == "这是一个问题"

    def test_long_text_routing(self):
        """测试长文本路由"""
        graph = build_routing_graph()
        long_text = "这是一段非常非常长的文本，用来测试长文本的路由功能是否正确工作"
        result = graph.invoke({"input_text": long_text, "category": "", "result": ""})
        assert result["category"] == "long_text"
        assert result["result"] == "这是长文本"

    def test_short_text_routing(self):
        """测试短文本路由"""
        graph = build_routing_graph()
        result = graph.invoke({"input_text": "你好", "category": "", "result": ""})
        assert result["category"] == "short_text"
        assert result["result"] == "这是短文本"

    @pytest.mark.parametrize("input_text,expected_category,expected_result", [
        ("你好吗?", "question", "这是一个问题"),
        ("这是一个很长很长的文本内容用来测试路由功能", "long_text", "这是长文本"),
        ("嗨", "short_text", "这是短文本"),
    ])
    def test_routing_parametrized(self, input_text, expected_category, expected_result):
        """参数化路由测试"""
        graph = build_routing_graph()
        result = graph.invoke({"input_text": input_text, "category": "", "result": ""})
        assert result["category"] == expected_category
        assert result["result"] == expected_result
```

## 五、测试人机协作工作流

### 5.1 测试 interrupt/resume

```python
# test_human_in_loop.py
import pytest
from typing import TypedDict
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver
from langgraph.types import interrupt, Command

class ReviewState(TypedDict):
    content: str
    review_result: str
    approved: bool

def draft_node(state: ReviewState) -> dict:
    """起草节点"""
    return {"content": f"草稿: {state['content']}"}

def review_node(state: ReviewState) -> Command:
    """审核节点：需要人工介入"""
    feedback = interrupt("请审核内容")
    return {"review_result": feedback, "approved": feedback == "approved"}

def publish_node(state: ReviewState) -> dict:
    """发布节点"""
    return {"content": f"已发布: {state['content']}"}

def reject_node(state: ReviewState) -> dict:
    """拒绝节点"""
    return {"content": f"已拒绝: {state['content']}"}

def route_after_review(state: ReviewState) -> str:
    """审核后路由"""
    if state.get("approved"):
        return "publish"
    return "reject"

def build_review_graph():
    """构建审核工作流图"""
    builder = StateGraph(ReviewState)
    builder.add_node("draft", draft_node)
    builder.add_node("review", review_node)
    builder.add_node("publish", publish_node)
    builder.add_node("reject", reject_node)

    builder.add_edge(START, "draft")
    builder.add_edge("draft", "review")
    builder.add_conditional_edges("review", route_after_review, {
        "publish": "publish",
        "reject": "reject",
    })
    builder.add_edge("publish", END)
    builder.add_edge("reject", END)

    checkpointer = MemorySaver()
    return builder.compile(
        checkpointer=checkpointer,
        interrupt_before=["review"]
    )

class TestHumanInLoop:
    """人机协作测试"""

    def test_interrupt_before_review(self):
        """测试在审核前中断"""
```

## 七、完整测试示例

```python
"""
完整测试示例：测试一个带工具调用的 Agent
"""
import pytest
from typing import TypedDict, Annotated
from unittest.mock import MagicMock, patch
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langgraph.checkpoint.memory import MemorySaver

# ===== 被测代码 =====

class AgentState(TypedDict):
    messages: Annotated[list, add_messages]
    tool_called: bool

def agent_node(state: AgentState) -> dict:
    """Agent 节点"""
    from langchain_openai import ChatOpenAI
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
    response = llm.invoke(state["messages"])
    return {"messages": [response]}

def tool_node(state: AgentState) -> dict:
    """工具节点"""
    last_message = state["messages"][-1]
    # 模拟工具执行
    result = f"工具结果: 处理了 {last_message.content}"
    return {"messages": [ToolMessage(content=result, tool_call_id="1")], "tool_called": True}

def should_use_tool(state: AgentState) -> str:
    """判断是否需要使用工具"""
    last = state["messages"][-1]
    if hasattr(last, "tool_calls") and last.tool_calls:
        return "tool"
    return "end"

# ===== 测试代码 =====

class TestAgent:
    """Agent 完整测试"""

    @pytest.fixture
    def agent_graph(self):
        """创建 Agent 图"""
        builder = StateGraph(AgentState)
        builder.add_node("agent", agent_node)
        builder.add_node("tool", tool_node)

        builder.add_edge(START, "agent")
        builder.add_conditional_edges("agent", should_use_tool, {
            "tool": "tool",
            "end": END,
        })
        builder.add_edge("tool", "agent")

        checkpointer = MemorySaver()
        return builder.compile(checkpointer=checkpointer)

    @patch("langchain_openai.ChatOpenAI")
    def test_agent_without_tool_call(self, mock_chat_cls, agent_graph):
        """测试 Agent 不调用工具的情况"""
        # Mock LLM 直接返回回答
        mock_llm = MagicMock()
        mock_llm.invoke.return_value = AIMessage(content="直接回答")
        mock_chat_cls.return_value = mock_llm

        config = {"configurable": {"thread_id": "test-no-tool"}}
        result = agent_graph.invoke(
            {"messages": [HumanMessage(content="你好")], "tool_called": False},
            config=config,
        )

        assert result["messages"][-1].content == "直接回答"
        assert result["tool_called"] is False

    @patch("langchain_openai.ChatOpenAI")
    def test_agent_with_tool_call(self, mock_chat_cls, agent_graph):
        """测试 Agent 调用工具的情况"""
        # Mock LLM：第一次返回工具调用，第二次返回最终回答
        mock_llm = MagicMock()
        tool_call_msg = AIMessage(
            content="",
            tool_calls=[{"name": "search", "args": {"query": "test"}, "id": "1"}]
        )
        final_msg = AIMessage(content="工具调用后的回答")
        mock_llm.invoke.side_effect = [tool_call_msg, final_msg]
        mock_chat_cls.return_value = mock_llm

        config = {"configurable": {"thread_id": "test-with-tool"}}
        result = agent_graph.invoke(
            {"messages": [HumanMessage(content="搜索一下")], "tool_called": False},
            config=config,
        )

        assert result["tool_called"] is True
        assert result["messages"][-1].content == "工具调用后的回答"

    def test_state_isolation(self, agent_graph):
        """测试不同线程的状态隔离"""
        config1 = {"configurable": {"thread_id": "isolation-1"}}
        config2 = {"configurable": {"thread_id": "isolation-2"}}

        with patch("langchain_openai.ChatOpenAI") as mock_cls:
            mock_llm = MagicMock()
            mock_llm.invoke.return_value = AIMessage(content="回答")
            mock_cls.return_value = mock_llm

            agent_graph.invoke(
                {"messages": [HumanMessage(content="线程1")], "tool_called": False},
                config=config1,
            )
            agent_graph.invoke(
                {"messages": [HumanMessage(content="线程2")], "tool_called": False},
                config=config2,
            )

        state1 = agent_graph.get_state(config1)
        state2 = agent_graph.get_state(config2)

        # 验证状态隔离
        msgs1 = [m.content for m in state1.values["messages"]]
        msgs2 = [m.content for m in state2.values["messages"]]

        assert "线程1" in msgs1
        assert "线程2" not in msgs1
        assert "线程2" in msgs2
        assert "线程1" not in msgs2
```

## 总结

| 测试类型 | 工具 | 说明 |
|----------|------|------|
| 节点单元测试 | `pytest` | 直接调用节点函数 |
| Mock LLM | `unittest.mock` / `FakeListLLM` | 避免真实 API 调用 |
| Mock 工具 | `unittest.mock.patch` | 模拟工具返回 |
| 图集成测试 | `graph.invoke()` + `MemorySaver` | 测试完整执行流程 |
| 路由测试 | `@pytest.mark.parametrize` | 测试不同路由分支 |
| 人机协作测试 | `interrupt` + `Command(resume=...)` | 测试中断与恢复 |
| 时间旅行测试 | `get_state_history` + `update_state` | 测试状态回退 |
| 快照测试 | `syrupy` | 防止输出意外变化 |
