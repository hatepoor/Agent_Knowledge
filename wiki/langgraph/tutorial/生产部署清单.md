# 生产部署清单

将 LangGraph 应用从开发环境迁移到生产环境需要系统性地考虑配置管理、持久化、性能优化、监控告警、安全防护和容器化部署等方面。本清单提供完整的检查项和代码示例，帮助你平稳上线。

## 部署前总览

```
开发环境                    生产环境
─────────                  ─────────
InMemorySaver       →      PostgreSQL Checkpointer
本地 API Key        →      环境变量 / 密钥管理
单进程运行          →      多 Worker + 负载均衡
无日志              →      结构化日志 + 集中收集
无监控              →      健康检查 + 指标监控
无限制              →      消息裁剪 + 速率限制
默认配置            →      连接池 + 超时配置
```

```python
# config_validation.py
"""
启动时验证配置是否正确
"""
import sys

def validate_config(settings: AppSettings) -> list[str]:
    """验证配置，返回警告列表"""
    warnings = []

    # 检查必填项
    if not settings.openai_api_key:
        raise ValueError("OPENAI_API_KEY 未设置")
    if not settings.database_url:
        raise ValueError("DATABASE_URL 未设置")

    # 检查安全警告
    if "localhost" in settings.database_url:
        warnings.append("警告: 生产环境不应使用 localhost 数据库")
    if settings.openai_api_key.startswith("sk-test"):
        warnings.append("警告: 似乎使用了测试 API Key")
    if settings.log_level == "DEBUG":
        warnings.append("警告: 生产环境不建议使用 DEBUG 日志级别")

    # 检查性能配置
    if settings.database_pool_size < 5:
        warnings.append("建议: 生产环境数据库连接池建议 >= 5")
    if settings.recursion_limit > 200:
        warnings.append("建议: recursion_limit 过高可能导致资源耗尽")

    return warnings

def startup_check():
    """启动前检查"""
    try:
        settings = AppSettings()
        warnings = validate_config(settings)

        for w in warnings:
            print(f"[配置警告] {w}", file=sys.stderr)

        print(f"[配置] 模型: {settings.openai_model}")
        print(f"[配置] 连接池: {settings.database_pool_size}")
        print(f"[配置] 日志级别: {settings.log_level}")
        print("[配置] 所有配置验证通过")

        return settings

    except Exception as e:
        print(f"[配置错误] {e}", file=sys.stderr)
        sys.exit(1)
```

## 二、选择正确的 Checkpointer

### 2.1 Checkpointer 选型指南

| Checkpointer | 适用场景 | 优点 | 缺点 |
|-------------|----------|------|------|
| `InMemorySaver` | 开发/测试 | 零配置、快速 | 重启丢失、不支持多进程 |
| `SqliteSaver` | 单机小规模 | 简单、无需额外服务 | 不支持并发写入 |
| `PostgresSaver` | 生产环境 | 高并发、持久化、可扩展 | 需要 PostgreSQL |
| `AsyncPostgresSaver` | 异步生产环境 | 异步、高并发 | 需要 PostgreSQL |

### 2.2 从 InMemorySaver 迁移到 PostgresSaver

```python
# checkpointer_setup.py
"""
生产环境 Checkpointer 配置
"""
from langgraph.checkpoint.postgres import PostgresSaver
from langgraph.checkpoint.postgres.aio import AsyncPostgresSaver
import psycopg2
from psycopg_pool import ConnectionPool

def create_sync_checkpointer(settings: AppSettings) -> PostgresSaver:
    """创建同步 PostgreSQL Checkpointer"""
    # 创建连接池
    pool = ConnectionPool(
        conninfo=settings.database_url,
        min_size=settings.database_pool_size,
        max_size=settings.database_pool_size + settings.database_max_overflow,
        timeout=settings.database_pool_timeout,
        # 连接健康检查
        check=ConnectionPool.check_connection,
    )

    # 创建 Checkpointer
    checkpointer = PostgresSaver(pool)

    # 初始化数据库表（首次部署时需要）
    checkpointer.setup()

    return checkpointer

async def create_async_checkpointer(settings: AppSettings) -> AsyncPostgresSaver:
    """创建异步 PostgreSQL Checkpointer"""
    from psycopg_pool import AsyncConnectionPool

    pool = AsyncConnectionPool(
        conninfo=settings.database_url,
        min_size=settings.database_pool_size,
        max_size=settings.database_pool_size + settings.database_max_overflow,
        timeout=settings.database_pool_timeout,
    )

    checkpointer = AsyncPostgresSaver(pool)
    await checkpointer.setup()

    return checkpointer
```

### 3.2 数据库连接池

```python
# db_pool.py
"""
数据库连接池管理
"""
from psycopg_pool import ConnectionPool
from contextlib import contextmanager
import threading

class DatabaseManager:
    """数据库管理器：线程安全的连接池"""

    def __init__(self, database_url: str, pool_size: int = 10, max_overflow: int = 20):
        self._pool = ConnectionPool(
            conninfo=database_url,
            min_size=pool_size,
            max_size=pool_size + max_overflow,
            timeout=30,
            # 连接存活检查
            check=ConnectionPool.check_connection,
            # 连接打开时的回调
            open=True,
        )
        self._lock = threading.Lock()

    @contextmanager
    def get_connection(self):
        """获取数据库连接（上下文管理器）"""
        conn = self._pool.getconn()
        try:
            yield conn
        except Exception:
            conn.rollback()
            raise
        finally:
            self._pool.putconn(conn)

    def close(self):
        """关闭连接池"""
        self._pool.close()

    def health_check(self) -> bool:
        """检查连接池健康状态"""
        try:
            with self.get_connection() as conn:
                with conn.cursor() as cur:
                    cur.execute("SELECT 1")
            return True
        except Exception:
            return False

# 全局数据库管理器
db_manager = None

def init_db(settings: AppSettings):
    """初始化数据库管理器"""
    global db_manager
    db_manager = DatabaseManager(
        database_url=settings.database_url,
        pool_size=settings.database_pool_size,
        max_overflow=settings.database_max_overflow,
    )
```

## 四、消息裁剪

### 4.1 为什么需要消息裁剪

长对话中，消息列表会不断增长，导致：
- Token 用量超出模型限制
- API 调用成本增加
- 响应延迟增大

### 4.2 消息裁剪策略

```python
# message_trimming.py
"""
消息裁剪：控制上下文长度
"""
from langchain_core.messages import SystemMessage, trim_messages
from langchain_openai import ChatOpenAI
from typing import Annotated
from typing_extensions import TypedDict
from langgraph.graph.message import add_messages

class State(TypedDict):
    messages: Annotated[list, add_messages]

def trim_messages_node(state: State) -> dict:
    """
    消息裁剪节点
    保留系统消息和最近的消息，裁剪中间部分
    """
    messages = state["messages"]

    # 分离系统消息和普通消息
    system_messages = [m for m in messages if isinstance(m, SystemMessage)]
    other_messages = [m for m in messages if not isinstance(m, SystemMessage)]

    # 裁剪普通消息
    trimmed = trim_messages(
        other_messages,
        max_tokens=4000,
        token_counter=ChatOpenAI(model="gpt-4o-mini"),
        allow_partial=False,     # 不允许部分消息（保持消息完整性）
        strategy="last",         # 保留最新的消息
        include_system=False,    # 系统消息单独处理
    )

    # 重新组合：系统消息 + 裁剪后的消息
    final_messages = system_messages + trimmed

    return {"messages": final_messages}
```

### 4.3 自定义裁剪策略

```python
# custom_trimming.py
"""
自定义消息裁剪策略
"""
from langchain_core.messages import BaseMessage, SystemMessage, HumanMessage

def smart_trim(messages: list[BaseMessage], max_messages: int = 20) -> list[BaseMessage]:
    """
    智能裁剪策略：
    1. 始终保留第一条系统消息
    2. 始终保留最后 N 条消息
    3. 中间的消息按重要性排序保留
    """
    if len(messages) <= max_messages:
        return messages

    # 保留系统消息
    system_msgs = [m for m in messages if isinstance(m, SystemMessage)]

    # 保留最近的 N 条非系统消息
    non_system = [m for m in messages if not isinstance(m, SystemMessage)]
    recent = non_system[-(max_messages - len(system_msgs)):]

    return system_msgs + recent

def summary_trim(messages: list[BaseMessage], max_tokens: int = 3000) -> list[BaseMessage]:
    """
    摘要裁剪策略：
    将旧消息总结为一条摘要消息
    """
    from langchain_openai import ChatOpenAI

    if len(messages) <= 5:
        return messages

    # 旧消息（需要总结的部分）
    old_messages = messages[:-5]
    recent_messages = messages[-5:]

    # 生成摘要
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
    old_text = "\n".join([f"{m.type}: {m.content}" for m in old_messages])

    summary_response = llm.invoke(
        f"请将以下对话历史总结为简洁的摘要（保留关键信息）：\n{old_text}"
    )

    # 创建摘要消息
    from langchain_core.messages import SystemMessage
    summary_msg = SystemMessage(content=f"[对话历史摘要] {summary_response.content}")

    return [summary_msg] + recent_messages
```

### 5.1 结构化日志

```python
# logging_setup.py
"""
生产环境结构化日志配置
"""
import logging
import json
import sys
from datetime import datetime

class JSONFormatter(logging.Formatter):
    """JSON 格式日志，便于日志收集系统解析"""

    def format(self, record):
        log_data = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno,
        }

        # 添加异常信息
        if record.exc_info:
            log_data["exception"] = self.formatException(record.exc_info)

        # 添加自定义字段
        if hasattr(record, "thread_id"):
            log_data["thread_id"] = record.thread_id
        if hasattr(record, "node_name"):
            log_data["node_name"] = record.node_name
        if hasattr(record, "duration_ms"):
            log_data["duration_ms"] = record.duration_ms

        return json.dumps(log_data, ensure_ascii=False)

def setup_logging(log_level: str = "INFO"):
    """配置生产环境日志"""
    handler = logging.StreamHandler(sys.stdout)
    handler.setFormatter(JSONFormatter())

    root_logger = logging.getLogger()
    root_logger.setLevel(getattr(logging, log_level.upper()))
    root_logger.addHandler(handler)

    # 降低第三方库日志级别
    logging.getLogger("httpx").setLevel(logging.WARNING)
    logging.getLogger("openai").setLevel(logging.WARNING)
    logging.getLogger("langchain").setLevel(logging.INFO)
    logging.getLogger("langgraph").setLevel(logging.INFO)

logger = logging.getLogger(__name__)
```

### 5.2 节点执行监控

```python
# monitoring.py
"""
节点执行监控装饰器
"""
import time
import functools
import logging

logger = logging.getLogger(__name__)

def monitor_node(node_func):
    """
    节点监控装饰器
    记录执行时间、成功/失败状态
    """
    @functools.wraps(node_func)
    def wrapper(state, *args, **kwargs):
        node_name = node_func.__name__
        start_time = time.perf_counter()

        logger.info(
            f"节点开始执行: {node_name}",
            extra={"node_name": node_name}
        )

        try:
            result = node_func(state, *args, **kwargs)

            duration_ms = (time.perf_counter() - start_time) * 1000
            logger.info(
                f"节点执行成功: {node_name}",
                extra={
                    "node_name": node_name,
                    "duration_ms": round(duration_ms, 2),
                }
            )

            return result

        except Exception as e:
            duration_ms = (time.perf_counter() - start_time) * 1000
            logger.error(
                f"节点执行失败: {node_name} - {e}",
                extra={
                    "node_name": node_name,
                    "duration_ms": round(duration_ms, 2),
                },
                exc_info=True,
            )
            raise

    return wrapper

# 使用示例
@monitor_node
def my_node(state: State) -> dict:
    """带监控的节点"""
    llm = get_llm_client()
    response = llm.invoke(state["messages"])
    return {"messages": [response]}
```

```python
    # 检查数据库连接

        llm = get_llm_client()

        checks["llm_api"] = f"error: {str(e)}"

    # 检查 Redis（如果使用）
    if settings.redis_url:
        try:
            import redis
            r = redis.from_url(settings.redis_url)
            r.ping()
            checks["redis"] = "ok"
        except Exception as e:
            checks["redis"] = f"error: {str(e)}"

    # 判断整体状态
    all_ok = all(v == "ok" for v in checks.values())
    status_code = 200 if all_ok else 503

    return Response(
        content=json.dumps({"status": "ready" if all_ok else "not_ready", "checks": checks}),
        status_code=status_code,
        media_type="application/json"
    )
```

```python
    # 编译图
    graph = build_graph(checkpointer)

    logger.info("正在关闭应用...")
    pool.close()

class ChatRequest(BaseModel):

    message: str
    thread_id: str

    """聊天响应模型"""
    reply: str
    thread_id: str

@app.post("/chat", response_model=ChatResponse)
async def chat(request: ChatRequest):
    """聊天接口"""
    try:
        config = {
            "configurable": {"thread_id": request.thread_id},
            "recursion_limit": settings.recursion_limit,
        }

        result = graph.invoke(
            {"messages": [HumanMessage(content=request.message)]},

        last_message = result["messages"][-1]
        reply = last_message.content if hasattr(last_message, "content") else str(last_message)

    except Exception as e:

        raise HTTPException(status_code=500, detail="处理失败，请稍后重试")

@app.get("/threads/{thread_id}/history")
async def get_thread_history(thread_id: str):
    """获取对话历史"""
    try:
        config = {"configurable": {"thread_id": thread_id}}
        state = graph.get_state(config)

        if state is None:
            raise HTTPException(status_code=404, detail="对话不存在")

        messages = [
            {"role": m.type, "content": m.content}
            for m in state.values.get("messages", [])
        ]

        return {"thread_id": thread_id, "messages": messages}

    except HTTPException:
        raise
    except Exception as e:
```

```python
# security.py
"""
安全最佳实践
"""
import re
import secrets

def validate_input(message: str) -> tuple[bool, str]:
    """
    输入验证：防止注入攻击
    """
    if not message or len(message.strip()) == 0:
        return False, "消息不能为空"

    if len(message) > 10000:
        return False, "消息长度不能超过 10000 字符"

    # 检测潜在的注入模式
    suspicious_patterns = [
        r"<script.*?>",           # XSS
        r"DROP\s+TABLE",          # SQL 注入
        r"eval\s*\(",             # 代码注入
        r"__import__",            # Python 注入
    ]

    for pattern in suspicious_patterns:
        if re.search(pattern, message, re.IGNORECASE):
            return False, "输入包含不允许的内容"

    return True, ""

def sanitize_thread_id(thread_id: str) -> str:
    """
    清理 thread_id：只允许字母、数字、连字符和下划线
    """
    if not re.match(r'^[a-zA-Z0-9_-]+$', thread_id):
        raise ValueError("thread_id 只能包含字母、数字、连字符和下划线")
    return thread_id

def generate_thread_id() -> str:
    """生成安全的 thread_id"""
    return secrets.token_urlsafe(16)
```

### 8.2 请求限流

```python
# rate_limiter.py
"""
简单的内存速率限制器
生产环境建议使用 Redis 实现分布式限流
"""
import time
from collections import defaultdict
from threading import Lock

class RateLimiter:
    """滑动窗口速率限制器"""

    def __init__(self, max_requests: int = 30, window_seconds: int = 60):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self._requests = defaultdict(list)
        self._lock = Lock()

    def is_allowed(self, client_id: str) -> bool:
        """检查是否允许请求"""
        now = time.time()
        cutoff = now - self.window_seconds

        with self._lock:
            # 清理过期记录
            self._requests[client_id] = [
                t for t in self._requests[client_id] if t > cutoff
            ]

            # 检查是否超限
            if len(self._requests[client_id]) >= self.max_requests:
                return False

            # 记录请求
            self._requests[client_id].append(now)
            return True

# 全局限流器
rate_limiter = RateLimiter(max_requests=30, window_seconds=60)
```
