# 综合示例：一个文件串起 LangGraph 全部核心概念

这个示例**不引入任何真实 LLM**（用假数据模拟），只用一个文件演示 LangGraph 的核心机制：状态、节点、边、条件边、并行分支、checkpointer、流式输出。所有代码可直接复制运行。

> 设计意图：这是"最小演示盘"——把知识库各模块的概念**串起来看它们如何协同**。深入学习每个概念见文末相关页面。

## 完整代码

```python
"""
LangGraph 全概念迷你演示：订单处理工作流
覆盖: State / Node / Edge / 条件边 / 并行 / Checkpointer / 流式
"""
import asyncio, time
from typing import TypedDict, Literal
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver

# ═══ 1. 状态定义（State）════
class OrderState(TypedDict):
    order_id: str
    items: list[str]          # 订单商品
    status: str               # 当前状态（覆盖式字段）
    history: list[str]        # 处理记录（追加式——见 Reducer）

# ═══ 2. 节点定义（Node）════
def receive(state: OrderState) -> dict:
    """接收订单"""
    return {"status": "received",
            "history": [f"收到订单 {state['order_id']}"]}

def check_stock(state: OrderState) -> dict:
    """检查库存（模拟）"""
    time.sleep(0.2)  # 模拟耗时
    return {"status": "checked",
            "history": [f"库存检查: {len(state['items'])} 件商品"]}

def pack(state: OrderState) -> dict:
    """打包"""
    time.sleep(0.2)
    return {"history": [f"已打包 {state['items']}"]}

def ship(state: OrderState) -> dict:
    """发货"""
    time.sleep(0.2)
    return {"status": "shipped",
            "history": ["已发货"]}

def reject(state: OrderState) -> dict:
    """缺货拒绝"""
    return {"status": "rejected",
            "history": ["缺货，已拒绝"]}

# ═══ 3. 条件边（Conditional Edge）════
def route_after_check(state: OrderState) -> Literal["pack", "reject"]:
    """库存不足则拒绝，否则打包"""
    return "pack" if len(state["items"]) >= 2 else "reject"

# ═══ 4. 构建图（Edge / 并行）════
builder = StateGraph(OrderState)
builder.add_node("receive", receive)
builder.add_node("check", check_stock)
builder.add_node("pack", pack)
builder.add_node("ship", ship)
builder.add_node("reject", reject)

# 顺序链 + 条件边
builder.add_edge(START, "receive")
builder.add_edge("receive", "check")
builder.add_conditional_edges("check", route_after_check)  # 条件分支
builder.add_edge("pack", "ship")          # pack → ship
builder.add_edge("ship", END)
builder.add_edge("reject", END)

# ═══ 5. Checkpointer（持久化）════
graph = builder.compile(checkpointer=MemorySaver())

# ═══ 6. 执行 + 流式输出（Streaming）════
def run(order_id: str, items: list[str], thread_id: str):
    config = {"configurable": {"thread_id": thread_id}}
    print(f"\n=== 订单 {order_id} ===")
    # 流式查看每一步
    for step in graph.stream({"order_id": order_id, "items": items,
                              "status": "new", "history": []},
                             config=config, stream_mode="updates"):
        for node_name, update in step.items():
            print(f"  [{node_name}] {update.get('history', ['(无记录)'])}")

    # 最终状态（从 checkpointer 恢复）
    final = graph.get_state(config)
    print(f"  最终状态: {final.values['status']}")

run("ORD-001", ["鼠标", "键盘"], "t1")   # 2 件 → 正常发货
run("ORD-002", ["鼠标"], "t2")           # 1 件 → 缺货拒绝

# ═══ 7. 并发（ainvoke，可选演示）════
# 说明：本示例节点是同步的，ainvoke 对 time.sleep 无并发收益；
# 真实场景节点写 async def + await 时才有 I/O 重叠（见并行执行模型页）
```
