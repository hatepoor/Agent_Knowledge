# 子图概述

子图（Subgraph）是 LangGraph 的模块化组合机制，允许你将一个图作为另一个图的节点使用。这种机制使得复杂的图可以分解为更小、更易管理的组件，提高代码的可重用性和可维护性。

## 核心概念

### 为什么使用子图？

- **模块化**：将复杂逻辑分解为独立的组件
- **可重用性**：同一个子图可以在多个地方使用
- **可维护性**：每个子图专注于特定功能
- **可测试性**：可以独立测试每个子图
- **团队协作**：不同团队可以开发不同的子图

### 基本思想

子图就像一个函数：
- 接收输入（状态）
- 执行一系列操作
- 返回输出（更新后的状态）

```python
# 子图就像一个黑盒函数
def subgraph(input_state) -> output_state:
    # 内部执行复杂的逻辑
    # 对外只暴露输入和输出
    return processed_state
```

## 基本用法

### 1. 创建子图

```python
from langgraph.graph import StateGraph, START, END
from typing_extensions import TypedDict

class SubgraphState(TypedDict):
    input_data: str
    processed_data: str
    result: str

# 创建子图
subgraph_builder = StateGraph(SubgraphState)

def process_node(state: SubgraphState):
    """处理节点"""
    processed = f"已处理: {state['input_data']}"
    return {"processed_data": processed}

def analyze_node(state: SubgraphState):
    """分析节点"""
    result = f"分析结果: {state['processed_data']}"
    return {"result": result}

subgraph_builder.add_node("process", process_node)
subgraph_builder.add_node("analyze", analyze_node)
subgraph_builder.add_edge(START, "process")
subgraph_builder.add_edge("process", "analyze")
subgraph_builder.add_edge("analyze", END)

# 编译子图
subgraph = subgraph_builder.compile()
```

### 2. 将子图添加为节点

```python
class ParentState(TypedDict):
    query: str
    subgraph_input: str
    subgraph_output: str
    final_result: str

# 创建父图
parent_builder = StateGraph(ParentState)

def prepare_node(state: ParentState):
    """准备子图输入"""
    return {"subgraph_input": f"准备: {state['query']}"}

def finalize_node(state: ParentState):
    """处理子图输出"""
    return {"final_result": f"最终: {state['subgraph_output']}"}

# 添加普通节点
parent_builder.add_node("prepare", prepare_node)
parent_builder.add_node("finalize", finalize_node)

# 添加子图作为节点
parent_builder.add_node("subgraph", subgraph)

# 设置边
parent_builder.add_edge(START, "prepare")
parent_builder.add_edge("prepare", "subgraph")
parent_builder.add_edge("subgraph", "finalize")
parent_builder.add_edge("finalize", END)

# 编译父图
parent_app = parent_builder.compile()
```

## 状态传递

### 1. 状态映射

子图和父图的状态可以不同，需要进行映射：

```python
class ParentState(TypedDict):
    query: str
    result: str

class SubgraphState(TypedDict):
    input_text: str
    output_text: str

# 创建子图
subgraph = create_subgraph()

# 创建父图，使用状态映射
parent_builder = StateGraph(ParentState)

# 添加子图节点，指定状态映射
parent_builder.add_node(
    "subgraph",
    subgraph,
    input_mapper=lambda state: {"input_text": state["query"]},
    output_mapper=lambda state: {"result": state["output_text"]}
)
```

### 2. 共享状态

子图和父图可以共享相同的状态结构：

```python
class SharedState(TypedDict):
    messages: list
    context: dict
    result: str

# 子图和父图使用相同的状态结构
subgraph = create_subgraph_with_state(SharedState)
parent = create_parent_with_state(SharedState)

# 状态自动传递，无需映射
parent.add_node("subgraph", subgraph)
```

## 完整示例

### 1. 简单的子图示例

```python
from langgraph.graph import StateGraph, START, END
from typing_extensions import TypedDict

# 子图状态
class SubgraphState(TypedDict):
    text: str
    word_count: int
    char_count: int

# 创建子图
def create_text_analysis_subgraph():
    """创建文本分析子图"""
    builder = StateGraph(SubgraphState)
    
    def count_words(state: SubgraphState):
        words = len(state["text"].split())
        return {"word_count": words}
    
    def count_chars(state: SubgraphState):
        chars = len(state["text"])
        return {"char_count": chars}
    
    builder.add_node("count_words", count_words)
    builder.add_node("count_chars", count_chars)
    
    builder.add_edge(START, "count_words")
    builder.add_edge("START", "count_chars")  # 并行执行
    builder.add_edge("count_words", END)
    builder.add_edge("count_chars", END)
    
    return builder.compile()

# 父图状态
class ParentState(TypedDict):
    input_text: str
    analysis_result: dict

# 创建父图
def create_parent_graph():
    """创建父图"""
    builder = StateGraph(ParentState)
    
    # 创建子图
    text_analysis = create_text_analysis_subgraph()
    
    def prepare_analysis(state: ParentState):
        return {"text": state["input_text"]}
    
    def compile_results(state: ParentState):
        return {
            "analysis_result": {
                "words": state.get("word_count", 0),
                "chars": state.get("char_count", 0)
            }
        }
    
    builder.add_node("prepare", prepare_analysis)
    builder.add_node("analyze", text_analysis)
    builder.add_node("compile", compile_results)
    
    builder.add_edge(START, "prepare")
    builder.add_edge("prepare", "analyze")
    builder.add_edge("analyze", "compile")
    builder.add_edge("compile", END)
    
    return builder.compile()

# 使用
app = create_parent_graph()
result = app.invoke({"input_text": "Hello World", "analysis_result": {}})
print(result)
```

### 2. 带状态映射的子图

```python
from langgraph.graph import StateGraph, START, END
from typing_extensions import TypedDict

# 子图状态
class ResearchState(TypedDict):
    query: str
    sources: list
    summary: str

# 父图状态
class MainState(TypedDict):
    user_question: str
    research_data: dict
    final_answer: str

# 创建研究子图
def create_research_subgraph():
    builder = StateGraph(ResearchState)
    
    def search_sources(state: ResearchState):
        # 模拟搜索
        sources = [f"来源1: {state['query']}", f"来源2: {state['query']}"]
        return {"sources": sources}
    
    def summarize(state: ResearchState):
        summary = f"关于 {state['query']} 的研究摘要"
        return {"summary": summary}
    
    builder.add_node("search", search_sources)
    builder.add_node("summarize", summarize)
    
    builder.add_edge(START, "search")
    builder.add_edge("search", "summarize")
    builder.add_edge("summarize", END)
    
    return builder.compile()

# 创建父图
def create_main_graph():
    builder = StateGraph(MainState)
    
    research_subgraph = create_research_subgraph()
    
    def prepare_research(state: MainState):
        # 映射状态：父图 -> 子图
        return {
            "query": state["user_question"],
            "sources": [],
            "summary": ""
        }
    
    def process_research_results(state: MainState):
        # 映射状态：子图 -> 父图
        return {
            "research_data": {
                "sources": state.get("sources", []),
                "summary": state.get("summary", "")
            }
        }
    
    def generate_answer(state: MainState):
        answer = f"基于研究数据生成答案: {state['research_data']}"
        return {"final_answer": answer}
    
    builder.add_node("prepare", prepare_research)
    builder.add_node("research", research_subgraph)
    builder.add_node("process", process_research_results)
    builder.add_node("answer", generate_answer)
    
    builder.add_edge(START, "prepare")
    builder.add_edge("prepare", "research")
    builder.add_edge("research", "process")
    builder.add_edge("process", "answer")
    builder.add_edge("answer", END)
    
    return builder.compile()
```

## 子图的优势

### 1. 代码重用

```python
# 创建通用的文本处理子图
text_processor = create_text_processing_subgraph()

# 在多个地方重用
graph1.add_node("process", text_processor)
graph2.add_node("process", text_processor)
graph3.add_node("process", text_processor)
```

### 2. 模块化设计

```python
# 每个子图专注于特定功能
research_module = create_research_subgraph()
analysis_module = create_analysis_subgraph()
writing_module = create_writing_subgraph()

# 组合成完整的系统
main_graph = StateGraph(State)
main_graph.add_node("research", research_module)
main_graph.add_node("analyze", analysis_module)
main_graph.add_node("write", writing_module)
```

### 3. 独立测试

```python
# 可以独立测试子图
def test_research_subgraph():
    subgraph = create_research_subgraph()
    result = subgraph.invoke({"query": "test", "sources": [], "summary": ""})
    assert "sources" in result
    assert "summary" in result
```
