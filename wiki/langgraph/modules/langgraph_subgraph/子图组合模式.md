# 子图组合模式

子图组合模式介绍了如何在 LangGraph 中使用子图来构建复杂的模块化系统。本文详细介绍各种子图使用模式、状态传递机制和最佳实践。

## 基本组合模式

### 1. 子图作为节点

最基本的模式是将子图直接作为父图的一个节点。

```python
from langgraph.graph import StateGraph, START, END
from typing_extensions import TypedDict

# 定义子图
class SubgraphState(TypedDict):
    input: str
    output: str

def create_subgraph():
    builder = StateGraph(SubgraphState)
    
    def process(state: SubgraphState):
        return {"output": f"处理: {state['input']}"}
    
    builder.add_node("process", process)
    builder.add_edge(START, "process")
    builder.add_edge("process", END)
    
    return builder.compile()

# 定义父图
class ParentState(TypedDict):
    query: str
    result: str

def create_parent_graph():
    builder = StateGraph(ParentState)
    subgraph = create_subgraph()
    
    def prepare(state: ParentState):
        return {"input": state["query"]}
    
    def finalize(state: ParentState):
        return {"result": state["output"]}
    
    builder.add_node("prepare", prepare)
    builder.add_node("subgraph", subgraph)
    builder.add_node("finalize", finalize)
    
    builder.add_edge(START, "prepare")
    builder.add_edge("prepare", "subgraph")
    builder.add_edge("subgraph", "finalize")
    builder.add_edge("finalize", END)
    
    return builder.compile()
```

### 2. 多个子图组合

可以将多个子图组合成一个更大的系统。

```python
class MainState(TypedDict):
    input: str
    research_result: str
    analysis_result: str
    final_output: str

def create_complete_system():
    builder = StateGraph(MainState)
    
    # 创建多个子图
    research_subgraph = create_research_subgraph()
    analysis_subgraph = create_analysis_subgraph()
    writing_subgraph = create_writing_subgraph()
    
    # 添加子图节点
    builder.add_node("research", research_subgraph)
    builder.add_node("analyze", analysis_subgraph)
    builder.add_node("write", writing_subgraph)
    
    # 设置流程
    builder.add_edge(START, "research")
    builder.add_edge("research", "analyze")
    builder.add_edge("analyze", "write")
    builder.add_edge("write", END)
    
    return builder.compile()
```

## 状态传递模式

### 1. 相同状态结构

子图和父图使用相同的状态结构，状态自动传递。

```python
class SharedState(TypedDict):
    messages: list
    context: dict
    result: str

# 子图和父图使用相同的状态
subgraph = create_subgraph_with_state(SharedState)
parent = create_parent_with_state(SharedState)

# 状态自动传递，无需映射
parent.add_node("subgraph", subgraph)
```

**优点**：
- 简单直接
- 无需状态映射
- 适合状态结构一致的场景

**缺点**：
- 子图和父图紧密耦合
- 状态结构必须完全一致

### 2. 状态映射

子图和父图使用不同的状态结构，需要显式映射。

```python
class ParentState(TypedDict):
    user_query: str
    final_answer: str

class SubgraphState(TypedDict):
    question: str
    research_data: dict
    answer: str

def create_parent_with_mapping():
    builder = StateGraph(ParentState)
    subgraph = create_subgraph()
    
    # 输入映射：父图 -> 子图
    def map_input(state: ParentState):
        return {
            "question": state["user_query"],
            "research_data": {},
            "answer": ""
        }
    
    # 输出映射：子图 -> 父图
    def map_output(state: ParentState):
        return {"final_answer": state["answer"]}
    
    builder.add_node("subgraph", subgraph)
    
    # 使用状态映射
    builder.add_edge(START, "subgraph")
    builder.add_edge("subgraph", END)
    
    return builder.compile()
```

**优点**：
- 子图和父图解耦
- 可以重用不同状态的子图
- 更灵活

**缺点**：
- 需要编写映射逻辑
- 稍微复杂一些

### 3. 部分状态传递

只传递子图需要的状态字段。

```python
class ParentState(TypedDict):
    query: str
    context: dict
    metadata: dict
    result: str

class SubgraphState(TypedDict):
    query: str
    context: dict

def create_parent_with_partial_state():
    builder = StateGraph(ParentState)
    subgraph = create_subgraph()
    
    # 只传递需要的字段
    def map_to_subgraph(state: ParentState):
        return {
            "query": state["query"],
            "context": state["context"]
            # 不传递 metadata
        }
    
    builder.add_node("subgraph", subgraph)
    builder.add_edge(START, "subgraph")
    builder.add_edge("subgraph", END)
    
    return builder.compile()
```

## 常见子图模式

### 1. 工具作为子图

将复杂的工具逻辑封装为子图。

```python
class ToolState(TypedDict):
    tool_name: str
    parameters: dict
    result: str
    error: str

def create_tool_subgraph(tool_name: str):
    """创建工具子图"""
    builder = StateGraph(ToolState)
    
    def validate_params(state: ToolState):
        # 验证参数
        if not validate(state["parameters"]):
            return {"error": "参数无效"}
        return {"error": ""}
    
    def execute_tool(state: ToolState):
        if state["error"]:
            return {"result": ""}
        
        try:
            result = execute_tool_logic(tool_name, state["parameters"])
            return {"result": result}
        except Exception as e:
            return {"error": str(e)}
    
    def format_result(state: ToolState):
        if state["error"]:
            return {"result": f"错误: {state['error']}"}
        return {"result": f"结果: {state['result']}"}
    
    builder.add_node("validate", validate_params)
    builder.add_node("execute", execute_tool)
    builder.add_node("format", format_result)
    
    builder.add_edge(START, "validate")
    builder.add_edge("validate", "execute")
    builder.add_edge("execute", "format")
    builder.add_edge("format", END)
    
    return builder.compile()

# 使用工具子图
class MainState(TypedDict):
    tool_calls: list
    tool_results: list

def create_main_with_tools():
    builder = StateGraph(MainState)
    
    search_tool = create_tool_subgraph("search")
    calculator_tool = create_tool_subgraph("calculator")
    
    builder.add_node("search", search_tool)
    builder.add_node("calculator", calculator_tool)
    
    builder.add_edge(START, "search")
    builder.add_edge("search", "calculator")
    builder.add_edge("calculator", END)
    
    return builder.compile()
```

### 2. 智能体作为子图

将智能体封装为子图，便于重用。

```python
class AgentState(TypedDict):
    messages: Annotated[list, add_messages]
    agent_name: str
    tools: list

def create_agent_subgraph(agent_name: str, tools: list):
    """创建智能体子图"""
    builder = StateGraph(AgentState)
    
    llm = ChatOpenAI(model="gpt-4")
    
    def think(state: AgentState):
        response = llm.invoke(state["messages"])
        return {"messages": [response]}
    
    def use_tools(state: AgentState):
        last_message = state["messages"][-1]
        
        if hasattr(last_message, "tool_calls"):
            # 执行工具调用
            results = execute_tool_calls(last_message.tool_calls, tools)
            return {"messages": results}
        
        return {"messages": []}
    
    def should_continue(state: AgentState):
        last_message = state["messages"][-1]
        
        if hasattr(last_message, "tool_calls") and last_message.tool_calls:
            return "tools"
        return "end"
    
    builder.add_node("think", think)
    builder.add_node("tools", use_tools)
    
    builder.add_edge(START, "think")
    builder.add_conditional_edges("think", should_continue, {
        "tools": "tools",
        "end": END
    })
    builder.add_edge("tools", "think")
    
    return builder.compile()

# 使用智能体子图
class MainState(TypedDict):
    messages: Annotated[list, add_messages]
    agent_results: dict

def create_multi_agent_system():
    builder = StateGraph(MainState)
    
    researcher = create_agent_subgraph("研究员", [search_tool])
    writer = create_agent_subgraph("写作者", [writing_tool])
    
    builder.add_node("researcher", researcher)
    builder.add_node("writer", writer)
    
    builder.add_edge(START, "researcher")
    builder.add_edge("researcher", "writer")
    builder.add_edge("writer", END)
    
    return builder.compile()
```

### 3. 工作流作为子图

将复杂的工作流封装为子图。

```python
class WorkflowState(TypedDict):
    task: str
    steps_completed: list
    current_step: str
    result: str

def create_approval_workflow():
    """创建审批工作流子图"""
    builder = StateGraph(WorkflowState)
    
    def submit_request(state: WorkflowState):
        return {"current_step": "submitted"}
    
    def review_request(state: WorkflowState):
        # 这里可以添加 interrupt() 等待人类审批
        return {"current_step": "reviewed"}
    
    def approve_request(state: WorkflowState):
        return {"current_step": "approved", "result": "已批准"}
    
    def reject_request(state: WorkflowState):
        return {"current_step": "rejected", "result": "已拒绝"}
    
    def route_after_review(state: WorkflowState):
        # 根据审批结果路由
        if state.get("approved", True):
            return "approve"
        return "reject"
    
    builder.add_node("submit", submit_request)
    builder.add_node("review", review_request)
    builder.add_node("approve", approve_request)
    builder.add_node("reject", reject_request)
    
    builder.add_edge(START, "submit")
    builder.add_edge("submit", "review")
    builder.add_conditional_edges("review", route_after_review, {
        "approve": "approve",
        "reject": "reject"
    })
    builder.add_edge("approve", END)
    builder.add_edge("reject", END)
    
    return builder.compile()

# 使用工作流子图
class MainState(TypedDict):
    request: str
    workflow_result: str

def create_main_with_workflow():
    builder = StateGraph(MainState)
    
    approval_workflow = create_approval_workflow()
    
    builder.add_node("workflow", approval_workflow)
    builder.add_edge(START, "workflow")
    builder.add_edge("workflow", END)
    
    return builder.compile()
```

## 高级组合模式

### 1. 条件子图

根据条件选择不同的子图。

```python
class MainState(TypedDict):
    task_type: str
    input_data: str
    result: str

def create_conditional_system():
    builder = StateGraph(MainState)
    
    # 创建不同的子图
    research_subgraph = create_research_subgraph()
    writing_subgraph = create_writing_subgraph()
    analysis_subgraph = create_analysis_subgraph()
    
    def route_by_task_type(state: MainState):
        task_type = state["task_type"]
        
        if task_type == "research":
            return "research"
        elif task_type == "writing":
            return "writing"
        elif task_type == "analysis":
            return "analysis"
        else:
            return "default"
    
    builder.add_node("research", research_subgraph)
    builder.add_node("writing", writing_subgraph)
    builder.add_node("analysis", analysis_subgraph)
    builder.add_node("default", default_node)
    
    builder.add_conditional_edges(START, route_by_task_type, {
        "research": "research",
        "writing": "writing",
        "analysis": "analysis",
        "default": "default"
    })
    
    builder.add_edge("research", END)
    builder.add_edge("writing", END)
    builder.add_edge("analysis", END)
    builder.add_edge("default", END)
    
    return builder.compile()
```

### 2. 并行子图

多个子图并行执行。

```python
class MainState(TypedDict):
    input: str
    research_result: str
    analysis_result: str
    combined_result: str

def create_parallel_system():
    builder = StateGraph(MainState)
    
    research_subgraph = create_research_subgraph()
    analysis_subgraph = create_analysis_subgraph()
    
    builder.add_node("research", research_subgraph)
    builder.add_node("analysis", analysis_subgraph)
    
    # 并行执行
    builder.add_edge(START, "research")
    builder.add_edge(START, "analysis")
    
    def combine_results(state: MainState):
        return {
            "combined_result": f"{state['research_result']} + {state['analysis_result']}"
        }
    
    builder.add_node("combine", combine_results)
    builder.add_edge("research", "combine")
    builder.add_edge("analysis", "combine")
    builder.add_edge("combine", END)
    
    return builder.compile()
```

### 3. 循环子图

子图可以循环执行。

```python
class MainState(TypedDict):
    input: str
    iteration: int
    result: str
    should_continue: bool

def create_iterative_system():
    builder = StateGraph(MainState)
    
    process_subgraph = create_process_subgraph()
    
    builder.add_node("process", process_subgraph)
    
    def check_iteration(state: MainState):
        if state["iteration"] < 3 and state["should_continue"]:
            return "continue"
        return "end"
    
    def increment_iteration(state: MainState):
        return {"iteration": state["iteration"] + 1}
    
    builder.add_node("increment", increment_iteration)
    
    builder.add_edge(START, "process")
    builder.add_conditional_edges("process", check_iteration, {
        "continue": "increment",
        "end": END
    })
    builder.add_edge("increment", "process")
    
    return builder.compile()
```

## 状态传递详解

### 1. 自动状态传递

当子图和父图使用相同的状态结构时，状态自动传递。

```python
class SharedState(TypedDict):
    messages: list
    context: dict

# 子图和父图使用相同的状态
subgraph = create_subgraph()
parent = create_parent()

# 状态自动传递
parent.add_node("subgraph", subgraph)
```

### 2. 手动状态映射

使用 `input_mapper` 和 `output_mapper` 进行状态映射。

```python
# 输入映射：父图 -> 子图
def input_mapper(parent_state):
    return {
        "subgraph_input": parent_state["parent_input"],
        "subgraph_context": parent_state["parent_context"]
    }

# 输出映射：子图 -> 父图
def output_mapper(subgraph_state):
    return {
        "parent_output": subgraph_state["subgraph_output"],
        "parent_result": subgraph_state["subgraph_result"]
    }

# 添加子图节点，指定映射
parent.add_node(
    "subgraph",
    subgraph,
    input_mapper=input_mapper,
    output_mapper=output_mapper
)
```

### 3. 状态合并

子图的输出可以与父图的现有状态合并。

```python
class ParentState(TypedDict):
    data: str
    results: Annotated[list, merge_results]

def merge_results(left, right):
    """合并结果列表"""
    return left + right

# 子图的输出会自动合并到父图的 results 列表
parent.add_node("subgraph", subgraph)
```

## 最佳实践

### 1. 清晰的接口定义

```python

class SubgraphInput(TypedDict):
    """子图输入接口"""
    query: str
    context: dict

class SubgraphOutput(TypedDict):
    """子图输出接口"""
    result: str
    confidence: float

def create_well_defined_subgraph():
    """定义清晰的子图"""
    builder = StateGraph(SubgraphInput)
    # ...
    return builder.compile()

# 不好的做法：接口模糊
def create_vague_subgraph():
    builder = StateGraph(dict)  # 不明确的类型
    # ...
    return builder.compile()
```

### 2. 单一职责

```python
# 好的做法：每个子图专注于一个功能
research_subgraph = create_research_subgraph()  # 只负责研究
analysis_subgraph = create_analysis_subgraph()  # 只负责分析
writing_subgraph = create_writing_subgraph()    # 只负责写作
```

## 完整示例：模块化内容生成系统

```python
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from typing import Annotated
from typing_extensions import TypedDict

# 研究子图
class ResearchState(TypedDict):
    topic: str
    sources: list
    summary: str

def create_research_subgraph():
    """研究子图"""
    builder = StateGraph(ResearchState)
    
    def search_sources(state: ResearchState):
        llm = ChatOpenAI(model="gpt-4")
        response = llm.invoke(f"搜索关于 {state['topic']} 的信息")
        sources = [response.content]
        return {"sources": sources}
    
    def summarize(state: ResearchState):
        llm = ChatOpenAI(model="gpt-4")
        response = llm.invoke(f"总结以下信息: {state['sources']}")
        return {"summary": response.content}
    
    builder.add_node("search", search_sources)
    builder.add_node("summarize", summarize)
    
    builder.add_edge(START, "search")
    builder.add_edge("search", "summarize")
    builder.add_edge("summarize", END)
    
    return builder.compile()

# 写作子图
class WritingState(TypedDict):
    research_summary: str
    outline: str
    draft: str

def create_writing_subgraph():
    """写作子图"""
    builder = StateGraph(WritingState)
    
    def create_outline(state: WritingState):
        llm = ChatOpenAI(model="gpt-4")
        response = llm.invoke(f"基于以下研究创建大纲: {state['research_summary']}")
        return {"outline": response.content}
    
    def write_draft(state: WritingState):
        llm = ChatOpenAI(model="gpt-4")
        response = llm.invoke(f"基于大纲写草稿: {state['outline']}")
        return {"draft": response.content}
    
    builder.add_node("outline", create_outline)
    builder.add_node("draft", write_draft)
    
    builder.add_edge(START, "outline")
    builder.add_edge("outline", "draft")
    builder.add_edge("draft", END)
    
    return builder.compile()

# 编辑子图
class EditingState(TypedDict):
    draft: str
    edited_version: str

def create_editing_subgraph():
    """编辑子图"""
    builder = StateGraph(EditingState)
    
    def edit_content(state: EditingState):
        llm = ChatOpenAI(model="gpt-4")
        response = llm.invoke(f"编辑以下内容: {state['draft']}")
        return {"edited_version": response.content}
    
    builder.add_node("edit", edit_content)
    builder.add_edge(START, "edit")
    builder.add_edge("edit", END)
    
    return builder.compile()

# 主系统
class MainState(TypedDict):
    topic: str
    research_data: dict
    content_draft: str
    final_content: str

def create_content_generation_system():
    """创建内容生成系统"""
    builder = StateGraph(MainState)
    
    # 创建子图
    research = create_research_subgraph()
    writing = create_writing_subgraph()
    editing = create_editing_subgraph()
    
    # 状态映射函数
    def map_to_research(state: MainState):
        return {"topic": state["topic"], "sources": [], "summary": ""}
    
    def map_research_to_writing(state: MainState):
        return {
            "research_summary": state.get("summary", ""),
            "outline": "",
            "draft": ""
        }
    
    def map_writing_to_editing(state: MainState):
        return {"draft": state.get("draft", ""), "edited_version": ""}
    
    def finalize(state: MainState):
        return {"final_content": state.get("edited_version", "")}
    
    # 添加节点
    builder.add_node("research", research)
    builder.add_node("writing", writing)
    builder.add_node("editing", editing)
    builder.add_node("finalize", finalize)
    
    # 设置边
    builder.add_edge(START, "research")
    builder.add_edge("research", "writing")
    builder.add_edge("writing", "editing")
    builder.add_edge("editing", "finalize")
    builder.add_edge("finalize", END)
    
    return builder.compile()

# 使用示例
app = create_content_generation_system()
result = app.invoke({
    "topic": "人工智能的未来",
    "research_data": {},
    "content_draft": "",
    "final_content": ""
})

print(f"最终内容: {result['final_content']}")
```
