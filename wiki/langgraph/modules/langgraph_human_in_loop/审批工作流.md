# 审批工作流

审批工作流是人机交互中最常见的模式，允许在关键操作前获得人类批准。本文介绍各种审批工作流的实现模式和最佳实践。

## 基本审批模式

### 1. 简单审批

最简单的审批模式，等待人类批准或拒绝。

```python
from langgraph.graph import StateGraph, START, END
from langgraph.types import interrupt, Command
from langgraph.checkpoint.memory import MemorySaver
from typing_extensions import TypedDict

class State(TypedDict):
    data: str
    approved: bool
    result: str

def generate_node(state: State):
    """生成需要审批的内容"""
    return {"data": "生成的内容"}

def approval_node(state: State):
    """简单审批节点"""
    decision = interrupt({
        "message": "请审批以下内容",
        "content": state["data"],
        "options": ["approve", "reject"]
    })
    
    if decision == "approve":
        return {"approved": True, "result": state["data"]}
    else:
        return {"approved": False, "result": "已拒绝"}

def send_node(state: State):
    """发送结果"""
    if state["approved"]:
        return {"result": f"已发送: {state['result']}"}
    else:
        return {"result": "未发送"}

# 构建图
graph = StateGraph(State)
graph.add_node("generate", generate_node)
graph.add_node("approval", approval_node)
graph.add_node("send", send_node)

graph.add_edge(START, "generate")
graph.add_edge("generate", "approval")
graph.add_edge("approval", "send")
graph.add_edge("send", END)

checkpointer = MemorySaver()
app = graph.compile(checkpointer=checkpointer)
```

### 2. 工具调用审批

在执行敏感工具前需要审批。

```python
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool

@tool
def send_email(to: str, subject: str, body: str):
    """发送邮件"""
    return f"邮件已发送给 {to}"

@tool
def delete_file(file_path: str):
    """删除文件"""
    return f"文件 {file_path} 已删除"

def tool_approval_node(state):
    """工具调用审批"""
    tool_call = state["pending_tool_call"]
    
    # 根据工具的风险级别决定是否需要审批
    high_risk_tools = ["delete_file", "send_email"]
    
    if tool_call["name"] in high_risk_tools:
        decision = interrupt({
            "type": "tool_approval",
            "tool": tool_call["name"],
            "arguments": tool_call["args"],
            "message": f"是否执行工具 {tool_call['name']}?"
        })
        
        if decision != "approve":
            return {"tool_result": "工具调用被拒绝"}
    
    # 执行工具
    tool_func = get_tool_by_name(tool_call["name"])
    result = tool_func.invoke(tool_call["args"])
    
    return {"tool_result": result}
```

### 3. 输出审核模式

在发送响应前审核内容。

```python
def review_output_node(state):
    """审核输出内容"""
    output = state["generated_output"]
    
    # 暂停等待审核
    feedback = interrupt({
        "type": "content_review",
        "content": output,
        "message": "请审核以下内容，或提供修改建议",
        "options": ["approve", "edit", "reject"]
    })
    
    if feedback == "approve":
        return {"final_output": output, "status": "approved"}
    elif feedback == "reject":
        return {"final_output": None, "status": "rejected"}
    else:
        # 假设 feedback 是修改后的内容
        return {"final_output": feedback, "status": "edited"}
```

## 多级审批

### 1. 顺序多级审批

多个审批者依次审批。

```python
class State(TypedDict):
    data: str
    manager_approved: bool
    director_approved: bool
    status: str

def manager_approval(state: State):
    """经理审批"""
    decision = interrupt({
        "level": 1,
        "message": "请经理审批",
        "data": state["data"]
    })
    
    if decision == "approve":
        return {"manager_approved": True}
    else:
        return {"status": "rejected_by_manager"}

def director_approval(state: State):
    """总监审批"""
    decision = interrupt({
        "level": 2,
        "message": "请总监审批",
        "data": state["data"]
    })
    
    if decision == "approve":
        return {"director_approved": True, "status": "fully_approved"}
    else:
        return {"status": "rejected_by_director"}

def route_after_manager(state: State):
    """经理审批后的路由"""
    if state.get("status") == "rejected_by_manager":
        return "rejected"
    return "director"

# 构建图
graph = StateGraph(State)
graph.add_node("manager_approval", manager_approval)
graph.add_node("director_approval", director_approval)

graph.add_edge(START, "manager_approval")
graph.add_conditional_edges("manager_approval", route_after_manager, {
    "director": "director_approval",
    "rejected": END
})
graph.add_edge("director_approval", END)
```

### 2. 并行多级审批

多个审批者同时审批。

```python
from langgraph.graph import StateGraph, START, END
from langgraph.types import interrupt
from typing import Annotated

class State(TypedDict):
    data: str
    approvals: Annotated[dict, merge_approvals]
    all_approved: bool

def merge_approvals(left, right):
    """合并审批结果"""
    return {**left, **right}

def reviewer_a(state: State):
    """审批者 A"""
    decision = interrupt({"reviewer": "A", "message": "请审批"})
    return {"approvals": {"A": decision == "approve"}}

def reviewer_b(state: State):
    """审批者 B"""
    decision = interrupt({"reviewer": "B", "message": "请审批"})
    return {"approvals": {"B": decision == "approve"}}

def reviewer_c(state: State):
    """审批者 C"""
    decision = interrupt({"reviewer": "C", "message": "请审批"})
    return {"approvals": {"C": decision == "approve"}}

def check_all_approved(state: State):
    """检查是否全部批准"""
    approvals = state.get("approvals", {})
    all_approved = all(approvals.values())
    return {"all_approved": all_approved}

# 构建图
graph = StateGraph(State)
graph.add_node("reviewer_a", reviewer_a)
graph.add_node("reviewer_b", reviewer_b)
graph.add_node("reviewer_c", reviewer_c)
graph.add_node("check", check_all_approved)

graph.add_edge(START, "reviewer_a")
graph.add_edge(START, "reviewer_b")
graph.add_edge(START, "reviewer_c")
graph.add_edge("reviewer_a", "check")
graph.add_edge("reviewer_b", "check")
graph.add_edge("reviewer_c", "check")
graph.add_edge("check", END)
```

### 3. 条件审批

根据条件决定需要哪些审批。

```python
def conditional_approval_node(state):
    """条件审批"""
    amount = state["amount"]
    risk_level = state["risk_level"]
    
    # 根据金额和风险级别决定审批流程
    if amount > 100000 or risk_level == "high":
        # 需要总监审批
        approver = "director"
    elif amount > 10000 or risk_level == "medium":
        # 需要经理审批
        approver = "manager"
    else:
        # 自动批准
        return {"status": "auto_approved"}
    
    decision = interrupt({
        "approver": approver,
        "amount": amount,
        "risk_level": risk_level,
        "message": f"需要 {approver} 审批"
    })
    
    return {"status": "approved" if decision == "approve" else "rejected"}
```

## 审批链模式

### 1. 审批链

多个步骤，每个步骤都需要审批。

```python
class State(TypedDict):
    step: int
    data: str
    results: list
    status: str

def step_with_approval(state: State):
    """带审批的步骤"""
    step = state["step"]
    
    # 执行步骤逻辑
    result = execute_step(step, state["data"])
    
    # 请求审批
    decision = interrupt({
        "step": step,
        "result": result,
        "message": f"步骤 {step} 已完成，是否继续？"
    })
    
    if decision == "approve":
        return {
            "step": step + 1,
            "results": state["results"] + [result],
            "status": "continue"
        }
    else:
        return {"status": "stopped"}

def route_after_step(state: State):
    """步骤后的路由"""
    if state["status"] == "stopped":
        return "end"
    elif state["step"] > 3:  # 总共 3 个步骤
        return "end"
    else:
        return "step"

# 构建图
graph = StateGraph(State)
graph.add_node("step", step_with_approval)

graph.add_edge(START, "step")
graph.add_conditional_edges("step", route_after_step, {
    "step": "step",
    "end": END
})
```

## 超时处理

### 1. 审批超时

```python
import asyncio
from datetime import datetime, timedelta

async def approval_with_timeout(state):
    """带超时的审批"""
    timeout_hours = 24
    
    try:
        # 等待审批，设置超时
        decision = await asyncio.wait_for(
            interrupt({
                "message": "请审批",
                "timeout": f"{timeout_hours}小时"
            }),
            timeout=timeout_hours * 3600
        )
        return {"decision": decision, "status": "manual"}
    except asyncio.TimeoutError:
        # 超时处理
        return {"decision": "timeout", "status": "auto_rejected"}
```

### 2. 自动升级

```python
def escalation_approval(state):
    """升级审批"""
    approval_count = state.get("approval_count", 0)
    
    if approval_count == 0:
        # 第一次：经理审批
        decision = interrupt({"level": "manager", "message": "请经理审批"})
    elif approval_count == 1:
        # 第二次：总监审批
        decision = interrupt({"level": "director", "message": "请总监审批"})
    else:
        # 第三次：VP 审批
        decision = interrupt({"level": "vp", "message": "请 VP 审批"})
    
    if decision == "approve":
        return {"status": "approved"}
    else:
        return {
            "approval_count": approval_count + 1,
            "status": "retry"
        }
```

## 最佳实践

### 1. 提供清晰的上下文

```python
# 好的做法
decision = interrupt({
    "message": "请审批以下数据处理请求",
    "data": {
        "operation": "delete_records",
        "table": "users",
        "count": 150,
        "reason": "数据清理"
    },
    "impact": "将永久删除 150 条用户记录",
    "rollback": "可以从备份恢复",
    "deadline": "2024-01-01 12:00",
    "options": ["approve", "reject", "request_more_info"]
})

# 不好的做法
decision = interrupt("审批？")
```

### 2. 验证人类输入

```python
def validated_approval(state):
    """带验证的审批"""
    while True:
        decision = interrupt({
            "message": "请审批",
            "options": ["approve", "reject"]
        })
        
        if decision in ["approve", "reject"]:
            return {"decision": decision}
        else:
            # 输入无效，继续等待
            continue
```

### 3. 记录审计日志

```python
def auditable_approval(state):
    """可审计的审批"""
    # 记录审批请求
    log_audit({
        "action": "approval_requested",
        "data": state["data"],
        "timestamp": datetime.now().isoformat(),
        "requester": state["user_id"]
    })
    
    decision = interrupt({"message": "请审批"})
    
    # 记录审批结果
```
