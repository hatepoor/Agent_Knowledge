# 错误处理与恢复

在生产环境中，LLM 调用、工具执行、网络请求都可能失败。LangGraph 提供了**多层次**的错误处理机制，从节点内部的 try/except 到图级别的错误传播、自动重试、以及基于 Checkpointer 的状态恢复，帮助你构建**健壮可靠**的 Agent 应用。

## 核心原则

LangGraph 的错误处理遵循以下原则：

1. **就近处理**：节点内部处理自己最了解的错误
2. **向上传播**：未处理的错误会传播到图的执行层
3. **状态保留**：Checkpointer 保存了每一步状态，可以从任意检查点恢复
4. **优雅降级**：失败时提供合理的回退方案，而不是直接崩溃

## 一、节点级错误处理

### 1.1 基本 try/except 模式

最直接的方式是在节点函数内部捕获和处理异常：

```python
from typing import TypedDict, Annotated
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, START, END

class State(TypedDict):
    """带错误信息的状态"""
    question: str
    answer: str
    error: str          # 记录错误信息
    retry_count: int    # 重试次数

def call_llm_node(state: State) -> dict:
    """
    调用 LLM 的节点，包含错误处理
    """
    try:
        llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
        response = llm.invoke(state["question"])
        return {
            "answer": response.content,
            "error": "",           # 成功时清空错误
            "retry_count": 0
        }
    except Exception as e:
        # 记录错误但不中断流程
        return {
            "answer": "",
            "error": f"LLM 调用失败: {str(e)}",
            "retry_count": state.get("retry_count", 0) + 1
        }

def handle_error_node(state: State) -> dict:
    """
    错误处理节点：当 LLM 调用失败时的降级方案
    """
    return {
        "answer": f"抱歉，暂时无法处理您的请求。错误信息：{state['error']}",
        "error": ""  # 清除错误标记，让流程继续
    }
```

### 1.2 分类异常处理

不同类型的异常需要不同的处理策略：

```python
import httpx
from langchain_core.exceptions import OutputParserException

def tool_calling_node(state: State) -> dict:
    """
    工具调用节点，分类处理各种异常
    """
    try:
        # 调用外部 API
        result = call_external_api(state["question"])
        return {"answer": result, "error": ""}

    except httpx.TimeoutException:
        # 超时：可以重试
        return {
            "error": "请求超时，请稍后重试",
            "retry_count": state.get("retry_count", 0) + 1
        }

    except httpx.HTTPStatusError as e:
        # HTTP 错误：根据状态码决定策略
        if e.response.status_code == 429:
            # 限流：等待后重试
            return {
                "error": "请求频率过高，请稍后重试",
                "retry_count": state.get("retry_count", 0) + 1
            }
        elif e.response.status_code >= 500:
            # 服务端错误：使用降级方案
            return {
                "answer": "服务暂时不可用，已切换到备用方案。",
                "error": ""
            }
        else:
            # 客户端错误：返回明确提示
            return {
                "answer": f"请求参数有误：{e.response.text}",
                "error": ""
            }

    except OutputParserException as e:
        # 解析错误：记录并返回原始数据
        return {
            "error": f"输出解析失败: {str(e)}",
            "retry_count": state.get("retry_count", 0) + 1
        }

    except Exception as e:
        # 兜底：未知错误
        return {
            "error": f"未知错误: {type(e).__name__}: {str(e)}",
            "retry_count": state.get("retry_count", 0) + 1
        }
```

### 1.3 带重试逻辑的节点

在节点内部实现有限次数的重试：

```python
import time

def call_llm_with_retry(state: State) -> dict:
    """
    带重试逻辑的 LLM 调用节点
    最多重试 3 次，每次间隔递增
    """
    max_retries = 3
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

    for attempt in range(max_retries + 1):
        try:
            response = llm.invoke(state["question"])
            return {"answer": response.content, "error": ""}
        except Exception as e:
            if attempt < max_retries:
                # 指数退避：1s, 2s, 4s
                wait_time = 2 ** attempt
                print(f"[重试] 第 {attempt + 1} 次失败，{wait_time}s 后重试...")
                time.sleep(wait_time)
            else:
                return {
                    "answer": "",
                    "error": f"经过 {max_retries} 次重试仍然失败: {str(e)}"
                }
```

## 二、图级错误传播

### 2.1 条件边实现错误路由

通过条件边根据错误状态动态路由到不同的处理节点：

```python
from langgraph.graph import StateGraph, START, END

class State(TypedDict):
    question: str
    answer: str
    error: str
    retry_count: int

def call_llm_node(state: State) -> dict:
    """调用 LLM 节点"""
    try:
        llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
        response = llm.invoke(state["question"])
        return {"answer": response.content, "error": ""}
    except Exception as e:
        return {"error": str(e), "retry_count": state.get("retry_count", 0) + 1}

def fallback_node(state: State) -> dict:
    """降级处理节点"""
    return {
        "answer": "AI 服务暂时不可用，请稍后再试。",
        "error": ""
    }

def retry_node(state: State) -> dict:
    """重试前的准备节点"""
    print(f"[重试] 当前重试次数: {state['retry_count']}")
    return {}  # 不修改状态，直接回到 call_llm

def should_retry_or_fallback(state: State) -> str:
    """
    条件路由函数：
    - 没有错误 → 结束
    - 有错误且重试次数 < 3 → 重试
    - 有错误且重试次数 >= 3 → 降级
    """
    if not state.get("error"):
        return "end"
    elif state.get("retry_count", 0) < 3:
        return "retry"
    else:
        return "fallback"

# 构建图
builder = StateGraph(State)
builder.add_node("call_llm", call_llm_node)
builder.add_node("fallback", fallback_node)
builder.add_node("retry", retry_node)

builder.add_edge(START, "call_llm")

# 条件边：根据错误状态决定下一步
builder.add_conditional_edges(
    "call_llm",
    should_retry_or_fallback,
    {
        "end": END,
        "retry": "retry",
        "fallback": "fallback"
    }
)

builder.add_edge("retry", "call_llm")       # 重试后重新调用 LLM
builder.add_edge("fallback", END)            # 降级后结束

graph = builder.compile()
```

### 2.2 错误传播到调用方

如果不希望图内部吞掉错误，可以让异常自然传播：

```python
def strict_llm_node(state: State) -> dict:
    """
    严格模式节点：不捕获异常，让错误传播到调用方
    适用于错误不可恢复、必须通知用户的场景
    """
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
    # 如果这里抛出异常，会直接传播到 graph.invoke() 的调用方
    response = llm.invoke(state["question"])
    return {"answer": response.content}

# 调用方处理
try:
    result = graph.invoke({"question": "你好"})
except Exception as e:
    print(f"图执行失败: {e}")
    # 通知用户、记录日志、触发告警等
```

## 三、`retry` 参数

### 3.1 基本用法

LangGraph 的 `add_node()` 方法支持 `retry` 参数，可以自动重试失败的节点：

```python
from langgraph.graph import StateGraph

builder = StateGraph(State)

# 添加节点并配置自动重试
builder.add_node(
    "call_llm",
    call_llm_node,
    retry={
        "max_attempts": 3,        # 最大重试次数
        "retry_on": Exception,    # 需要重试的异常类型
    }
)
```

### 3.2 retry 参数详解

| 参数 | 类型 | 说明 |
|------|------|------|
| `max_attempts` | `int` | 最大尝试次数（包含首次执行），默认 1（不重试） |
| `retry_on` | `type[Exception]` 或 `tuple` | 触发重试的异常类型，可以是单个异常类或异常类元组 |

### 3.3 指定异常类型重试

```python
import httpx

builder.add_node(
    "api_call",
    api_call_node,
    retry={
        "max_attempts": 5,
        # 只对网络相关异常重试，不对业务逻辑异常重试
        "retry_on": (httpx.TimeoutException, httpx.ConnectError),
    }
)
```

### 3.4 retry 的执行流程

```
首次执行 → 失败 → 重试 1 → 失败 → 重试 2 → 成功 → 继续
                              ↓ 失败
                          重试 3 → 失败 → 抛出异常
```

> **注意**：`retry` 是在节点级别自动执行的，不需要在节点函数内编写重试逻辑。如果节点函数内部已经有重试逻辑，两者会叠加。

## 四、`recursion_limit` 防止无限循环

### 4.1 什么是递归限制

LangGraph 在执行图时会跟踪**超级步骤（super steps）** 的数量。当图中存在循环边（如 Agent 的 observe → think → act → observe 循环），理论上可能无限执行。`recursion_limit` 用于设置执行上限，防止无限循环。

### 4.2 默认值与设置

```python
# 编译时设置（影响所有后续调用）
graph = builder.compile()

# 调用时设置（覆盖编译时的默认值）
# 默认值为 25，表示最多执行 25 个超级步骤
result = graph.invoke(
    {"question": "你好"},
    config={"recursion_limit": 50}  # 提高限制
)
```

### 4.3 什么是"超级步骤"

一个超级步骤是图中一轮并行执行的节点集合：

```
超级步骤 1: START → node_A
超级步骤 2: node_A → node_B, node_C （并行执行）
超级步骤 3: node_B, node_C → node_D
超级步骤 4: node_D → END
```

在 Agent 循环中，每一轮 think → act → observe 通常消耗 2-3 个超级步骤。

> 并行节点的物理执行方式（同步顺序 / 异步真并发）、屏障同步与输入快照机制，详见 [[langgraph/modules/langgraph_core/并行执行模型]]。

### 4.4 超过限制的后果

当超过 `recursion_limit` 时，LangGraph 会抛出 `GraphRecursionError`：

```python
from langgraph.errors import GraphRecursionError

try:
    result = graph.invoke(
        {"question": "复杂问题"},
        config={"recursion_limit": 10}
    )
except GraphRecursionError as e:
    print(f"执行超过递归限制: {e}")
    # 可以获取部分结果（使用 checkpointer 时）
```

### 4.5 合理设置 recursion_limit

| 场景 | 建议值 | 说明 |
|------|--------|------|
| 简单线性工作流 | 10-15 | 没有循环，步骤固定 |
| 单轮 Agent | 20-30 | 可能有 5-10 次工具调用 |
| 多轮对话 Agent | 50-100 | 需要考虑对话轮数 |
| 复杂推理 Agent | 100-200 | 多步推理，大量工具调用 |

### 4.6 在条件边中实现安全的循环控制

```python
def should_continue(state: State) -> str:
    """
    安全的循环控制：同时检查业务逻辑和执行次数
    """
    messages = state.get("messages", [])

    # 业务逻辑判断
    last_message = messages[-1] if messages else None
    if last_message and hasattr(last_message, "tool_calls") and last_message.tool_calls:
        return "continue"  # 有工具调用，继续循环

    return "end"  # 没有工具调用，结束

# 配合 recursion_limit 使用
graph = builder.compile()
result = graph.invoke(
    {"messages": [("user", "帮我查一下天气")]},
    config={"recursion_limit": 30}  # 安全上限
)
```

## 五、基于 Checkpointer 的错误恢复

### 5.1 Checkpointer 如何帮助错误恢复

Checkpointer 在每一步执行后保存完整状态快照。当某个节点失败时，可以：

1. **查看失败时的状态**：了解上下文
2. **修改状态后重试**：修复问题后从检查点恢复
3. **回退到之前的状态**：放弃失败的路径

### 5.2 从失败状态恢复

```python
from langgraph.checkpoint.memory import MemorySaver

# 使用 Checkpointer 编译图
checkpointer = MemorySaver()
graph = builder.compile(checkpointer=checkpointer)

config = {"configurable": {"thread_id": "user-123"}}

# 第一次执行：假设在某个节点失败
try:
    result = graph.invoke(
        {"question": "复杂问题"},
        config=config
    )
except Exception as e:
    print(f"执行失败: {e}")

    # 查看当前状态（失败前的最后状态）
    current_state = graph.get_state(config)
    print(f"失败时的状态: {current_state.values}")
    print(f"失败的节点: {current_state.next}")  # 下一步要执行的节点

    # 修复问题后，更新状态并恢复执行
    graph.update_state(
        config,
        {"error": "", "retry_count": 0}  # 清除错误状态
    )

    # 从上次失败的地方继续执行
    result = graph.invoke(None, config=config)
```

### 5.3 时间旅行：回退到历史状态

```python
# 获取完整的状态历史
state_history = list(graph.get_state_history(config))

print(f"共有 {len(state_history)} 个历史状态")

# 查看每个历史状态
for i, snapshot in enumerate(state_history):
    print(f"步骤 {i}: 节点={snapshot.next}, 值={snapshot.values}")

# 回退到第 2 步的状态
target_snapshot = state_history[2]
graph.update_state(config, target_snapshot.values, as_of=target_snapshot)

# 从回退的状态重新执行
result = graph.invoke(None, config=config)
```

### 5.4 修改失败节点输入后重试

```python
# 获取当前状态
current_state = graph.get_state(config)

# 如果知道是某个输入导致失败，可以修改后重试
graph.update_state(
    config,
    {
        "question": "更简单的问题",  # 换一个更容易处理的输入
        "error": "",
        "retry_count": 0
    }
)

# 重新执行
result = graph.invoke(None, config=config)
```

## 六、优雅降级模式

### 6.1 多级降级策略

```python
class State(TypedDict):
    question: str
    answer: str
    model_tier: str  # 当前使用的模型级别

def call_primary_llm(state: State) -> dict:
    """尝试使用高级模型"""
    try:
        llm = ChatOpenAI(model="gpt-4o", temperature=0)
        response = llm.invoke(state["question"])
        return {"answer": response.content, "model_tier": "primary"}
    except Exception:
        return {"model_tier": "primary_failed"}

def call_secondary_llm(state: State) -> dict:
    """降级到备用模型"""
    try:
        llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
        response = llm.invoke(state["question"])
        return {"answer": response.content, "model_tier": "secondary"}
    except Exception:
        return {"model_tier": "secondary_failed"}

def return_cached_response(state: State) -> dict:
    """最终降级：返回缓存的响应"""
    return {
        "answer": "根据已有信息，我的回答是...",
        "model_tier": "cached"
    }

def route_after_primary(state: State) -> str:
    """路由逻辑"""
    if state.get("model_tier") == "primary":
        return "end"
    elif state.get("model_tier") == "primary_failed":
        return "secondary"
    else:
        return "cached"

def route_after_secondary(state: State) -> str:
    if state.get("model_tier") == "secondary":
        return "end"
    else:
        return "cached"

# 构建多级降级图
builder = StateGraph(State)
builder.add_node("primary", call_primary_llm)
builder.add_node("secondary", call_secondary_llm)
builder.add_node("cached", return_cached_response)

builder.add_edge(START, "primary")
builder.add_conditional_edges("primary", route_after_primary, {
    "end": END, "secondary": "secondary", "cached": "cached"
})
builder.add_conditional_edges("secondary", route_after_secondary, {
    "end": END, "cached": "cached"
})
builder.add_edge("cached", END)

graph = builder.compile()
```

### 6.2 工具降级模式

```python
def search_with_fallback(state: State) -> dict:
    """
    搜索工具降级：主搜索失败时切换到备用搜索
    """
    # 尝试主搜索
    try:
        results = primary_search(state["question"])
        if results:
            return {"search_results": results, "search_engine": "primary"}
    except Exception:
        pass

    # 尝试备用搜索
    try:
        results = secondary_search(state["question"])
        if results:
            return {"search_results": results, "search_engine": "secondary"}
    except Exception:
        pass

    # 最终降级：无搜索结果
    return {
        "search_results": [],
        "search_engine": "none",
        "error": "所有搜索引擎均不可用"
    }
```

## 七、常见错误场景与解决方案

### 7.1 LLM API 超时

```python
# 解决方案：设置超时 + 重试
llm = ChatOpenAI(
    model="gpt-4o-mini",
    timeout=30,          # 30 秒超时
    max_retries=3,       # LangChain 内置重试
    request_timeout=30,  # 兼容旧版本
)
```

### 7.2 工具调用返回异常格式

```python
def robust_tool_node(state: State) -> dict:
    """
    健壮的工具调用节点：处理异常返回格式
    """
    try:
        result = tool.invoke(state["tool_input"])

        # 验证返回格式
        if not isinstance(result, str):
            result = str(result)

        # 检查返回内容是否为空
        if not result.strip():
            result = "工具执行成功但未返回结果"

        return {"tool_output": result, "error": ""}

    except Exception as e:
        return {
            "tool_output": f"工具执行失败: {str(e)}",
            "error": str(e)
        }
```

### 7.3 状态字段缺失

```python
def safe_node(state: State) -> dict:
    """
    安全访问状态字段：使用 get() 提供默认值
    """
    # 不好的做法：直接访问可能不存在的字段
    # data = state["optional_field"]  # 可能 KeyError

    # 好的做法：使用 get() 提供默认值
    data = state.get("optional_field", "默认值")

    # 更好的做法：在类型定义中明确标注可选字段
    return {"result": f"处理结果: {data}"}
```

### 7.4 消息列表过长导致 Token 超限

```python
from langchain_core.messages import trim_messages

def trim_messages_node(state: State) -> dict:
    """
    裁剪消息列表，防止 Token 超限
    """
    messages = state.get("messages", [])

    # 保留最近的消息
    trimmed = trim_messages(
        messages,
        max_tokens=4000,
        token_counter=ChatOpenAI(model="gpt-4o-mini"),
        # 保留系统消息
        allow_partial=False,
        # 策略：保留最新的消息
        strategy="last",
    )

    return {"messages": trimmed}
```

### 7.5 并发节点中的错误隔离

```python
from langgraph.constants import Send

def fan_out_node(state: State) -> list:
    """
    扇出节点：将任务分发到多个并行执行
    每个子任务独立处理错误
    """
    items = state["items"]
    return [
        Send("process_item", {"item": item, "error": None})
        for item in items
    ]

def process_item_node(state: dict) -> dict:
    """
    单个项目处理节点：错误隔离
    """
    try:
        result = process(state["item"])
        return {"result": result, "error": None}
    except Exception as e:
        # 只标记当前项目失败，不影响其他项目
        return {"result": None, "error": str(e)}
```

## 八、完整示例：健壮的生产级 Agent

```python
"""
生产级 Agent 示例：综合错误处理策略
"""
from typing import TypedDict, Annotated
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver
from langgraph.prebuilt import create_react_agent

class State(TypedDict):
    """生产级状态"""
    messages: list
    error: str
    retry_count: int
    fallback_used: bool

def agent_node(state: State) -> dict:
    """
    Agent 节点：带完整错误处理
    """
    try:
        llm = ChatOpenAI(
            model="gpt-4o-mini",
            temperature=0,
            timeout=30,
            max_retries=2,
        )
        response = llm.invoke(state["messages"])
        return {
            "messages": [response],
            "error": "",
        }
    except Exception as e:
        return {
            "error": str(e),
            "retry_count": state.get("retry_count", 0) + 1,
        }

def error_handler_node(state: State) -> dict:
    """错误处理节点"""
    error_msg = state.get("error", "未知错误")
    fallback_msg = AIMessage(
        content=f"抱歉，处理过程中遇到问题：{error_msg}。已记录日志，请稍后重试。"
    )
    return {
        "messages": [fallback_msg],
        "error": "",
        "fallback_used": True,
    }

def should_continue(state: State) -> str:
    """路由决策"""
    if state.get("error"):
        if state.get("retry_count", 0) < 3:
            return "retry"
        else:
            return "error_handler"
    return "end"

# 构建图
builder = StateGraph(State)
builder.add_node("agent", agent_node, retry={
    "max_attempts": 2,
    "retry_on": Exception,
})
builder.add_node("error_handler", error_handler_node)

builder.add_edge(START, "agent")
builder.add_conditional_edges("agent", should_continue, {
    "end": END,
    "retry": "agent",
    "error_handler": "error_handler",
})
builder.add_edge("error_handler", END)

# 使用 Checkpointer 支持状态恢复
checkpointer = MemorySaver()
graph = builder.compile(checkpointer=checkpointer)

# 使用示例
config = {"configurable": {"thread_id": "prod-user-001"}}
try:
    result = graph.invoke(
        {
            "messages": [HumanMessage(content="你好")],
            "error": "",
            "retry_count": 0,
            "fallback_used": False,
        },
        config=config,
    )
    print(f"结果: {result['messages'][-1].content}")
except Exception as e:
    print(f"严重错误: {e}")
    # 从检查点恢复
    state = graph.get_state(config)
    print(f"最后状态: {state.values}")
```

## 总结

| 策略 | 适用场景 | 实现方式 |
|------|----------|----------|
| 节点内 try/except | 可预见的局部错误 | 节点函数内部处理 |
| 条件边路由 | 需要根据错误类型走不同路径 | `add_conditional_edges` |
| `retry` 参数 | 临时性失败（网络超时等） | `add_node(retry={...})` |
| `recursion_limit` | 防止循环失控 | `invoke(config={"recursion_limit": N})` |
| Checkpointer 恢复 | 需要从失败中恢复 | `get_state` + `update_state` |
| 优雅降级 | 服务不可用时提供替代方案 | 多级降级图结构 |
