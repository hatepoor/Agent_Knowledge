# 节点（Node）详解

节点是 LangGraph 图中的**执行单元**，负责执行具体操作并返回状态更新。每个节点都是一个函数，接收当前状态作为输入，返回状态的部分更新。

## 核心概念

### 节点的本质

节点是**纯函数**：
- **输入**：当前状态（State）
- **输出**：状态的部分更新（字典）
- **副作用**：可以有（调用 API、打印日志等），但应该最小化

```python
def my_node(state: State) -> dict:
    """
    节点函数签名
    - 接收完整状态作为参数
    - 返回字典，表示要更新的字段
    """
    # 执行操作
    result = do_something(state["input"])
    
    # 返回状态更新（只包含要更新的字段）
    return {"output": result}
```

### 部分更新（Partial Updates）

节点**不需要**返回完整状态，只需返回**变化的字段**：

```python
class State(TypedDict):
    messages: list
    count: int
    user_name: str

def increment_node(state: State) -> dict:
    """只更新 count 字段"""
    return {"count": state["count"] + 1}
    # 不需要返回 messages 和 user_name
```

LangGraph 会自动将返回的字典**合并**到当前状态中。

## 节点类型

### 类型一：普通函数节点

最常见的节点类型：

```python
def process_data(state: State) -> dict:
    """处理数据节点"""
    data = state["raw_data"]
    processed = data.upper()
    return {"processed_data": processed}

builder.add_node("process", process_data)
```

### 类型二：异步函数节点

适合 I/O 密集型操作（调用 API、数据库查询等）：

```python
async def fetch_data(state: State) -> dict:
    """异步获取数据节点"""
    async with httpx.AsyncClient() as client:
        response = await client.get(state["url"])
        data = response.json()
    return {"fetched_data": data}

builder.add_node("fetch", fetch_data)
```

### 类型三：类节点

适合需要封装复杂逻辑或状态的节点：

```python
class DataProcessor:
    """数据处理器类"""
    
    def __init__(self, config: dict):
        self.config = config
    
    def invoke(self, state: State) -> dict:
        """执行处理"""
        data = state["data"]
        # 使用配置进行处理
        processed = self.transform(data)
        return {"processed": processed}
    
    def transform(self, data):
        """转换数据"""
        return data.upper() if self.config.get("uppercase") else data

# 使用类节点
processor = DataProcessor({"uppercase": True})
builder.add_node("process", processor.invoke)
```

### 类型四：Lambda 节点

适合简单的单行逻辑：

```python
builder.add_node("double", lambda state: {"count": state["count"] * 2})
```

### 类型五：子图节点

将子图作为节点使用，实现模块化：

```python
# 创建子图
subgraph_builder = StateGraph(SubState)
subgraph_builder.add_node("sub_task", sub_task_node)
subgraph_builder.add_edge(START, "sub_task")
subgraph_builder.add_edge("sub_task", END)
subgraph = subgraph_builder.compile()

# 将子图作为节点添加到主图
main_builder = StateGraph(MainState)
main_builder.add_node("subgraph_node", subgraph)
```

## 节点签名

### 标准签名

```python
def node_function(state: State) -> dict:
    """
    标准节点签名
    - state: 当前状态（完整状态）
    - return: 状态的部分更新（字典）
    """
    pass
```

### 带配置的签名

如果节点需要访问运行时配置（如 thread_id）：

```python
from langchain_core.runnables import RunnableConfig

def node_with_config(state: State, config: RunnableConfig) -> dict:
    """带配置的节点"""
    thread_id = config["configurable"]["thread_id"]
    user_id = config["configurable"]["user_id"]
    
    # 使用配置
    result = process_for_user(state["data"], user_id)
    return {"result": result}

builder.add_node("process", node_with_config)
```

### 异步签名

```python
async def async_node(state: State) -> dict:
    """异步节点"""
    result = await some_async_operation()
    return {"result": result}

builder.add_node("async_task", async_node)
```

## 完整示例：多类型节点

```python
"""
完整示例：展示不同类型的节点
"""
from typing import TypedDict
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
import httpx

# 定义状态
class State(TypedDict):
    query: str
    search_results: list
    summary: str
    translated: str

# 初始化 LLM
model = ChatOpenAI(model="gpt-4o-mini", temperature=0)

# 类型一：普通函数节点
def search_node(state: State) -> dict:
    """搜索节点：模拟搜索"""
    query = state["query"]
    # 模拟搜索结果
    results = [
        f"结果1: 关于'{query}'的信息...",
        f"结果2: 更多'{query}'的内容...",
    ]
    return {"search_results": results}

# 类型二：异步函数节点
async def fetch_details(state: State) -> dict:
    """异步获取详情节点"""
    # 模拟异步操作
    import asyncio
    await asyncio.sleep(0.1)
    
    # 实际场景中这里会调用 API
    details = f"详细信息: {state['query']}"
    return {"search_results": state["search_results"] + [details]}

# 类型三：类节点
class Summarizer:
    """摘要生成器"""
    
    def __init__(self, model):
        self.model = model
    
    def invoke(self, state: State) -> dict:
        """生成摘要"""
        results = "\n".join(state["search_results"])
        prompt = f"请总结以下搜索结果：\n{results}"
        response = self.model.invoke(prompt)
        return {"summary": response.content}

summarizer = Summarizer(model)

# 类型四：Lambda 节点
format_node = lambda state: {
    "summary": f"【摘要】{state['summary']}"
}

# 类型五：带配置的节点
from langchain_core.runnables import RunnableConfig

def translate_node(state: State, config: RunnableConfig) -> dict:
    """翻译节点：使用配置中的目标语言"""
    target_lang = config["configurable"].get("target_lang", "English")
    prompt = f"Translate to {target_lang}: {state['summary']}"
    response = model.invoke(prompt)
    return {"translated": response.content}

# 构建图
builder = StateGraph(State)

# 添加不同类型的节点
builder.add_node("search", search_node)
builder.add_node("fetch", fetch_details)
builder.add_node("summarize", summarizer.invoke)
builder.add_node("format", format_node)
builder.add_node("translate", translate_node)

# 添加边
builder.add_edge(START, "search")
builder.add_edge("search", "fetch")
builder.add_edge("fetch", "summarize")
builder.add_edge("summarize", "format")
builder.add_edge("format", "translate")
builder.add_edge("translate", END)

# 编译
graph = builder.compile()

# 执行
config = {"configurable": {"target_lang": "English"}}
result = graph.invoke(
    {"query": "LangGraph 状态管理"},
    config=config
)

print(f"摘要: {result['summary']}")
print(f"翻译: {result['translated']}")
```

## 最佳实践

### 1. 单一职责

每个节点只做一件事：

```python
# 错误：一个节点做太多事
def bad_node(state: State) -> dict:
    data = fetch_data()
    processed = process_data(data)
    saved = save_data(processed)
    notified = send_notification(saved)
    return {"result": notified}

# 正确：拆分为多个节点
def fetch_node(state: State) -> dict:
    return {"data": fetch_data()}

def process_node(state: State) -> dict:
    return {"processed": process_data(state["data"])}

def save_node(state: State) -> dict:
    return {"saved": save_data(state["processed"])}

def notify_node(state: State) -> dict:
    return {"notified": send_notification(state["saved"])}
```

### 2. 幂等性

节点应该是幂等的——相同输入产生相同输出：

```python
# 错误：依赖外部状态（非幂等）
counter = 0
def bad_node(state: State) -> dict:
    global counter
    counter += 1
    return {"count": counter}

# 正确：只依赖输入状态（幂等）
def good_node(state: State) -> dict:
    return {"count": state["count"] + 1}
```

### 3. 错误处理

在节点内部处理错误，避免图崩溃：

```python
def robust_node(state: State) -> dict:
    """健壮的节点：处理错误"""
    try:
        result = risky_operation(state["data"])
        return {"result": result, "error": None}
    except Exception as e:
        # 记录错误，返回默认值
        return {"result": None, "error": str(e)}
```

### 4. 避免大状态

不要在状态中存储大量数据：

```python
# 错误：在状态中存储大文件
def bad_node(state: State) -> dict:
    with open("large_file.txt") as f:
        content = f.read()  # 可能很大
    return {"file_content": content}

# 正确：存储引用或摘要
def good_node(state: State) -> dict:
    file_path = "large_file.txt"
    summary = summarize_file(file_path)
    return {"file_path": file_path, "summary": summary}
```

## 节点与 Reducer

当多个节点更新同一个字段时，Reducer 决定如何合并：

```python
from typing import Annotated
from langgraph.graph import add_messages

class State(TypedDict):
    # 使用 add_messages reducer：新消息追加到列表
    messages: Annotated[list, add_messages]
    
    # 无 reducer：新值覆盖旧值
    count: int

def node_a(state: State) -> dict:
    return {"messages": ["消息A"], "count": 1}

def node_b(state: State) -> dict:
    return {"messages": ["消息B"], "count": 2}

# 如果 A 和 B 都执行：
# - messages: ["消息A", "消息B"]（追加）
# - count: 2（覆盖）
```
