# 主管模式

主管模式（Supervisor Pattern）是多智能体系统中最常见的协作模式，由一个中央协调者（主管）负责分配任务给专业智能体。主管根据当前状态决定哪个智能体应该执行任务，并协调整个工作流程。

## 核心架构

```
         ┌─────────────┐
         │   主管      │
         │ Supervisor  │
         └──────┬──────┘
                │
       ┌────────┼────────┐
       │        │        │
  ┌────▼──┐ ┌──▼───┐ ┌──▼───┐
  │智能体A│ │智能体B│ │智能体C│
  │Agent A│ │Agent B│ │Agent C│
  └───────┘ └──────┘ └──────┘
```

## 工作原理

1. **任务接收**：主管接收初始任务
2. **分析决策**：主管分析任务，决定分配给哪个智能体
3. **任务分配**：主管将任务分配给专业智能体
4. **结果收集**：智能体完成任务，返回结果给主管
5. **评估决策**：主管评估结果，决定是否需要更多工作
6. **任务完成**：当任务完成时，主管返回最终结果

## 基本实现

### 1. 简单主管模式

```python
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from typing import Annotated
from typing_extensions import TypedDict

class State(TypedDict):
    messages: Annotated[list, add_messages]
    next_agent: str

# 创建专业智能体
def researcher(state: State):
    """研究员智能体 - 负责信息收集"""
    llm = ChatOpenAI(model="gpt-4")
    
    # 添加系统提示
    system_msg = {
        "role": "system",
        "content": "你是研究员，负责收集和分析信息。提供详细的研究结果。"
    }
    
    response = llm.invoke([system_msg] + state["messages"])
    return {
        "messages": [response],
        "next_agent": "supervisor"
    }

def writer(state: State):
    """写作者智能体 - 负责内容创作"""
    llm = ChatOpenAI(model="gpt-4")
    
    system_msg = {
        "role": "system",
        "content": "你是写作者，负责将研究结果转化为清晰、有条理的内容。"
    }
    
    response = llm.invoke([system_msg] + state["messages"])
    return {
        "messages": [response],
        "next_agent": "supervisor"
    }

def supervisor(state: State):
    """主管智能体 - 协调任务分配"""
    llm = ChatOpenAI(model="gpt-4")
    
    # 主管决策提示
    supervisor_prompt = """
    你是一个主管，负责协调团队工作。
    
    基于当前对话，决定下一步：
    - 如果需要研究或信息收集，选择 "researcher"
    - 如果需要写作或内容创作，选择 "writer"
    - 如果任务已完成，选择 "FINISH"
    
    只返回选项名称，不要解释。
    """
    
    response = llm.invoke(state["messages"] + [{"role": "user", "content": supervisor_prompt}])
    next_agent = response.content.strip()
    
    return {"next_agent": next_agent}

def route_after_supervisor(state: State):
    """主管决策后的路由"""
    next_agent = state.get("next_agent", "FINISH")
    
    if next_agent == "FINISH":
        return "end"
    elif next_agent == "researcher":
        return "researcher"
    elif next_agent == "writer":
        return "writer"
    else:
        return "end"

# 构建图
graph = StateGraph(State)
graph.add_node("researcher", researcher)
graph.add_node("writer", writer)
graph.add_node("supervisor", supervisor)

# 设置边
graph.add_edge(START, "supervisor")
graph.add_conditional_edges(
    "supervisor",
    route_after_supervisor,
    {
        "researcher": "researcher",
        "writer": "writer",
        "end": END
    }
)
graph.add_edge("researcher", "supervisor")
graph.add_edge("writer", "supervisor")

# 编译图
app = graph.compile()

# 使用示例
result = app.invoke({
    "messages": [{"role": "user", "content": "写一篇关于 AI 的文章"}],
    "next_agent": ""
})
```

### 2. 带工具的主管模式（官方推荐 API：create_agent）

> 注意：旧版 `create_react_agent(llm, tools, state_modifier=...)` 已过时。官方推荐使用 LangChain v1.0+ 的 `create_agent`，`name` 参数是 Supervisor 分发的依据。

```python
from langchain.agents import create_agent
from langchain_core.tools import tool

# 定义工具
@tool
def search_web(query: str):
    """搜索网络信息"""
    return f"搜索结果: 关于 {query} 的信息..."

@tool
def analyze_data(data: str):
    """分析数据"""
    return f"分析结果: {data} 的关键点是..."

@tool
def write_content(topic: str, style: str):
    """撰写内容"""
    return f"已撰写关于 {topic} 的 {style} 风格内容"

# 创建带工具的智能体（新版 API）
def create_specialist_agent(name: str, tools: list, system_prompt: str):
    """创建专业智能体"""
    llm = ChatOpenAI(model="gpt-4")
    return create_agent(
        model=llm,
        tools=tools,
        system_prompt=system_prompt,   # 新版：system_prompt 替代 state_modifier
        name=name,                     # 必须指定 name，Supervisor 靠它分发
    )

# 创建专业智能体
research_agent = create_specialist_agent(
    "researcher",                      # ← name 用英文标识符（路由映射的 key）
    [search_web, analyze_data],
    "你是研究员，使用工具收集和分析信息。"
)

writing_agent = create_specialist_agent(
    "writer",
    [write_content],
    "你是写作者，使用工具创建高质量内容。"
)

# 主管节点
def supervisor_node(state):
    """主管节点"""
    llm = ChatOpenAI(model="gpt-4")
    
    prompt = """
    分析当前任务，决定分配给哪个智能体：
    - researcher: 需要信息收集和分析
    - writer: 需要内容创作
    - FINISH: 任务完成
    
    只返回智能体名称或 FINISH。
    """
    
    response = llm.invoke(state["messages"] + [{"role": "user", "content": prompt}])
    next_agent = response.content.strip().lower()
    
    return {"next_agent": next_agent}

# 包装智能体为节点
def researcher_node(state):
    result = research_agent.invoke({"messages": state["messages"]})
    return {"messages": result["messages"], "next_agent": "supervisor"}

def writer_node(state):
    result = writing_agent.invoke({"messages": state["messages"]})
    return {"messages": result["messages"], "next_agent": "supervisor"}

# 构建图
graph = StateGraph(State)
graph.add_node("researcher", researcher_node)
graph.add_node("writer", writer_node)
graph.add_node("supervisor", supervisor_node)

graph.add_edge(START, "supervisor")
graph.add_conditional_edges("supervisor", route_after_supervisor, {
    "researcher": "researcher",
    "writer": "writer",
    "end": END
})
graph.add_edge("researcher", "supervisor")
graph.add_edge("writer", "supervisor")

app = graph.compile()
```

## 高级主管模式

### 1. 多轮决策

```python
def advanced_supervisor(state):
    """高级主管 - 支持多轮决策"""
    llm = ChatOpenAI(model="gpt-4")
    
    # 分析任务复杂度
    analysis_prompt = """
    分析当前任务：
    1. 任务复杂度（简单/中等/复杂）
    2. 需要哪些智能体
    3. 执行顺序
    
    返回 JSON 格式的计划。
    """
    
    plan_response = llm.invoke(
        state["messages"] + [{"role": "user", "content": analysis_prompt}]
    )
    
    # 根据计划执行
    plan = parse_json(plan_response.content)
    
    return {
        "plan": plan,
        "current_step": 0,
        "next_agent": plan["agents"][0]
    }

def execute_step(state):
    """执行计划的一步"""
    current_step = state["current_step"]
    plan = state["plan"]
    
    # 执行当前步骤
    agent_name = plan["agents"][current_step]
    result = execute_agent(agent_name, state["messages"])
    
    # 更新步骤
    next_step = current_step + 1
    
    if next_step >= len(plan["agents"]):
        return {"next_agent": "FINISH", "messages": [result]}
    else:
        return {
            "current_step": next_step,
            "next_agent": plan["agents"][next_step],
            "messages": [result]
        }
```

### 2. 动态团队组建

```python
def dynamic_supervisor(state):
    """动态主管 - 根据任务组建团队"""
    llm = ChatOpenAI(model="gpt-4")
    
    # 分析任务需求
    task_analysis = llm.invoke(
        state["messages"] + [{"role": "user", "content": "分析任务需要哪些专业能力"}]
    )
    
    # 根据需求选择智能体
    required_skills = extract_skills(task_analysis.content)
    selected_agents = select_agents(required_skills)
    
    return {
        "team": selected_agents,
        "next_agent": selected_agents[0] if selected_agents else "FINISH"
    }
```

### 3. 质量检查主管

```python
def quality_check_supervisor(state):
    """带质量检查的主管"""
    llm = ChatOpenAI(model="gpt-4")
    
    # 评估当前结果质量
    quality_prompt = """
    评估当前结果的质量（1-10分）：
    - 9-10分: 优秀，可以完成
    - 7-8分: 良好，小幅改进
    - 5-6分: 一般，需要改进
    - <5分: 差，需要重做
    
    返回分数和建议。
    """
    
    quality_response = llm.invoke(
        state["messages"] + [{"role": "user", "content": quality_prompt}]
    )
    
    quality_score = extract_score(quality_response.content)
    
    if quality_score >= 9:
        return {"next_agent": "FINISH"}
    elif quality_score >= 7:
        return {"next_agent": "refiner"}
    else:
        return {"next_agent": "creator"}
```

## 官方快捷方式：langgraph-supervisor 包

不想手写 Supervisor 节点 + 路由逻辑时，官方提供了现成封装（`pip install langgraph-supervisor`）。`create_supervisor` 内部自动完成：Supervisor 节点、路由条件边、Worker 回环——底层就是我们上面手写的图。

```python
from langgraph_supervisor import create_supervisor

supervisor_workflow = create_supervisor(
    agents=[research_agent, writing_agent],   # 就是 create_agent 的产物（需有 name）
    model=llm,
    system_prompt="你是全局调度主管，解析需求、拆分任务、分配给对应专家、整合结果。",
    prompt="你可以访问以下团队成员：{team_members}。根据任务选择下一个要调用的成员，或输出 FINISH。",
)
app = supervisor_workflow.compile()

result = app.invoke({"messages": [{"role": "user", "content": "研究 AI 并写一篇文章"}]})
```

**何时用 create_supervisor vs 手写**：

| 方式 | 优点 | 缺点 |
|------|------|------|
| `create_supervisor` | 零样板代码，开箱即用 | 定制空间小（路由逻辑固定为 LLM 决策） |
| 手写 StateGraph | 完全控制（规则路由、并行、质量门禁） | 需自己写节点/边/回环 |

> 生产建议：需要"确定性规则路由"（如按状态判断走哪步）时手写；纯 LLM 智能路由时用 `create_supervisor`。

## 完整示例：研究写作系统

```python
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from typing import Annotated
from typing_extensions import TypedDict

class State(TypedDict):
    messages: Annotated[list, add_messages]
    next_agent: str
    research_data: str
    draft: str
    iteration: int

def researcher(state: State):
    """研究员 - 收集信息"""
    llm = ChatOpenAI(model="gpt-4")
    
    system_msg = {
        "role": "system",
        "content": "你是研究员，负责收集和分析信息。提供详细的研究结果。"
    }
    
    response = llm.invoke([system_msg] + state["messages"])
    
    return {
        "messages": [response],
        "research_data": response.content,
        "next_agent": "supervisor"
    }

def writer(state: State):
    """写作者 - 创建初稿"""
    llm = ChatOpenAI(model="gpt-4")
    
    system_msg = {
        "role": "system",
        "content": f"""你是写作者，基于以下研究结果创建文章：
        
        {state['research_data']}
        
        创建清晰、有条理的文章。"""
    }
    
    response = llm.invoke([system_msg] + state["messages"])
    
    return {
        "messages": [response],
        "draft": response.content,
        "next_agent": "supervisor"
    }

def editor(state: State):
    """编辑 - 改进内容"""
    llm = ChatOpenAI(model="gpt-4")
    
    system_msg = {
        "role": "system",
        "content": f"""你是编辑，改进以下文章：
        
        {state['draft']}
        
        提高清晰度、流畅性和准确性。"""
    }
    
    response = llm.invoke([system_msg] + state["messages"])
    
    return {
        "messages": [response],
        "draft": response.content,
        "next_agent": "supervisor"
    }

def supervisor(state: State):
    """主管 - 协调工作流程"""
    llm = ChatOpenAI(model="gpt-4")
    
    iteration = state.get("iteration", 0)
    
    # 根据当前状态决定下一步
    if not state.get("research_data"):
        next_agent = "researcher"
    elif not state.get("draft"):
        next_agent = "writer"
    elif iteration < 2:  # 最多编辑 2 次
        next_agent = "editor"
    else:
        next_agent = "FINISH"
    
    return {
        "next_agent": next_agent,
        "iteration": iteration + 1
    }

def route_after_supervisor(state: State):
    """路由决策"""
    next_agent = state.get("next_agent", "FINISH")
    
    if next_agent == "FINISH":
        return "end"
    return next_agent

# 构建图
graph = StateGraph(State)
graph.add_node("researcher", researcher)
graph.add_node("writer", writer)
graph.add_node("editor", editor)
graph.add_node("supervisor", supervisor)

graph.add_edge(START, "supervisor")
graph.add_conditional_edges("supervisor", route_after_supervisor, {
    "researcher": "researcher",
    "writer": "writer",
    "editor": "editor",
    "end": END
})
graph.add_edge("researcher", "supervisor")
graph.add_edge("writer", "supervisor")
graph.add_edge("editor", "supervisor")

app = graph.compile()

# 使用示例
result = app.invoke({
    "messages": [{"role": "user", "content": "写一篇关于量子计算的科普文章"}],
    "next_agent": "",
    "research_data": "",
    "draft": "",
    "iteration": 0
})

print(f"最终文章: {result['draft']}")
```

## 适用场景

### 使用主管模式当：

- **任务分工明确**：每个智能体有清晰的角色
- **需要中央控制**：需要统一的协调和决策
- **工作流程固定**：任务按照预定义的流程执行
- **系统规模较小**：智能体数量在 3-5 个
- **易于调试**：需要清晰的执行路径

### 不适用场景：

- 智能体需要直接通信
- 任务高度动态，难以预测
- 需要高度灵活性的场景
- 智能体数量很多（>10）

## 最佳实践

### 1. 清晰的角色定义

```python
# 好的做法：name 用英文标识符，角色写进 system_prompt
```

## 生产踩坑（真实项目高频问题）

| 坑 | 现象 | 修复 |
|----|------|------|
| **Supervisor 无限循环** | 在 researcher ↔ writer 之间反复路由，永不结束 | prompt 明确完成条件（"当 research_data 和 draft 均非空时必须输出 FINISH"）；State 加 `iteration` 计数器，超过上限强制 `FINISH` |
| **Worker 输出不被 Supervisor 理解** | Code Agent 输出了代码，Supervisor 判断"未完成"继续派发 | 每个 Worker 把**关键结果写入 State 专属字段**（`research_data`/`draft`），Supervisor 读字段而非读冗长消息列表 |
| **并行 Worker 状态覆盖** | 两个并行 Agent 同时写 `messages`，后写的覆盖先写的 | `messages: Annotated[list, add_messages]` 用 reducer 合并（并发安全标准方案） |
| **路由 key 不匹配** | Supervisor 返回 `"researcher"` 但节点注册名是 `"研究员"` | `create_agent(name=...)` 与 `add_conditional_edges` 的映射 key 必须完全一致，统一用英文标识符 |
| **断点恢复缺失** | 长任务中断后从零开始 | `graph.compile(checkpointer=SqliteSaver(...))`，多 Agent 长任务必须持久化 |
| **高风险操作无审批** | Agent 直接发邮件/写库 | 用 `interrupt_before="send_email_node"` 加人工审批节点 |
