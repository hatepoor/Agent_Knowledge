# 网络模式

网络模式（Network Pattern）是多智能体系统的去中心化协作模式，智能体之间可以直接通信和交接任务，没有中央协调者。每个智能体可以根据任务需要，将控制权交给任何其他智能体。

## 核心架构

```
  ┌─────────┐     ┌─────────┐
  │智能体A  │◄───►│智能体B  │
  │Agent A  │     │Agent B  │
  └────┬────┘     └────┬────┘
       │               │
       │    ┌─────────┐│
       └───►│智能体C  │◄┘
            │Agent C  │
            └─────────┘
```

## 工作原理

1. **直接通信**：智能体之间直接交换信息
2. **任务交接**：智能体可以将控制权交给其他智能体
3. **去中心化**：没有中央协调者
4. **动态路由**：智能体根据任务需要决定交给谁
5. **灵活协作**：智能体可以动态组建协作关系

## 基本实现

### 1. 使用 handoff 工具

LangGraph 提供了 `handoff` 工具来实现智能体之间的交接。

```python
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langgraph.prebuilt import create_react_agent
from typing import Annotated
from typing_extensions import TypedDict

class State(TypedDict):
    messages: Annotated[list, add_messages]

# 创建专业智能体
def create_agent(name: str, tools: list, can_handoff_to: list):
    """创建可以交接的智能体"""
    llm = ChatOpenAI(model="gpt-4")
    
    # 添加工具
    all_tools = tools + [
        create_handoff_tool(agent_name=target)
        for target in can_handoff_to
    ]
    
    return create_react_agent(llm, all_tools)

# 创建交接工具
def create_handoff_tool(agent_name: str):
    """创建交接工具"""
    @tool
    def handoff_to_agent(query: str):
        """将任务交接给其他智能体"""
        return f"handoff:{agent_name}:{query}"
    
    handoff_to_agent.name = f"handoff_to_{agent_name}"
    return handoff_to_agent

# 创建智能体
agent_a = create_agent(
    "Agent A",
    [tool_a],
    can_handoff_to=["agent_b", "agent_c"]
)

agent_b = create_agent(
    "Agent B",
    [tool_b],
    can_handoff_to=["agent_a", "agent_c"]
)

agent_c = create_agent(
    "Agent C",
    [tool_c],
    can_handoff_to=["agent_a", "agent_b"]
)
```

### 2. 网络模式图

```python
from langgraph.graph import StateGraph, START, END

def agent_a_node(state: State):
    """智能体 A 节点"""
    result = agent_a.invoke({"messages": state["messages"]})
    
    # 检查是否需要交接
    last_message = result["messages"][-1]
    if "handoff:agent_b" in str(last_message):
        return {"messages": result["messages"], "next_agent": "agent_b"}
    elif "handoff:agent_c" in str(last_message):
        return {"messages": result["messages"], "next_agent": "agent_c"}
    else:
        return {"messages": result["messages"], "next_agent": "end"}

def agent_b_node(state: State):
    """智能体 B 节点"""
    result = agent_b.invoke({"messages": state["messages"]})
    
    last_message = result["messages"][-1]
    if "handoff:agent_a" in str(last_message):
        return {"messages": result["messages"], "next_agent": "agent_a"}
    elif "handoff:agent_c" in str(last_message):
        return {"messages": result["messages"], "next_agent": "agent_c"}
    else:
        return {"messages": result["messages"], "next_agent": "end"}

def agent_c_node(state: State):
    """智能体 C 节点"""
    result = agent_c.invoke({"messages": state["messages"]})
    
    last_message = result["messages"][-1]
    if "handoff:agent_a" in str(last_message):
        return {"messages": result["messages"], "next_agent": "agent_a"}
    elif "handoff:agent_b" in str(last_message):
        return {"messages": result["messages"], "next_agent": "agent_b"}
    else:
        return {"messages": result["messages"], "next_agent": "end"}

def route_after_agent(state: State):
    """路由决策"""
    next_agent = state.get("next_agent", "end")
    
    if next_agent == "agent_a":
        return "agent_a"
    elif next_agent == "agent_b":
        return "agent_b"
    elif next_agent == "agent_c":
        return "agent_c"
    else:
        return "end"

# 构建图
graph = StateGraph(State)
graph.add_node("agent_a", agent_a_node)
graph.add_node("agent_b", agent_b_node)
graph.add_node("agent_c", agent_c_node)

# 所有智能体都可以相互连接
graph.add_edge(START, "agent_a")  # 从 A 开始

for agent in ["agent_a", "agent_b", "agent_c"]:
    graph.add_conditional_edges(agent, route_after_agent, {
        "agent_a": "agent_a",
        "agent_b": "agent_b",
        "agent_c": "agent_c",
        "end": END
    })

app = graph.compile()
```

## 高级网络模式

### 1. 使用 LangGraph 的 handoff 机制

```python
from langgraph.prebuilt import create_react_agent
from langchain_openai import ChatOpenAI

# 创建智能体，使用内置 handoff 工具
researcher = create_react_agent(
    ChatOpenAI(model="gpt-4"),
    [search_tool],
    name="researcher"
)

writer = create_react_agent(
    ChatOpenAI(model="gpt-4"),
    [writing_tool],
    name="writer"
)

editor = create_react_agent(
    ChatOpenAI(model="gpt-4"),
    [editing_tool],
    name="editor"
)

# 创建网络，智能体可以相互交接
network = create_react_agent(
    ChatOpenAI(model="gpt-4"),
    [],
    name="network"
)

# 使用 handoff 工具连接智能体
from langgraph.prebuilt import handoff

network_with_handoff = create_react_agent(
    ChatOpenAI(model="gpt-4"),
    [
        handoff(researcher),
        handoff(writer),
        handoff(editor)
    ]
)
```

### 2. 动态网络

```python
class DynamicNetworkState(TypedDict):
    messages: Annotated[list, add_messages]
    available_agents: list
    current_agent: str

def dynamic_router(state: DynamicNetworkState):
    """动态路由 - 根据任务选择智能体"""
    llm = ChatOpenAI(model="gpt-4")
    
    # 分析任务
    task_analysis = llm.invoke(
        state["messages"] + [{"role": "user", "content": "分析任务需要哪些专业能力"}]
    )
    
    # 动态选择智能体
    required_skills = extract_skills(task_analysis.content)
    selected_agents = select_agents(required_skills, state["available_agents"])
    
    return {
        "current_agent": selected_agents[0],
        "available_agents": selected_agents
    }

def agent_node(state: DynamicNetworkState):
    """智能体节点"""
    current_agent = state["current_agent"]
    
    # 执行当前智能体
    result = execute_agent(current_agent, state["messages"])
    
    # 决定下一步
    next_agent = decide_next_agent(result, state["available_agents"])
    
    return {
        "messages": result["messages"],
        "current_agent": next_agent
    }
```

### 3. 带约束的网络

```python
class ConstrainedNetworkState(TypedDict):
    messages: Annotated[list, add_messages]
    handoff_history: list
    max_handoffs: int

def constrained_agent_node(state: ConstrainedNetworkState):
    """带约束的智能体节点"""
    # 检查交接次数
    if len(state["handoff_history"]) >= state["max_handoffs"]:
        return {
            "messages": state["messages"] + [("assistant", "达到最大交接次数，任务结束")],
            "next_agent": "end"
        }
    
    # 执行智能体
    result = execute_agent(state["current_agent"], state["messages"])
    
    # 记录交接历史
    handoff_history = state["handoff_history"] + [state["current_agent"]]
    
    return {
        "messages": result["messages"],
        "handoff_history": handoff_history,
        "next_agent": decide_next(result)
    }
```

## 完整示例：客服系统

```python
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from typing import Annotated
from typing_extensions import TypedDict

class State(TypedDict):
    messages: Annotated[list, add_messages]
    next_agent: str
    customer_info: dict
    issue_type: str

# 定义工具
@tool
def check_order_status(order_id: str):
    """查询订单状态"""
    return f"订单 {order_id} 状态: 已发货"

@tool
def process_refund(order_id: str, reason: str):
    """处理退款"""
    return f"订单 {order_id} 退款已处理: {reason}"

@tool
def transfer_to_human(query: str):
    """转接人工客服"""
    return f"已转接人工客服: {query}"

@tool
def create_handoff_tool(target_agent: str):
    """创建交接工具"""
    @tool
    def handoff(query: str):
        """交接给其他智能体"""
        return f"handoff:{target_agent}:{query}"
    
    handoff.name = f"handoff_to_{target_agent}"
    return handoff

# 创建智能体
def create_customer_service_agent(name: str, tools: list, handoff_targets: list):
    """创建客服智能体"""
    llm = ChatOpenAI(model="gpt-4")
    
    all_tools = tools + [
        create_handoff_tool(target) for target in handoff_targets
    ]
    
    system_prompt = f"""
    你是{name}，负责处理客户问题。
    如果需要其他专家帮助，使用交接工具。
    """
    
    return create_react_agent(llm, all_tools, state_modifier=system_prompt)

# 创建各个客服智能体
general_support = create_customer_service_agent(
    "通用客服",
    [check_order_status],
    ["technical_support", "billing_support"]
)

technical_support = create_customer_service_agent(
    "技术支持",
    [],
    ["general_support", "billing_support"]
)

billing_support = create_customer_service_agent(
    "账单客服",
    [process_refund],
    ["general_support", "technical_support"]
)

# 节点函数
def general_support_node(state: State):
    result = general_support.invoke({"messages": state["messages"]})
    return handle_handoff(result, "general_support")

def technical_support_node(state: State):
    result = technical_support.invoke({"messages": state["messages"]})
    return handle_handoff(result, "technical_support")

def billing_support_node(state: State):
    result = billing_support.invoke({"messages": state["messages"]})
    return handle_handoff(result, "billing_support")

def handle_handoff(result, current_agent):
    """处理交接逻辑"""
    last_message = str(result["messages"][-1])
    
    if "handoff:general_support" in last_message:
        return {"messages": result["messages"], "next_agent": "general_support"}
    elif "handoff:technical_support" in last_message:
        return {"messages": result["messages"], "next_agent": "technical_support"}
    elif "handoff:billing_support" in last_message:
        return {"messages": result["messages"], "next_agent": "billing_support"}
    else:
        return {"messages": result["messages"], "next_agent": "end"}

def route_after_agent(state: State):
    """路由决策"""
    next_agent = state.get("next_agent", "end")
    
    if next_agent == "general_support":
        return "general_support"
    elif next_agent == "technical_support":
        return "technical_support"
    elif next_agent == "billing_support":
        return "billing_support"
    else:
        return "end"

# 构建图
graph = StateGraph(State)
graph.add_node("general_support", general_support_node)
graph.add_node("technical_support", technical_support_node)
graph.add_node("billing_support", billing_support_node)

graph.add_edge(START, "general_support")

for agent in ["general_support", "technical_support", "billing_support"]:
    graph.add_conditional_edges(agent, route_after_agent, {
        "general_support": "general_support",
        "technical_support": "technical_support",
        "billing_support": "billing_support",
        "end": END
    })

app = graph.compile()

# 使用示例
result = app.invoke({
    "messages": [{"role": "user", "content": "我的订单有问题，需要退款"}],
    "next_agent": "",
    "customer_info": {},
    "issue_type": ""
})
```

## 适用场景

### 使用网络模式当：

- **任务动态变化**：无法预测任务流程
- **需要直接协作**：智能体需要直接交换信息
- **灵活性优先**：需要高度灵活的协作方式
- **智能体数量适中**：3-7 个智能体
- **去中心化决策**：不需要中央控制

### 不适用场景：

- 需要严格的流程控制
- 智能体数量很多（>10）
- 需要清晰的审计跟踪
- 任务流程固定且可预测

## 网络模式 vs 主管模式

| 特性 | 网络模式 | 主管模式 |
|------|----------|----------|
| **控制方式** | 去中心化 | 中央控制 |
| **灵活性** | 高 | 低 |
| **复杂度** | 高 | 低 |
| **可预测性** | 低 | 高 |
| **调试难度** | 高 | 低 |
| **适用规模** | 中小型 | 小型 |
| **通信开销** | 高 | 低 |

## 最佳实践

### 1. 明确的交接规则

```python
# 好的做法：明确的交接规则
def clear_handoff_rules(state):
    """清晰的交接规则"""
    if is_technical_issue(state):
        return "technical_support"
    elif is_billing_issue(state):
        return "billing_support"
    else:
        return "general_support"

# 不好的做法：模糊的交接
def vague_handoff(state):
    """模糊的交接"""
    return llm.invoke("决定交给谁")  # 不可预测
```

### 2. 防止无限循环

```python
def prevent_infinite_loop(state):
    """防止无限循环"""
    handoff_count = state.get("handoff_count", 0)
    
    if handoff_count >= 5:  # 最多交接 5 次
        return {"next_agent": "end", "reason": "达到最大交接次数"}
    
    return {
        "handoff_count": handoff_count + 1,
        "next_agent": decide_next_agent(state)
    }
```

### 3. 记录交接历史

```python
def track_handoffs(state):
    """跟踪交接历史"""
    current_agent = state["current_agent"]
    handoff_history = state.get("handoff_history", [])
    
    # 记录交接
    new_history = handoff_history + [{
        "agent": current_agent,
        "timestamp": datetime.now().isoformat(),
        "reason": state.get("handoff_reason", "unknown")
    }]
    
    return {"handoff_history": new_history}
```
