# 层级模式

层级模式（Hierarchical Pattern）是多智能体系统的多层级协作模式，通过嵌套的主管结构实现复杂任务的管理。管理者智能体监督专业团队，形成树状的组织结构。

## 核心架构

```
            ┌─────────────┐
            │   总监      │
            │   Director  │
            └──────┬──────┘
                   │
          ┌────────┼────────┐
          │        │        │
     ┌────▼──┐ ┌──▼───┐ ┌──▼───┐
     │经理A  │ │经理B  │ │经理C  │
     │Mgr A  │ │Mgr B  │ │Mgr C  │
     └───┬───┘ └───┬──┘ └───┬──┘
         │         │         │
    ┌────┼────┐   ┌┼┐   ┌────┼────┐
    │    │    │   │││   │    │    │
   ┌▼┐ ┌▼┐ ┌▼┐ ┌▼▼▼┐ ┌▼┐ ┌▼┐ ┌▼┐
   │S│ │S│ │S│ │S S│ │S│ │S│ │S│
   └─┘ └─┘ └─┘ └───┘ └─┘ └─┘ └─┘
```

## 工作原理

1. **多层级结构**：总监 → 经理 → 专员
2. **分层管理**：每个管理者只管理直接下属
3. **任务分解**：总监将任务分解给经理，经理再分解给专员
4. **结果汇总**：专员结果汇总到经理，再汇总到总监
5. **职责分离**：每层专注于不同层次的任务

## 基本实现

### 1. 简单的两层结构

```python
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from typing import Annotated
from typing_extensions import TypedDict

class State(TypedDict):
    messages: Annotated[list, add_messages]
    next_agent: str
    task_type: str

# 创建专员智能体
def create_specialist(name: str, tools: list):
    """创建专员智能体"""
    llm = ChatOpenAI(model="gpt-4")
    
    def specialist_node(state: State):
        system_msg = {
            "role": "system",
            "content": f"你是{name}，专注于特定领域的任务。"
        }
        
        response = llm.invoke([system_msg] + state["messages"])
        return {"messages": [response], "next_agent": "manager"}
    
    return specialist_node

# 创建经理智能体
def create_manager(name: str, specialists: list):
    """创建经理智能体"""
    llm = ChatOpenAI(model="gpt-4")
    
    def manager_node(state: State):
        # 经理决定分配给哪个专员
        prompt = f"""
        你是{name}，管理以下专员：{', '.join(specialists)}。
        
        基于当前任务，决定分配给哪个专员。
        只返回专员名称。
        """
        
        response = llm.invoke(state["messages"] + [{"role": "user", "content": prompt}])
        next_specialist = response.content.strip()
        
        return {"messages": [response], "next_agent": next_specialist}
    
    return manager_node

# 创建专员
researcher_a = create_specialist("研究员A", [search_tool])
researcher_b = create_specialist("研究员B", [analysis_tool])
writer = create_specialist("写作者", [writing_tool])

# 创建经理
research_manager = create_manager(
    "研究经理",
    ["researcher_a", "researcher_b"]
)

content_manager = create_manager(
    "内容经理",
    ["writer"]
)

# 构建图
graph = StateGraph(State)

# 添加节点
graph.add_node("researcher_a", researcher_a)
graph.add_node("researcher_b", researcher_b)
graph.add_node("writer", writer)
graph.add_node("research_manager", research_manager)
graph.add_node("content_manager", content_manager)

# 设置边
graph.add_edge(START, "research_manager")

graph.add_conditional_edges("research_manager", lambda s: s["next_agent"], {
    "researcher_a": "researcher_a",
    "researcher_b": "researcher_b"
})

graph.add_edge("researcher_a", "research_manager")
graph.add_edge("researcher_b", "research_manager")

graph.add_edge("research_manager", "content_manager")
graph.add_conditional_edges("content_manager", lambda s: s["next_agent"], {
    "writer": "writer"
})

graph.add_edge("writer", "content_manager")
graph.add_edge("content_manager", END)

app = graph.compile()
```

### 2. 完整的三层结构

```python
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from typing import Annotated
from typing_extensions import TypedDict

class State(TypedDict):
    messages: Annotated[list, add_messages]
    next_agent: str
    level: str  # director, manager, specialist

# 总监节点
def director_node(state: State):
    """总监 - 高层决策"""
    llm = ChatOpenAI(model="gpt-4")
    
    prompt = """
    你是总监，负责高层决策和任务分配。
    
    分析任务，决定分配给哪个经理：
    - research_manager: 研究相关任务
    - content_manager: 内容相关任务
    - tech_manager: 技术相关任务
    
    只返回经理名称。
    """
    
    response = llm.invoke(state["messages"] + [{"role": "user", "content": prompt}])
    next_manager = response.content.strip()
    
    return {"messages": [response], "next_agent": next_manager, "level": "director"}

# 经理节点
def create_manager_node(name: str, managed_specialists: list):
    """创建经理节点"""
    llm = ChatOpenAI(model="gpt-4")
    
    def manager_node(state: State):
        prompt = f"""
        你是{name}，管理以下专员：{', '.join(managed_specialists)}。
        
        基于当前任务，决定分配给哪个专员。
        只返回专员名称。
        """
        
        response = llm.invoke(state["messages"] + [{"role": "user", "content": prompt}])
        next_specialist = response.content.strip()
        
        return {"messages": [response], "next_agent": next_specialist, "level": "manager"}
    
    return manager_node

# 专员节点
def create_specialist_node(name: str, expertise: str):
    """创建专员节点"""
    llm = ChatOpenAI(model="gpt-4")
    
    def specialist_node(state: State):
        system_msg = {
            "role": "system",
            "content": f"你是{name}，专长领域：{expertise}。执行具体任务。"
        }
        
        response = llm.invoke([system_msg] + state["messages"])
        return {"messages": [response], "next_agent": "manager", "level": "specialist"}
    
    return specialist_node

# 创建专员
data_analyst = create_specialist_node("数据分析师", "数据分析、统计")
research_assistant = create_specialist_node("研究助理", "信息收集、文献检索")
content_writer = create_specialist_node("内容写手", "文章写作、内容创作")
tech_specialist = create_specialist_node("技术专员", "技术开发、代码编写")

# 创建经理
research_manager = create_manager_node(
    "研究经理",
    ["data_analyst", "research_assistant"]
)

content_manager = create_manager_node(
    "内容经理",
    ["content_writer"]
)

tech_manager = create_manager_node(
    "技术经理",
    ["tech_specialist"]
)

# 路由函数
def route_from_director(state: State):
    next_agent = state.get("next_agent", "")
    if "research" in next_agent:
        return "research_manager"
    elif "content" in next_agent:
        return "content_manager"
    elif "tech" in next_agent:
        return "tech_manager"
    return END

def route_from_manager(manager_name: str):
    def route(state: State):
        next_agent = state.get("next_agent", "")
        return next_agent if next_agent != "manager" else manager_name
    return route

# 构建图
graph = StateGraph(State)

# 添加所有节点
graph.add_node("director", director_node)
graph.add_node("research_manager", research_manager)
graph.add_node("content_manager", content_manager)
graph.add_node("tech_manager", tech_manager)
graph.add_node("data_analyst", data_analyst)
graph.add_node("research_assistant", research_assistant)
graph.add_node("content_writer", content_writer)
graph.add_node("tech_specialist", tech_specialist)

# 总监到经理
graph.add_edge(START, "director")
graph.add_conditional_edges("director", route_from_director, {
    "research_manager": "research_manager",
    "content_manager": "content_manager",
    "tech_manager": "tech_manager"
})

# 经理到专员
graph.add_conditional_edges("research_manager", route_from_manager("research_manager"), {
    "data_analyst": "data_analyst",
    "research_assistant": "research_assistant",
    "research_manager": "research_manager"
})

graph.add_conditional_edges("content_manager", route_from_manager("content_manager"), {
    "content_writer": "content_writer",
    "content_manager": "content_manager"
})

graph.add_conditional_edges("tech_manager", route_from_manager("tech_manager"), {
    "tech_specialist": "tech_specialist",
    "tech_manager": "tech_manager"
})

# 专员返回经理
graph.add_edge("data_analyst", "research_manager")
graph.add_edge("research_assistant", "research_manager")
graph.add_edge("content_writer", "content_manager")
graph.add_edge("tech_specialist", "tech_manager")

# 经理返回总监或结束
def route_back_to_director(state: State):
    # 检查是否需要其他经理
    if state.get("needs_other_manager"):
        return "director"
    return END

graph.add_conditional_edges("research_manager", route_back_to_director, {
    "director": "director",
    END: END
})

graph.add_conditional_edges("content_manager", route_back_to_director, {
    "director": "director",
    END: END
})

graph.add_conditional_edges("tech_manager", route_back_to_director, {
    "director": "director",
    END: END
})

app = graph.compile()
```

def dynamic_director(state: State):
    """动态总监 - 根据任务组建团队"""
    llm = ChatOpenAI(model="gpt-4")
    
    # 分析任务需求
    task_analysis = llm.invoke(

    )
    
    # 动态选择经理和专员
    required_skills = extract_skills(task_analysis.content)

### 2. 跨团队协作

```python
def cross_team_collaboration(state: State):
    """跨团队协作"""
    current_manager = state["current_manager"]
    
    # 检查是否需要其他团队的支持
    if needs_cross_team_support(state):
        # 请求其他团队支持
        support_request = {
            "from_team": current_manager,
            "to_team": determine_support_team(state),
            "task": state["current_task"]
        }
        
        return {
            "next_agent": "director",
            "support_request": support_request
        }
    
    return {"next_agent": "end"}
```

### 3. 负载均衡

```python
def load_balanced_manager(state: State):
    """负载均衡经理"""
    specialists = state["available_specialists"]
    
    # 选择负载最轻的专员
    least_loaded = min(
        specialists,
        key=lambda s: get_current_load(s)
    )
    
    return {"next_agent": least_loaded}
```

## 完整示例：企业项目管理系统

```python
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from typing import Annotated
from typing_extensions import TypedDict

class ProjectState(TypedDict):
    messages: Annotated[list, add_messages]
    project_info: dict
    next_agent: str
    current_phase: str
    team_assignments: dict

# 总监 - 项目总监
def project_director(state: ProjectState):
    """项目总监 - 高层决策"""
    llm = ChatOpenAI(model="gpt-4")
    
    prompt = """
    你是项目总监，负责整体项目管理。
    
    分析项目需求，决定分配给哪个部门经理：
    - research_manager: 市场调研
    - development_manager: 技术开发
    - marketing_manager: 营销推广
    
    只返回经理名称。
    """
    
    response = llm.invoke(state["messages"] + [{"role": "user", "content": prompt}])
    next_manager = response.content.strip()
    
    return {
        "messages": [response],
        "next_agent": next_manager,
        "current_phase": "planning"
    }

# 部门经理
def create_department_manager(department: str, team_members: list):
    """创建部门经理"""
    llm = ChatOpenAI(model="gpt-4")
    
    def manager_node(state: ProjectState):
        prompt = f"""
        你是{department}部门经理，管理团队成员：{', '.join(team_members)}。
        
        基于当前任务，决定分配给哪个团队成员。
        只返回成员名称。
        """
        
        response = llm.invoke(state["messages"] + [{"role": "user", "content": prompt}])
        next_member = response.content.strip()
        
        return {"messages": [response], "next_agent": next_member}
    
    return manager_node

# 团队成员
def create_team_member(role: str, skills: list):
    """创建团队成员"""
    llm = ChatOpenAI(model="gpt-4")
    
    def member_node(state: ProjectState):
        system_msg = {
            "role": "system",
            "content": f"你是{role}，技能：{', '.join(skills)}。执行具体任务。"
        }
        
        response = llm.invoke([system_msg] + state["messages"])
        return {"messages": [response], "next_agent": "manager"}
    
    return member_node

# 创建团队
research_team = {
    "manager": create_department_manager("研究部", ["analyst", "researcher"]),
    "members": {
        "analyst": create_team_member("数据分析师", ["数据分析", "统计"]),
        "researcher": create_team_member("市场研究员", ["市场调研", "竞品分析"])
    }
}

development_team = {
    "manager": create_department_manager("开发部", ["frontend_dev", "backend_dev"]),
    "members": {
        "frontend_dev": create_team_member("前端开发", ["React", "UI开发"]),
        "backend_dev": create_team_member("后端开发", ["Python", "API开发"])
    }
}

marketing_team = {
    "manager": create_department_manager("营销部", ["content_creator", "social_media"]),
    "members": {
        "content_creator": create_team_member("内容创作", ["写作", "编辑"]),
        "social_media": create_team_member("社交媒体", ["运营", "推广"])
    }
}

# 构建图
graph = StateGraph(ProjectState)

# 添加总监
graph.add_node("director", project_director)

# 添加经理
graph.add_node("research_manager", research_team["manager"])
graph.add_node("development_manager", development_team["manager"])
graph.add_node("marketing_manager", marketing_team["manager"])

# 添加成员
for team in [research_team, development_team, marketing_team]:
    for member_name, member_node in team["members"].items():
        graph.add_node(member_name, member_node)

# 设置边
graph.add_edge(START, "director")

# 总监到经理
graph.add_conditional_edges("director", lambda s: s["next_agent"], {
    "research_manager": "research_manager",
    "development_manager": "development_manager",
    "marketing_manager": "marketing_manager"
})

# 经理到成员
graph.add_conditional_edges("research_manager", lambda s: s["next_agent"], {
    "analyst": "analyst",
    "researcher": "researcher",
    "manager": "director"
})

graph.add_conditional_edges("development_manager", lambda s: s["next_agent"], {
    "frontend_dev": "frontend_dev",
    "backend_dev": "backend_dev",
    "manager": "director"
})

graph.add_conditional_edges("marketing_manager", lambda s: s["next_agent"], {
    "content_creator": "content_creator",
    "social_media": "social_media",
    "manager": "director"
})

# 成员返回经理
graph.add_edge("analyst", "research_manager")
graph.add_edge("researcher", "research_manager")
graph.add_edge("frontend_dev", "development_manager")
graph.add_edge("backend_dev", "development_manager")
graph.add_edge("content_creator", "marketing_manager")
graph.add_edge("social_media", "marketing_manager")

# 经理返回总监
def route_manager_to_director(state: ProjectState):
    # 检查是否需要其他部门支持
    if state.get("needs_cross_department"):
        return "director"
    return END

for manager in ["research_manager", "development_manager", "marketing_manager"]:
    graph.add_conditional_edges(manager, route_manager_to_director, {
        "director": "director",
        END: END
    })

app = graph.compile()

# 使用示例
result = app.invoke({
    "messages": [{"role": "user", "content": "开发一个新的移动应用"}][[,
    "project_info": {},
    "next_agent": "",
    "current_phase": "",
    "team_assignments": {}
})
```

## 适用场景

### 使用层级模式当：

- **系统规模大**：需要管理多个团队
- **职责层次清晰**：总监、经理、专员职责明确
- **需要扩展性**：可以轻松添加新的团队
- **复杂任务**：需要多层级协调
- **企业级应用**：模拟真实组织结构

### 不适用场景：

- 系统规模小（<3个智能体）
- 任务简单，不需要多层管理
- 需要快速响应，层级会降低效率
- 智能体之间需要频繁直接通信

## 层级模式 vs 其他模式

| 特性 | 层级模式 | 主管模式 | 网络模式 |
|------|----------|----------|----------|
| **层级数** | 多层 | 单层 | 无层级 |
| **控制方式** | 分层管理 | 中央控制 | 去中心化 |
| **复杂度** | 高 | 中 | 中 |
| **可扩展性** | 高 | 中 | 高 |
| **适用规模** | 大型 | 小型 | 中型 |
| **灵活性** | 中 | 低 | 高 |
| **管理开销** | 高 | 低 | 中 |

## 最佳实践

### 1. 清晰的职责划分

```python
# 好的做法：清晰的职责
director = create_director(
    role="项目总监",
    responsibilities="高层决策、资源分配、跨部门协调"
)

manager = create_manager(
    role="部门经理",
    responsibilities="团队管理、任务分配、质量控制"
)

specialist = create_specialist(
    role="专员",
    responsibilities="具体任务执行、专业技能应用"
)

# 不好的做法：职责模糊
agent = create_agent(role="智能体", responsibilities="做所有事")
```

### 2. 限制管理层级

```python
# 好的做法：限制层级数
MAX_HIERARCHY_LEVELS = 3

def check_hierarchy_depth(state):
    """检查层级深度"""
    current_level = state.get("level", 0)
    
    if current_level >= MAX_HIERARCHY_LEVELS:
        return {"next_agent": "end", "reason": "达到最大层级"}
    
    return continue_normal_flow(state)

# 不好的做法：无限层级
def unlimited_hierarchy(state):
    return create_more_managers(state)  # 可能导致无限嵌套
```

### 3. 有效的沟通机制

```python
def effective_communication(state):
    """有效的沟通"""
    # 向上汇报
    if needs_director_attention(state):
        return {
            "next_agent": "director",
            "report": generate_report(state)
        }
    
    # 向下分配
    elif needs_team_execution(state):
        return {
            "next_agent": select_team_member(state),
            "task": define_task(state)
        }
    
    # 横向协调
    elif needs_cross_team_support(state):
        return {
            "next_agent": "director",
            "coordination_request": generate_request(state)
        }
```
