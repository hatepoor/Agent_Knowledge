# 异步流式处理

LangGraph 提供了完整的异步流式处理支持，让你能够与非阻塞 I/O 操作无缝集成。这对于构建高性能的 Web 应用、处理并发请求或与其他异步库集成非常重要。

## 核心异步方法

### astream()

`astream()` 是 `stream()` 的异步版本，提供相同的三种流式模式（values、updates、messages），但以异步迭代器的方式返回结果。

```python
import asyncio
from langgraph.graph import StateGraph, START, END
from typing import Annotated
from typing_extensions import TypedDict

class State(TypedDict):
    messages: Annotated[list, add_messages]
    count: int

async def async_node(state: State):
    # 模拟异步操作
    await asyncio.sleep(1)
    return {"count": state["count"] + 1}

# 构建图
graph = StateGraph(State)
graph.add_node("async_node", async_node)
graph.add_edge(START, "async_node")
graph.add_edge("async_node", END)

app = graph.compile()

# 异步流式处理
async def main():
    async for chunk in app.astream({"messages": [], "count": 0}, stream_mode="values"):
        print(chunk)

asyncio.run(main())
```

### astream_events()

`astream_events()` 提供了更细粒度的事件流，可以捕获 LangChain 和 LangGraph 内部的所有事件。这对于调试、监控和构建复杂的流式应用非常有用。

```python
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, START, END
from typing import Annotated
from typing_extensions import TypedDict

class State(TypedDict):
    messages: Annotated[list, add_messages]

llm = ChatOpenAI(model="gpt-4")

def chatbot(state: State):
    return {"messages": [llm.invoke(state["messages"])]}

graph = StateGraph(State)
graph.add_node("chatbot", chatbot)
graph.add_edge(START, "chatbot")
graph.add_edge("chatbot", END)

app = graph.compile()

# 使用 astream_events 获取详细事件
async def main():
    async for event in app.astream_events(
        {"messages": [("user", "Hello")]}, 
        version="v2"
    ):
        event_kind = event["event"]
        
        if event_kind == "on_chat_model_stream":
            # LLM 流式输出
            content = event["data"]["chunk"].content
            if content:
                print(content, end="", flush=True)
        
        elif event_kind == "on_chain_start":
            # 链开始执行
            print(f"\n[开始执行: {event['name']}]")
        
        elif event_kind == "on_chain_end":
            # 链执行结束
            print(f"\n[执行完成: {event['name']}]")

asyncio.run(main())
```

## async/await 模式

### 基本模式

```python
import asyncio
from langgraph.graph import StateGraph, START, END

async def process_stream():
    """基本的异步流式处理模式"""
    async for chunk in app.astream(inputs, stream_mode="values"):
        # 处理每个 chunk
        yield chunk

# 使用异步生成器
async def main():
    async for result in process_stream():
        print(result)

asyncio.run(main())
```

### 并发处理

```python
import asyncio
from langgraph.graph import StateGraph, START, END

async def process_multiple_streams():
    """并发处理多个流"""
    inputs_list = [
        {"query": "问题1"},
        {"query": "问题2"},
        {"query": "问题3"}
    ]
    
    # 并发创建多个流
    tasks = [
        app.astream(inputs, stream_mode="values")
        for inputs in inputs_list
    ]
    
    # 并发处理所有流
    results = await asyncio.gather(*[
        collect_stream(task) for task in tasks
    ])
    
    return results

async def collect_stream(stream):
    """收集流的所有结果"""
    results = []
    async for chunk in stream:
        results.append(chunk)
    return results

asyncio.run(process_multiple_streams())
```

### 超时处理

```python
import asyncio
from langgraph.graph import StateGraph, START, END

async def stream_with_timeout(timeout_seconds=30):
    """带超时的异步流式处理"""
    try:
        async with asyncio.timeout(timeout_seconds):
            async for chunk in app.astream(inputs, stream_mode="values"):
                yield chunk
    except asyncio.TimeoutError:
        print(f"流式处理超时（{timeout_seconds}秒）")
        raise

async def main():
    try:
        async for chunk in stream_with_timeout(30):
            print(chunk)
    except asyncio.TimeoutError:
        # 处理超时
        pass

asyncio.run(main())
```

## 与 Web 框架集成

### FastAPI

FastAPI 是最常用的 Python Web 框架之一，与 LangGraph 的异步流式处理完美集成。

```python
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
from langgraph.graph import StateGraph, START, END
from langchain_openai import ChatOpenAI
from typing import Annotated
from typing_extensions import TypedDict

app = FastAPI()

class State(TypedDict):
    messages: Annotated[list, add_messages]

llm = ChatOpenAI(model="gpt-4")

def chatbot(state: State):
    return {"messages": [llm.invoke(state["messages"])]}

graph = StateGraph(State)
graph.add_node("chatbot", chatbot)
graph.add_edge(START, "chatbot")
graph.add_edge("chatbot", END)

langgraph_app = graph.compile()

@app.post("/chat")
async def chat(message: str):
    """流式聊天接口"""
    
    async def generate():
        async for chunk in langgraph_app.astream(
            {"messages": [("user", message)]}, 
            stream_mode="messages"
        ):
            if chunk[0].content:
                yield f"data: {chunk[0].content}\n\n"
    
    return StreamingResponse(
        generate(),
        media_type="text/event-stream"
    )

@app.post("/chat/events")
async def chat_events(message: str):
    """使用 astream_events 的详细事件接口"""
    
    async def generate():
        async for event in langgraph_app.astream_events(
            {"messages": [("user", message)]}, 
            version="v2"
        ):
            if event["event"] == "on_chat_model_stream":
                content = event["data"]["chunk"].content
                if content:
                    yield f"data: {content}\n\n"
    
    return StreamingResponse(
        generate(),
        media_type="text/event-stream"
    )
```

## 前端集成

### JavaScript/TypeScript

```javascript
// 使用 EventSource 接收流式响应
const eventSource = new EventSource('/chat?message=Hello');

eventSource.onmessage = (event) => {
    console.log('收到消息:', event.data);
    // 更新 UI
    document.getElementById('response').textContent += event.data;
};

eventSource.onerror = (error) => {
    console.error('流式错误:', error);
    eventSource.close();
};
```

### React

```jsx
import { useState } from 'react';

function ChatComponent() {
    const [response, setResponse] = useState('');
    const [isStreaming, setIsStreaming] = useState(false);

    const sendMessage = async (message) => {
        setIsStreaming(true);
        setResponse('');

        const eventSource = new EventSource(`/chat?message=${encodeURIComponent(message)}`);

        eventSource.onmessage = (event) => {
            setResponse(prev => prev + event.data);
        };

        eventSource.onerror = () => {
            setIsStreaming(false);
            eventSource.close();
        };
    };

    return (
        <div>
            <button onClick={() => sendMessage('Hello')}>
                发送消息
            </button>
            <div>{response}</div>
            {isStreaming && <div>正在生成...</div>}
        </div>
    );
}
```

### 事件过滤

async def stream_filtered_events():
    """过滤特定类型的事件"""
    async for event in app.astream_events(inputs, version="v2"):
        event_kind = event["event"]
        
        # 只处理 LLM 流式输出
        if event_kind == "on_chat_model_stream":

            if content:
                yield content
        
        # 忽略其他事件
        # 可以添加更多过滤条件

### 事件转换

```python
async def stream_transformed_events():
    """转换事件格式"""
    async for event in app.astream_events(inputs, version="v2"):
        if event["event"] == "on_chat_model_stream":
            # 转换为自定义格式
            yield {
                "type": "token",

                "timestamp": event["time"]
            }
```

### 错误处理

```python
async def stream_with_error_handling():
    """带错误处理的流式处理"""
    try:
        async for event in app.astream_events(inputs, version="v2"):
            try:
                if event["event"] == "on_chat_model_stream":
                    yield event["data"]["chunk"].content
            except KeyError as e:
                print(f"事件格式错误: {e}")
                continue
    except Exception as e:
        print(f"流式处理错误: {e}")
        raise
```

## 性能优化

### 批量处理

```python
async def batch_stream(inputs_list, batch_size=5):
    """批量处理多个流"""
    for i in range(0, len(inputs_list), batch_size):
        batch = inputs_list[i:i + batch_size]
        
        # 并发处理批次
        tasks = [
            app.astream(inputs, stream_mode="values")
            for inputs in batch
        ]
        
        # 收集结果
        results = await asyncio.gather(*[
            collect_stream(task) for task in tasks
        ][[)
        
        yield results
```

### 背压控制

```python
async def stream_with_backpressure(max_queue_size=10):
    """带背压控制的流式处理"""
    queue = asyncio.Queue(maxsize=max_queue_size)
    
    async def producer():
        async for chunk in app.astream(inputs, stream_mode="values"):
            await queue.put(chunk)
        await queue.put(None)  # 结束信号
    
    async def consumer():
        while True:
            chunk = await queue.get()
            if chunk is None:
                break
            yield chunk
    
    # 启动生产者和消费者
    producer_task = asyncio.create_task(producer())
    
    async for result in consumer():
        yield result
    
    await producer_task
```
