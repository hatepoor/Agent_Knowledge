# 状态定义

## 概述

LangGraph 中的状态定义决定了图中流转的数据结构。选择合适的状态定义方式，直接影响代码的可维护性、类型安全性和开发效率。

LangGraph 支持三种主要的状态定义方式：

1. **TypedDict** — 简单轻量，推荐大多数场景使用
2. **Pydantic BaseModel** — 自带数据验证和序列化
3. **MessagesState** — 内置的聊天消息状态

---

## 方式一：TypedDict（推荐）

`TypedDict` 是 Python 标准库中的类型提示工具，定义一个字典的键值类型。这是 LangGraph 中最常用、最推荐的状态定义方式。

### 基本用法

```python
from typing import TypedDict, Annotated
from langgraph.graph.message import add_messages

# 定义状态结构
class GraphState(TypedDict):
    """图的状态定义"""
    messages: Annotated[list, add_messages]  # 消息列表，使用 add_messages 归约
    query: str                                # 用户查询
    context: str                              # 检索上下文
    answer: str                               # 生成的回答
    retry_count: int                          # 重试计数
    is_complete: bool                         # 是否完成

# 创建图时使用
from langgraph.graph import StateGraph

graph = StateGraph(GraphState)
graph.add_node("retrieve", retrieve_node)
graph.add_node("generate", generate_node)
graph.set_entry_point("retrieve")
graph.add_edge("retrieve", "generate")
```

### 带默认值的 TypedDict

```python
from typing import TypedDict, Annotated, Optional
from langgraph.graph.message import add_messages

class StateWithDefaults(TypedDict):
    messages: Annotated[list, add_messages]
    query: str
    # 注意：TypedDict 本身不支持默认值
    # 需要在图初始化时通过输入参数提供初始值
    max_retries: int
    temperature: float

# 初始状态在调用图时传入
# result = graph.invoke({
#     "messages": [],
#     "query": "什么是 LangGraph？",
#     "max_retries": 3,
#     "temperature": 0.7,
# })
```

### 嵌套 TypedDict

```python
from typing import TypedDict, Annotated, Optional

class Metadata(TypedDict):
    """元数据"""
    source: str          # 来源
    timestamp: str       # 时间戳
    user_id: str         # 用户ID

class GraphState(TypedDict):
    """带嵌套结构的图状态"""
    messages: Annotated[list, add_messages]
    query: str
    metadata: Metadata          # 嵌套的 TypedDict
    optional_field: Optional[str]  # 可选字段
```

### 为什么推荐 TypedDict

- **零依赖**：Python 标准库，无需额外安装
- **语法简洁**：像定义普通类一样直观
- **IDE 友好**：主流 IDE 都能正确推断类型
- **性能好**：运行时就是普通 `dict`，没有额外开销
- **灵活**：支持 `Annotated` 类型标注 Reducer

## 方式二：Pydantic BaseModel

当需要严格的数据验证、序列化、或复杂的字段约束时，可以使用 Pydantic `BaseModel`。

### 基本用法

```python
from pydantic import BaseModel, Field
from typing import Annotated
from langgraph.graph.message import add_messages

class PydanticState(BaseModel):
    """使用 Pydantic 定义状态"""
    messages: Annotated[list, add_messages] = Field(default_factory=list)
    query: str = Field(..., min_length=1, description="用户查询，不能为空")
    context: str = Field(default="", description="检索到的上下文")
    answer: str = Field(default="", description="生成的回答")
    retry_count: int = Field(default=0, ge=0, le=10, description="重试次数，0-10")
    temperature: float = Field(default=0.7, ge=0.0, le=2.0, description="生成温度")

# 创建图时使用（和 TypedDict 用法相同）
from langgraph.graph import StateGraph

graph = StateGraph(PydanticState)
```

### 带验证器的 Pydantic 状态

```python
from pydantic import BaseModel, Field, field_validator
from typing import Annotated, Optional
from langgraph.graph.message import add_messages

class ValidatedState(BaseModel):
    """带自定义验证的状态"""
    messages: Annotated[list, add_messages] = Field(default_factory=list)
    query: str = Field(..., min_length=1)
    answer: str = Field(default="")
    tags: list[str] = Field(default_factory=list)

    @field_validator("query")
    @classmethod
    def query_must_not_be_empty(cls, v: str) -> str:
        """验证查询不为空且去除首尾空格"""
        cleaned = v.strip()
        if not cleaned:
            raise ValueError("查询内容不能为空")
        return cleaned

    @field_validator("tags")
    @classmethod
    def tags_must_be_lowercase(cls, v: list[str]) -> list[str]:
        """验证标签必须是小写"""
        return [tag.lower() for tag in v]
```

### Pydantic 状态的注意事项

```python
# Pydantic BaseModel 在 LangGraph 中的行为：
# 1. 每次状态更新时，会创建新的 Pydantic 实例
# 2. 字段验证会在每次更新时执行
# 3. 性能开销比 TypedDict 大（验证 + 实例化）

# 节点返回的仍然是 dict，LangGraph 会自动转换
def my_node(state: ValidatedState) -> dict:
    # state 是 ValidatedState 实例，可以直接用 . 访问属性
    query = state.query       # 属性访问（Pydantic 风格）
    # 也可以用字典访问
    query = state["query"]    # 字典访问（也可以）
    return {"answer": "新回答"}
```

## 方式三：MessagesState（内置聊天状态）

对于聊天/对话场景，LangGraph 提供了内置的 `MessagesState`，已经预定义了 `messages` 字段并配置了 `add_messages` Reducer。

### 基本用法

```python
from langgraph.graph import StateGraph, MessagesState

# 直接使用内置 MessagesState，无需自定义
graph = StateGraph(MessagesState)

# 节点接收的状态包含 messages 字段
def chatbot(state: MessagesState) -> dict:
    """聊天机器人节点"""
    messages = state["messages"]  # 获取消息列表
    # 调用 LLM...
    response = llm.invoke(messages)
    return {"messages": [response]}  # 返回新消息（追加）

graph.add_node("chatbot", chatbot)
graph.set_entry_point("chatbot")
```

### 扩展 MessagesState

当内置的 `MessagesState` 不够用时，可以继承并扩展：

```python
from typing import Annotated, TypedDict
from langchain_core.messages import AnyMessage
from langgraph.graph.message import add_messages

# 方式一：继承 MessagesState
from langgraph.graph import MessagesState

class ExtendedChatState(MessagesState):
    """扩展的聊天状态，添加额外字段"""
    user_name: str              # 用户名
    conversation_topic: str     # 对话主题
    is_resolved: bool           # 是否已解决

# 方式二：手动定义等价结构
class ManualChatState(TypedDict):
    """手动定义的聊天状态（等价于 MessagesState + 额外字段）"""
    messages: Annotated[list[AnyMessage], add_messages]
    user_name: str
    conversation_topic: str
    is_resolved: bool
```

### MessagesState 的完整聊天示例

```python
from langgraph.graph import StateGraph, MessagesState, START, END
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-4o-mini")

def chatbot(state: MessagesState) -> dict:
    """聊天机器人"""
    response = llm.invoke(state["messages"])
    return {"messages": [response]}

# 构建图
graph = StateGraph(MessagesState)
graph.add_node("chatbot", chatbot)
graph.add_edge(START, "chatbot")
graph.add_edge("chatbot", END)
app = graph.compile()

# 运行（单轮）
result = app.invoke({
    "messages": [("user", "你好，请介绍一下自己")]
})

# 运行（多轮，需要配合 Checkpointer）
# 详见持久化章节
```

## 对比与选择指南

| 特性 | TypedDict | Pydantic BaseModel | MessagesState |
|------|-----------|--------------------|---------------|
| 复杂度 | 低 | 中 | 最低 |
| 数据验证 | 无 | 强（自动验证） | 无 |
| 序列化 | 手动 | 内置 | 手动 |
| 性能 | 最好 | 有额外开销 | 好 |
| 类型提示 | 完整 | 完整 | 完整 |
| 属性访问 | 仅 `[]` | `.` 和 `[]` | 仅 `[]` |
| 适用场景 | 通用 | 需要验证/序列化 | 纯聊天场景 |

### 选择建议

```
需要聊天功能？
├── 只需要 messages 字段 → 使用 MessagesState
└── 需要 messages + 其他字段 → 扩展 MessagesState 或 TypedDict

需要数据验证？
├── 是 → 使用 Pydantic BaseModel
└── 否 → 使用 TypedDict（推荐默认选择）
```

## 完整实战示例

```python
from typing import TypedDict, Annotated, Optional
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langgraph.checkpoint.memory import MemorySaver

# ===== 1. 定义状态 =====
class RAGState(TypedDict):
    """RAG 应用的状态定义"""
    messages: Annotated[list, add_messages]  # 对话消息
    query: str                                # 用户查询
    documents: list[dict]                     # 检索到的文档
    answer: str                               # 最终回答
    source: str                               # 回答来源
    confidence: float                         # 置信度

# ===== 2. 定义节点 =====
def retrieve(state: RAGState) -> dict:
    """检索节点：根据查询检索相关文档"""
    query = state["query"]
    # 模拟检索
    docs = [
        {"content": "文档1内容", "score": 0.95},
        {"content": "文档2内容", "score": 0.87},
    ]
    return {"documents": docs}

def generate(state: RAGState) -> dict:
    """生成节点：基于检索结果生成回答"""
    docs = state["documents"]
    query = state["query"]
    # 模拟生成
    answer = f"基于 {len(docs)} 篇文档，回答：..."
    return {
        "answer": answer,
        "source": "知识库",
        "confidence": 0.92,
    }

# ===== 3. 构建图 =====
graph = StateGraph(RAGState)
graph.add_node("retrieve", retrieve)
graph.add_node("generate", generate)

graph.add_edge(START, "retrieve")
graph.add_edge("retrieve", "generate")
graph.add_edge("generate", END)

# ===== 4. 编译并运行 =====
app = graph.compile()

result = app.invoke({
    "query": "LangGraph 是什么？",
    "messages": [],
})

print(result["answer"])
print(f"置信度: {result['confidence'][[}")
```
