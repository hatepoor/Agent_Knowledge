## 一、镜像构建

### 1.1 Dockerfile（双语言 + 国内源）

```bash
mkdir -p /opt/sandbox && cd /opt/sandbox
cat > Dockerfile << 'EOF'
FROM python:3.11-slim

# 换国内 apt 源（Debian 新格式配置文件在 debian.sources，不是 sources.list）
RUN sed -i 's|deb.debian.org|mirrors.aliyun.com|g; s|security.debian.org|mirrors.aliyun.com|g' \
    /etc/apt/sources.list.d/debian.sources

# 安装 Node.js 18
RUN apt-get update \
    && apt-get install -y --no-install-recommends nodejs npm \
    && rm -rf /var/lib/apt/lists/*

# Python 依赖（走国内 pip 源）
RUN pip install --no-cache-dir -i https://mirrors.aliyun.com/pypi/simple/ \
    pandas numpy requests

WORKDIR /workspace
EOF
```

### 1.2 构建

```bash
cd /opt/sandbox && docker build -t agent-sandbox:1.0 .
```

## 二、沙箱管理类（sandbox_manager.py）

```bash
cat > /opt/sandbox/sandbox_manager.py << 'EOF'
"""沙箱管理器：在 Docker 隔离容器中执行 Python/JS 代码"""
import time
import uuid
from dataclasses import dataclass

import docker

IMAGE = "agent-sandbox:1.0"


@dataclass
class SandboxConfig:
    image: str = IMAGE
    memory_limit: str = "256m"       # 内存上限
    cpu_limit: float = 1.0           # 最多 1 核
    timeout: int = 30                # 单次执行超时秒数
    network_enabled: bool = False    # 默认禁网
    workdir: str | None = None       # 宿主机工作目录（挂载到容器 /workspace）


@dataclass
class SandboxResult:
    stdout: str          # 输出（stdout+stderr 合并）
    stderr: str          # 异常/超时信息（正常时为空）
    exit_code: int       # 0=成功, 124=超时, 137=内存超限被杀, -1=沙箱异常
    container_id: str    # 容器 ID（日志追踪用）
    duration_ms: float   # 执行耗时


class DockerSandbox:
    """每次执行新建容器，跑完自动销毁（remove=True）"""

    def __init__(self, config: SandboxConfig | None = None):
        self.config = config or SandboxConfig()
        self.client = docker.from_env()
        if self.config.workdir:
            import pathlib
            pathlib.Path(self.config.workdir).mkdir(parents=True, exist_ok=True)

    def execute(
        self,
        code: str,
        *,
        language: str = "python",
        timeout: int | None = None,
        env: dict | None = None,
        network_enabled: bool | None = None,
        workdir: str | None = None,
    ) -> SandboxResult:
        container_id = str(uuid.uuid4())[:12]
        timeout = timeout or self.config.timeout
        net = (
            network_enabled if network_enabled is not None
            else self.config.network_enabled
        )
        workdir = workdir or self.config.workdir

        if language == "python":
            cmd = ["python", "-c", code]
        elif language == "js":
            cmd = ["node", "-e", code]
        else:
            return SandboxResult("", f"不支持的语言: {language}", -1, container_id, 0)

        # 挂载配置：有工作目录 → bind mount 到 /workspace；没有 → tmpfs 临时盘
        if workdir:
            import pathlib
            pathlib.Path(workdir).mkdir(parents=True, exist_ok=True)
            mounts = [docker.types.Mount(
                target="/workspace",
                source=workdir,
                type="bind",
            )]
            tmpfs = None
        else:
            mounts = []
            tmpfs = {"/workspace": "size=100m"}

        start = time.time()

        # ── 1. 创建并启动容器（后台运行）──
        try:
            container = self.client.containers.run(
                image=self.config.image,
                command=cmd,
                mem_limit=self.config.memory_limit,
                nano_cpus=int(self.config.cpu_limit * 1e9),
                network_mode="none" if not net else "bridge",
                environment=env or {},
                working_dir="/workspace",
                read_only=True,                          # 根文件系统只读
                tmpfs=tmpfs,
                mounts=mounts,
                cap_drop=["ALL"],                        # 丢弃所有内核能力
                security_opt=["no-new-privileges:true"],
                detach=True,                             # 后台运行，手动等结果
                stdout=True,
                stderr=True,
            )
        except Exception as e:
            return SandboxResult("", f"容器创建失败: {e}", -1, container_id,
                                 (time.time() - start) * 1000)

        # ── 2. 等待执行完成（带超时）──
        try:
            wait_result = container.wait(timeout=timeout)
            exit_code = wait_result["StatusCode"] if isinstance(wait_result, dict) else wait_result
        except Exception:
            try:
                container.kill()
            except Exception:
                pass
            try:
                container.remove(force=True)
            except Exception:
                pass
            return SandboxResult("", f"执行超时（>{timeout}s），容器已终止", 124,
                                 container_id, (time.time() - start) * 1000)

        # ── 3. 取日志并清理 ──
        logs = container.logs(stdout=True, stderr=True).decode("utf-8", "ignore").strip()
        try:
            container.remove(force=True)
        except Exception:
            pass

        return SandboxResult(logs, "", exit_code, container_id,
                             (time.time() - start) * 1000)


# 全局单例：复用 Docker 连接，所有调用共享
sandbox = DockerSandbox()
EOF
```

**docker SDK 两个坑（已踩过）**：
1. `containers.run()` **没有 `timeout` 参数** → 必须 `detach=True` + `container.wait(timeout=n)`，超时后 `kill()` + `remove(force=True)`，退出码 124
2. `container.wait()` 返回的是 **dict**（`{'StatusCode': N}`）不是 int → 要取 `["StatusCode"]`

### 自测

```bash
cd /opt/sandbox && python3 - << 'EOF'
from sandbox_manager import sandbox
tests = [
    ("print(1+1)", "python"),
    ("console.log(2*3)", "js"),
    ("open('/etc/x','w')", "python"),
    ("x = 'a' * 1024 * 1024 * 1024", "python"),
]
for code, lang in tests:
    r = sandbox.execute(code, language=lang)
    print(f"{lang}: exit={r.exit_code} out={r.stdout[:40]!r} err={r.stderr[:40]!r}")
EOF
```

预期：前两个 exit=0，第三个 exit=1（只读），第四个 exit=137（OOM）。

## 三、HTTP API（sandbox_api.py）

```bash
cat > /opt/sandbox/sandbox_api.py << 'EOF'
"""沙箱 HTTP API：代码执行 + 文件读写（带 Token 认证）"""
import os
from pathlib import Path

from fastapi import FastAPI, Header, HTTPException
from pydantic import BaseModel

from sandbox_manager import sandbox

app = FastAPI(title="Sandbox API")

SANDBOX_TOKEN = os.environ.get("SANDBOX_TOKEN", "dev-token-change-me")
WORKDIR = Path("/opt/sandbox/data")   # 文件工作区（execute 时挂载到容器 /workspace）


class ExecRequest(BaseModel):
    code: str                 # 要执行的代码
    language: str = "python"  # python 或 js
    timeout: int | None = None


class FileWriteRequest(BaseModel):
    filename: str      # 相对路径，如 "hello.py" 或 "data/notes.md"
    content: str       # 文件内容


def check_auth(authorization: str) -> None:
    if authorization != f"Bearer {SANDBOX_TOKEN}":
        raise HTTPException(status_code=401, detail="未授权")


def safe_path(filename: str) -> Path:
    """路径安全校验：只允许在工作区内，禁止 ../ 穿越和绝对路径"""
    p = (WORKDIR / filename).resolve()
    if not p.is_relative_to(WORKDIR):
        raise HTTPException(status_code=400, detail="非法路径：不允许访问工作区之外")
    return p


@app.post("/execute")
def execute(req: ExecRequest, authorization: str = Header(default="")):
    check_auth(authorization)
    r = sandbox.execute(req.code, language=req.language, timeout=req.timeout,
                        workdir=str(WORKDIR))
    return {
        "stdout": r.stdout,
        "stderr": r.stderr,
        "exit_code": r.exit_code,
        "duration_ms": r.duration_ms,
    }


@app.post("/files/write")
def files_write(req: FileWriteRequest, authorization: str = Header(default="")):
    check_auth(authorization)
    p = safe_path(req.filename)
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(req.content, encoding="utf-8")
    return {"ok": True, "path": str(p.relative_to(WORKDIR))}


@app.get("/files/read")
def files_read(filename: str, authorization: str = Header(default="")):
    check_auth(authorization)
    p = safe_path(filename)
    if not p.exists():
        raise HTTPException(status_code=404, detail="文件不存在")
    return {"filename": filename, "content": p.read_text(encoding="utf-8")}


@app.get("/health")
def health():
    return {"status": "ok"}
EOF
```

三个接口：

| 接口 | 方法 | 用途 |
|------|------|------|
| `/execute` | POST | 执行 Python/JS 代码，工作区挂载到 /workspace |
| `/files/write` | POST | 写文件到工作区（`{"filename": "...", "content": "..."}`） |
| `/files/read` | GET | 读工作区文件（`?filename=...`） |

认证方式：请求头 `Authorization: Bearer <TOKEN>`，不带 Token 返回 401。

## 四、systemd 服务守护

```bash
cat > /etc/systemd/system/sandbox-api.service << 'EOF'
[Unit]
Description=Sandbox HTTP API
After=docker.service
Requires=docker.service

[Service]
WorkingDirectory=/opt/sandbox
Environment=SANDBOX_TOKEN=34e8320d83817136fb9b3df1f3e78cd5fabd68d6ad6c2d36adc0056b535e65a1
ExecStart=/usr/local/bin/uvicorn sandbox_api:app --host 0.0.0.0 --port 8000
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF

systemctl daemon-reload
systemctl enable --now sandbox-api   # 开机自启 + 立即启动
```

## 六、本地 Agent 工具（tools/sandbox_tool.py）

```python
"""远程沙箱工具：让 Agent 在服务器 Docker 沙箱中安全执行代码"""
import httpx
from langchain_core.tools import tool

SANDBOX_URL = "http://8.130.113.15:8000"
SANDBOX_TOKEN = "34e8320d83817136fb9b3df1f3e78cd5fabd68d6ad6c2d36adc0056b535e65a1"
REQUEST_TIMEOUT = 60  # 秒


@tool
async def sandbox_execute(code: str, language: str = "python") -> str:
    """在远程隔离沙箱中执行 Python 或 JavaScript 代码，返回执行输出。

    代码在服务器的 Docker 容器中运行：网络隔离、只读文件系统、
    内存限制 256MB、默认 30 秒超时。适合数据分析、脚本执行、
    代码验证等需要安全运行代码的场景。

    参数:
        code: 要执行的代码
        language: 语言，python 或 js
    """
    async with httpx.AsyncClient(timeout=REQUEST_TIMEOUT) as client:
        resp = await client.post(
            f"{SANDBOX_URL}/execute",
            json={"code": code, "language": language},
            headers={"Authorization": f"Bearer {SANDBOX_TOKEN}"},
        )

        if resp.status_code == 401:
            return "[沙箱认证失败] Token 无效，请检查配置"
        if resp.status_code == 422:
            return "[沙箱参数错误] 请求格式不正确"
        resp.raise_for_status()
        data = resp.json()

    if data["exit_code"] == 0:
        return data["stdout"] or "[执行完成，无输出]"
    if data["exit_code"] == 124:
        return f"[执行超时] {data['stderr']}"
    if data["exit_code"] == 137:
        return "[内存超限] 代码占用内存超过 256MB 限制，被系统终止"
    return f"[沙箱执行失败 exit={data['exit_code']}]\n{data['stderr'] or data['stdout']}"


@tool
async def sandbox_write_file(filename: str, content: str) -> str:
    """在远程沙箱的工作区中写入文件（支持子目录）。写入后可被 sandbox_execute 读取或执行。

    参数:
        filename: 文件名或相对路径，如 "hello.py" 或 "data/notes.md"
        content: 文件内容
    """
    async with httpx.AsyncClient(timeout=30) as client:
        resp = await client.post(
            f"{SANDBOX_URL}/files/write",
            json={"filename": filename, "content": content},
            headers={"Authorization": f"Bearer {SANDBOX_TOKEN}"},
        )
        if resp.status_code == 401:
            return "[沙箱认证失败] Token 无效"
        resp.raise_for_status()
    return f"文件 {filename} 写入成功"


@tool
async def sandbox_read_file(filename: str) -> str:
    """读取远程沙箱工作区中的文件内容。

    参数:
        filename: 文件名或相对路径
    """
    async with httpx.AsyncClient(timeout=30) as client:
        resp = await client.get(
            f"{SANDBOX_URL}/files/read",
            params={"filename": filename},
            headers={"Authorization": f"Bearer {SANDBOX_TOKEN}"},
        )
        if resp.status_code == 401:
            return "[沙箱认证失败] Token 无效"
        if resp.status_code == 404:
            return f"[文件不存在] {filename}"
        resp.raise_for_status()
    return resp.json()["content"]
```

## 八、故障排查

| 症状 | 原因 | 解决 |
|------|------|------|
| `docker build` 报 `not found` | 拉不到 Docker Hub | 检查 daemon.json 加速器 + `systemctl restart docker` |
| apt 构建极慢（886s+） | 未换国内 apt 源 | Dockerfile 里 sed 换 debian.sources |
| `run() got an unexpected keyword argument 'timeout'` | docker SDK 高级 API 无 timeout | detach + wait(timeout=n) 方案 |
| wait 返回 `{'StatusCode': 0}` | wait 返回 dict | 取 `["StatusCode"]` |
| 本地连不上 :8000 | 安全组/防火墙 | 阿里云控制台放行 8000；`firewall-cmd --add-port=8000/tcp` |
| 401 未授权 | Token 不对 | systemd 的 Environment 和本地工具必须一致 |
| 容器残留 | 执行被 kill | `docker rm -f $(docker ps -a \| grep sandbox \| awk '{print $1}')` |
| 服务器重启后服务没了 | systemd 未 enable | `systemctl enable sandbox-api` |
