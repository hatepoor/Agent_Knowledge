# 沙箱多租户隔离方案

生产环境中，多个 Agent / 多个用户共享同一台 Docker daemon。必须确保 A 用户的容器不能访问 B 用户的文件、不能耗尽共享资源、不能互相干扰。

## 一、隔离的四个维度

```
租户 A 的 Agent               租户 B 的 Agent
  │                              │
  ├── 独立 Docker 网络            ├── 独立 Docker 网络
  ├── 独立资源配额                ├── 独立资源配额
  ├── 独立 skills 目录            ├── 独立 skills 目录
  └── 独立执行日志                └── 独立执行日志
         │                              │
         ▼                              ▼
    ┌─────────────────────────────────────┐
    │     共享 Docker daemon（单实例）      │
    └─────────────────────────────────────┘
```

| 维度 | 隔离手段 | 风险（不隔离） |
|------|---------|---------------|
| **网络** | 独立 Docker bridge 网络 | A 能访问 B 的容器端口 |
| **资源** | 每租户独立 CPU/内存配额 | A 耗尽资源导致 B 无法执行 |
| **文件** | 每租户独立 skills 目录挂载 | A 能读取 B 的技能文件 |
| **日志** | 每租户独立 trace_id + 日志流 | 审计日志混在一起无法追溯 |

## 二、租户配置模型

```python
from dataclasses import dataclass
from pathlib import Path


@dataclass
class TenantConfig:
    """单租户的沙箱配置——每个 Agent/用户一份"""

    tenant_id: str                  # 租户唯一标识（如 "user_001"）
    skills_dir: str                 # 该租户的 skills 目录路径
    memory_limit: str = "256m"      # 该租户的内存上限
    cpu_limit: float = 1.0          # 该租户的 CPU 上限
    network_name: str = ""          # 该租户的 Docker 网络名
    max_concurrent: int = 3         # 该租户最大并发容器数
    allowed_domains: list[str] = None  # 网络白名单（None = 禁网）

    def __post_init__(self):
        if not self.network_name:
            self.network_name = f"sandbox-net-{self.tenant_id}"


class TenantManager:
    """多租户管理器：为每个租户创建独立网络和配额"""

    def __init__(self):
        self.client = docker.from_env()
        self._tenants: dict[str, TenantConfig] = {}
        self._networks: dict[str, str] = {}  # tenant_id → network_id
        self._semaphores: dict[str, asyncio.Semaphore] = {}

    def register_tenant(self, config: TenantConfig):
        """注册新租户：创建独立 Docker 网络"""
        # 1. 创建专属网络
        network = self.client.networks.create(
            name=config.network_name,
            driver="bridge",
            internal=False,  # 需要出站时设为 False
        )
        self._networks[config.tenant_id] = network.id

        # 2. 创建并发信号量
        self._semaphores[config.tenant_id] = asyncio.Semaphore(config.max_concurrent)

        # 3. 存储
        self._tenants[config.tenant_id] = config

        logger.info("tenant.registered",
            tenant_id=config.tenant_id,
            network=config.network_name,
        )

    def get_config(self, tenant_id: str) -> TenantConfig:
        return self._tenants.get(tenant_id)

    def get_semaphore(self, tenant_id: str) -> asyncio.Semaphore:
        return self._semaphores.get(tenant_id)
```

## 三、多租户沙箱执行器

```python
class MultiTenantSandbox:
    """多租户沙箱：每次执行按租户隔离"""

    def __init__(self, tenant_manager: TenantManager):
        self.tm = tenant_manager
        self.client = docker.from_env()

    async def execute(
        self,
        tenant_id: str,
        code: str,
        *,
        language: str = "python",
        timeout: int = 30,
        network_needed: bool = False,
    ) -> SandboxResult:
        """在指定租户的隔离环境中执行代码"""

        config = self.tm.get_config(tenant_id)
        if not config:
            raise ValueError(f"租户未注册: {tenant_id}")

        semaphore = self.tm.get_semaphore(tenant_id)

        # 并发限制：每个租户最多 N 个容器同时执行
        async with semaphore:
            return await asyncio.to_thread(
                self._execute_sync,
                config, code, language, timeout, network_needed,
            )

    def _execute_sync(
        self,
        config: TenantConfig,
        code: str,
        language: str,
        timeout: int,
        network_needed: bool,
    ) -> SandboxResult:
        """同步执行（在线程池中调用）"""

        # 构建租户专属挂载
        mounts = [
            docker.types.Mount(
                target="/skills",
                source=str(Path(config.skills_dir).absolute()),
                type="bind",
                read_only=True,
            ),
        ]

        # 网络策略
        if network_needed and config.allowed_domains:
            network_mode = config.network_name  # 用租户的白名单网络
        elif network_needed:
            network_mode = "bridge"  # 需要网络但无白名单 → 用默认 bridge
        else:
            network_mode = "none"    # 默认禁网

        cmd = ["python", "-c", code] if language == "python" else ["bash", "-c", code]

        start = time.time()
        try:
            container = self.client.containers.run(
                image="python:3.11-slim",
                command=cmd,
                mem_limit=config.memory_limit,
                nano_cpus=int(config.cpu_limit * 1e9),
                network=network_mode,       # ← 租户专属网络
                mounts=mounts,              # ← 租户专属 skills 目录
                read_only=True,
                tmpfs={"/workspace": "size=100m"},
                cap_drop=["ALL"],
                security_opt=["no-new-privileges"],
                environment={
                    "TENANT_ID": config.tenant_id,  # 容器内可感知租户
                    "PYTHONIOENCODING": "utf-8",
                },
                working_dir="/workspace",
                timeout=timeout,
                remove=True,
                detach=False,
                stdout=True,
                stderr=True,
            )
            return SandboxResult(
                stdout=container.decode("utf-8").strip() if container else "",
                exit_code=0,
                container_id=str(uuid.uuid4())[:12],
                duration_ms=(time.time() - start) * 1000,
            )
        except docker.errors.ContainerError as e:
            return SandboxResult(
                stdout=e.stdout.decode("utf-8").strip() if e.stdout else "",
                stderr=e.stderr.decode("utf-8").strip() if e.stderr else str(e),
                exit_code=e.exit_status,
                duration_ms=(time.time() - start) * 1000,
            )
```

## 四、多租户 Agent 集成

```python
from fastapi import FastAPI, Depends, Header, HTTPException
from pydantic import BaseModel

app = FastAPI()
tm = TenantManager()
sandbox = MultiTenantSandbox(tm)

# 预注册租户
tm.register_tenant(TenantConfig(
    tenant_id="team_alpha",
    skills_dir="./skills_data/alpha",
    memory_limit="256m",
    cpu_limit=1.0,
    max_concurrent=3,
))
tm.register_tenant(TenantConfig(
    tenant_id="team_beta",
    skills_dir="./skills_data/beta",
    memory_limit="512m",      # Beta 团队需要更多内存（数据分析）
    cpu_limit=2.0,            # Beta 团队双核
    max_concurrent=5,
))

# 从请求头提取租户 ID
async def get_tenant(x_tenant_id: str = Header(...)):
    if not tm.get_config(x_tenant_id):
        raise HTTPException(status_code=403, detail="租户未注册")
    return x_tenant_id

class ExecuteRequest(BaseModel):
    code: str
    language: str = "python"
    timeout: int = 30
    network_needed: bool = False

@app.post("/sandbox/execute")
async def execute(req: ExecuteRequest, tenant_id: str = Depends(get_tenant)):
    """多租户隔离的沙箱执行"""
    result = await sandbox.execute(
        tenant_id=tenant_id,
        code=req.code,
        language=req.language,
        timeout=req.timeout,
        network_needed=req.network_needed,
    )
    return {
        "stdout": result.stdout,
        "stderr": result.stderr,
        "exit_code": result.exit_code,
        "duration_ms": result.duration_ms,
        "tenant_id": tenant_id,
    }
```

## 五、网络白名单（每租户独立）

```python
import subprocess

def setup_tenant_network(config: TenantConfig):
    """为租户创建带白名单的 Docker 网络"""

    # 1. 创建专属 bridge 网络
    subprocess.run([
        "docker", "network", "create",
        "--driver", "bridge",
        "--subnet", f"172.{hash(config.tenant_id) % 255}.0.0/16",
        config.network_name,
    ], check=True)

    # 2. iptables 白名单（只允许访问指定域名）
    if config.allowed_domains:
        for domain in config.allowed_domains:
            # 解析域名 IP
            ip = socket.gethostbyname(domain)
            subprocess.run([
                "iptables", "-I", "FORWARD",
                "-i", f"br-{config.network_name[:12]}",
                "-d", ip, "-j", "ACCEPT",
            ], check=True)

        # 拒绝其他所有出站
        subprocess.run([
            "iptables", "-A", "FORWARD",
            "-i", f"br-{config.network_name[:12]}",
            "-j", "DROP",
        ], check=True)
```

## 六、资源配额策略

| 租户类型 | memory_limit | cpu_limit | max_concurrent | timeout |
|---------|-------------|-----------|---------------|---------|
| 免费用户 | 128m | 0.5 | 1 | 10s |
| 个人付费 | 256m | 1.0 | 3 | 30s |
| 团队版 | 512m | 2.0 | 5 | 60s |
| 企业版 | 1g | 4.0 | 10 | 120s |

```python
TENANT_TIERS = {
    "free":    {"memory_limit": "128m", "cpu_limit": 0.5, "max_concurrent": 1, "timeout": 10},
    "pro":     {"memory_limit": "256m", "cpu_limit": 1.0, "max_concurrent": 3, "timeout": 30},
    "team":    {"memory_limit": "512m", "cpu_limit": 2.0, "max_concurrent": 5, "timeout": 60},
    "enterprise": {"memory_limit": "1g", "cpu_limit": 4.0, "max_concurrent": 10, "timeout": 120},
}

def create_tenant(tenant_id: str, tier: str, skills_dir: str):
    config = TenantConfig(
        tenant_id=tenant_id,
        skills_dir=skills_dir,
        **TENANT_TIERS[tier],
    )
    tm.register_tenant(config)
```

## 七、审计与追溯

```python
@dataclass
class AuditEntry:
    timestamp: str
    tenant_id: str
    container_id: str
    code_hash: str          # 代码的 SHA256（不存原文）
    exit_code: int
    duration_ms: float
    network_used: bool
    trace_id: str

# 每次执行记录审计
def execute_with_audit(self, tenant_id: str, code: str, **kwargs):
    import hashlib
    result = self.execute(tenant_id, code, **kwargs)

    audit = AuditEntry(
        timestamp=datetime.now().isoformat(),
        tenant_id=tenant_id,
        container_id=result.container_id,
        code_hash=hashlib.sha256(code.encode()).hexdigest()[:16],
        exit_code=result.exit_code,
        duration_ms=result.duration_ms,
        network_used=kwargs.get("network_needed", False),
        trace_id=str(uuid.uuid4()),
    )
    logger.info("audit.execute", **asdict(audit))
    return result
```

## 八、清理与生命周期

```python
class TenantManager:
    def cleanup_tenant(self, tenant_id: str):
        """租户注销时清理所有资源"""
        config = self._tenants.get(tenant_id)
        if not config:
            return

        # 1. 删除 Docker 网络
        try:
            network = self.client.networks.get(config.network_name)
            network.remove()
        except Exception:
            pass

        # 2. 清理信号量
        self._semaphores.pop(tenant_id, None)

        # 3. 删除租户配置
        del self._tenants[tenant_id]

        logger.info("tenant.cleanup", tenant_id=tenant_id)

    def cleanup_all(self):
        """应用关闭时清理所有租户"""
        for tid in list(self._tenants.keys()):
            self.cleanup_tenant(tid)
```

## 九、隔离验证

部署后验证隔离是否生效：

```bash
# 1. 网络隔离验证：A 租户的容器不能 ping B 租户的容器
docker run --rm --network=sandbox-net-team_alpha python:3.11-slim \
    python -c "import socket; socket.gethostbyname('container_in_beta_network')"

# 2. 文件隔离验证：A 租户的容器看不到 B 租户的 skills
docker run --rm -v ./skills_data/alpha:/skills:ro python:3.11-slim \
    ls /skills/
# 应只看到 alpha 的技能文件

# 3. 资源隔离验证：A 租户的容器不能超过配额
docker run --rm --memory=256m python:3.11-slim \
    python -c "x = [0] * 10**9"  # 应被 OOM kill
```
