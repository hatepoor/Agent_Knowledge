# 调试工具与技巧

> 并发代码的调试需要专用工具。本页讲解 faulthandler、线程状态 dump 和线程安全日志。

## faulthandler（崩溃调试）

```python
import faulthandler

# 启用：段错误时打印线程调用栈
faulthandler.enable()

# 定期 dump 线程状态
faulthandler.dump_traceback_later(
    timeout=30,        # 30 秒后自动 dump
    repeat=True,
    file=open("thread_dump.log", "w")
)
```

## 线程状态 dump

```python
import threading
import sys
import traceback

def dump_threads():
    """打印所有活跃线程的调用栈"""
    print(f"\n=== 活跃线程: {threading.active_count()} ===")
    for thread in threading.enumerate():
        if thread is threading.main_thread():
            continue
        print(f"\n线程: {thread.name}")
        frame = sys._current_frames().get(thread.ident)
        if frame:
            traceback.print_stack(frame)

# 在死锁或卡住时调用
dump_threads()
```

## 线程安全日志

```python
import logging
import threading

# logging 模块是线程安全的
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s [%(threadName)s] %(message)s'
)

logger = logging.getLogger(__name__)

def worker(task_id):
    logger.info(f"开始任务 {task_id}")
    logger.info(f"完成任务 {task_id}")

threads = [threading.Thread(target=worker, args=(i,), name=f"Worker-{i}")
           for i in range(5)]
for t in threads: t.start()
for t in threads: t.join()
```

## 死锁看门狗

```python
import threading
import signal
import sys
import traceback

def deadlock_watchdog(timeout=10):
    """超时后打印所有线程状态"""
    def handler(signum, frame):
        print("\n=== 死锁检测 ===")
        for thread in threading.enumerate():
            if thread is threading.main_thread():
                continue
            print(f"\n线程: {thread.name}")
            frame = sys._current_frames().get(thread.ident)
            if frame:
                traceback.print_stack(frame)

    signal.signal(signal.SIGALRM, handler)
    signal.alarm(timeout)
```
