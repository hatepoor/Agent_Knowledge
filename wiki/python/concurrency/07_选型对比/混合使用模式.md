# 混合使用模式

> 实际场景中 I/O 和 CPU 往往混合出现。本页讲解如何组合使用多种并发模型。

## asyncio + ThreadPoolExecutor

在 asyncio 中调用同步阻塞库：

```python
import asyncio
from concurrent.futures import ThreadPoolExecutor
import requests

async def main():
    loop = asyncio.get_event_loop()

    with ThreadPoolExecutor(max_workers=4) as pool:
        tasks = [
            loop.run_in_executor(pool, requests.get, url)
            for url in URLs
        ]
        results = await asyncio.gather(*tasks)
        return [r.text for r in results]

asyncio.run(main())
```

## asyncio + ProcessPoolExecutor

CPU 密集任务不阻塞事件循环：

```python
import asyncio
from concurrent.futures import ProcessPoolExecutor

def cpu_heavy(n):
    return sum(i * i for i in range(n))

async def main():
    loop = asyncio.get_event_loop()

    with ProcessPoolExecutor() as pool:
        result = await loop.run_in_executor(pool, cpu_heavy, 10_000_000)
        print(f"计算结果: {result}")

        # 同时可以做其他 I/O 操作
        await asyncio.sleep(0.1)

asyncio.run(main())
```

## ThreadPool + ProcessPool

I/O 用线程池，CPU 用进程池：

```python
from concurrent.futures import ThreadPoolExecutor, ProcessPoolExecutor

def download(url):
    import time; time.sleep(1)
    return f"data_from_{url}"

def process_data(data):
    return sum(ord(c) for c in data * 10000)

if __name__ == "__main__":
    urls = [f"url_{i}" for i in range(8)]

    with ThreadPoolExecutor(max_workers=4) as io_pool, \
         ProcessPoolExecutor() as cpu_pool:

        # 阶段 1: 并行下载
        all_data = list(io_pool.map(download, urls))

        # 阶段 2: 并行处理
        results = list(cpu_pool.map(process_data, all_data))
```

## 同一任务的三种实现对比

```python
URLS = [f"https://httpbin.org/delay/{i%3}" for i in range(10)]

# threading 方式
def fetch_threading():
    def fetch(url):
        return requests.get(url).text
    with ThreadPoolExecutor() as ex:
        return list(ex.map(fetch, URLS))

# asyncio 方式
async def fetch_asyncio():
    async with aiohttp.ClientSession() as session:
        tasks = [fetch(session, url) for url in URLS]
        return await asyncio.gather(*tasks)

# asyncio + Executor 方式（混合）
async def fetch_hybrid():
    loop = asyncio.get_event_loop()
    with ThreadPoolExecutor() as pool:
        tasks = [loop.run_in_executor(pool, requests.get, url) for url in URLS]
        responses = await asyncio.gather(*tasks)
        return [r.text for r in responses]
```
