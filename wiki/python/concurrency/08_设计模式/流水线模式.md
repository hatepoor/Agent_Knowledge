# 流水线模式

> 数据经过多阶段串行处理，每个阶段是独立线程，阶段间通过 Queue 传递，实现流水线并行。

## 原理

```
数据流：  ↓       ↓       ↓       ↓       ↓
┌─────┐ → ┌─────┐ → ┌─────┐ → ┌─────┐ → ┌─────┐
│下载  │   │解析  │   │处理  │   │存储  │   │通知  │
│Stage1│   │Stage2│   │Stage3│   │Stage4│   │Stage5│
└─────┘   └─────┘   └─────┘   └─────┘   └─────┘
   ↓         ↓          ↓          ↓          ↓
 Queue1    Queue2     Queue3    Queue4    Result
```

## 实现

```python
import threading
import queue
import time
import random

class Pipeline:
    def __init__(self):
        self.queues = []
        self.workers = []

    def add_stage(self, name, func, num_workers=1):
        input_q = self.queues[-1] if self.queues else queue.Queue()
        output_q = queue.Queue()
        if not self.queues:
            self.queues.append(input_q)
        self.queues.append(output_q)

        def worker(worker_id):
            while True:
                item = input_q.get()
                if item is None:
                    output_q.put(None)
                    input_q.task_done()
                    break
                try:
                    result = func(item)
                    if result is not None:
                        output_q.put(result)
                except Exception as e:
                    print(f"  [{name}-{worker_id}] 错误: {e}")
                input_q.task_done()

        for i in range(num_workers):
            t = threading.Thread(target=worker, args=(i,), name=f"{name}-{i}")
            self.workers.append(t)
            t.start()
        return input_q

    def feed(self, data):
        self.queues[0].put(data)

    def get_output_queue(self):
        return self.queues[-1]

    def stop(self):
        self.queues[0].put(None)
        for t in self.workers:
            t.join(timeout=5)

# 定义各阶段
def download(url):
    time.sleep(random.uniform(0.1, 0.3))
    return f"<html>{url}</html>"

def parse(html):
    time.sleep(0.05)
    return {"title": html[:20], "links": 5}

def save(data):
    time.sleep(0.05)
    return f"saved:{data['title']}"

# 构建流水线
pipeline = Pipeline()
input_q = pipeline.add_stage("下载", download, num_workers=3)
pipeline.add_stage("解析", parse, num_workers=2)
pipeline.add_stage("存储", save, num_workers=1)

output_q = pipeline.get_output_queue()
for i in range(10):
    pipeline.feed(f"url_{i}")
pipeline.stop()
```
