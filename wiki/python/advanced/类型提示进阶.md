# 类型提示进阶

> Python 类型提示（Type Hints）在 LangChain 中广泛使用，从工具定义到 Agent 状态管理，类型提示提供了代码可读性、IDE 支持和运行时验证。

## 基础类型提示

```python
from typing import Optional, Union, List, Dict, Tuple, Set

# 基本类型
name: str = "Alice"
age: int = 30
height: float = 1.75
is_active: bool = True

# 容器类型
names: List[str] = ["Alice", "Bob"]
ages: Dict[str, int] = {"Alice": 30, "Bob": 25}
coords: Tuple[float, float] = (1.0, 2.0)
unique_ids: Set[int] = {1, 2, 3}

# 可选类型
middle_name: Optional[str] = None  # 等价于 str | None
```

## 函数类型提示

```python
from typing import Optional, List

def greet(name: str, greeting: str = "Hello") -> str:
    """打招呼函数"""
    return f"{greeting}, {name}!"

def process_items(
    items: List[str],
    *,  # 强制后续参数使用关键字
    max_count: int = 10,
    filter_func: Optional[callable] = None
) -> List[str]:
    """处理项目列表"""
    result = items[:max_count]
    if filter_func:
        result = [x for x in result if filter_func(x)]
    return result
```

## 高级类型

### Union 与 Optional

```python
from typing import Union, Optional

# Union：多种类型之一
def parse(value: Union[int, str, float]) -> str:
    return str(value)

# Optional：等价于 Union[T, None]
def get_user(id: int) -> Optional[dict]:
    # 可能返回 dict 或 None
    pass

# Python 3.10+ 新语法
def parse(value: int | str | float) -> str:
    return str(value)

def get_user(id: int) -> dict | None:
    pass
```

### Literal 类型

```python
from typing import Literal

def set_mode(mode: Literal["read", "write", "append"]) -> None:
    """只接受特定的字面量值"""
    pass

set_mode("read")    # OK
# set_mode("delete")  # 类型检查错误
```

### Annotated（带元数据的类型）

```python
from typing import Annotated
from pydantic import Field

# 结合 Pydantic 使用
class User:
    age: Annotated[int, Field(gt=0, le=150)]
    name: Annotated[str, Field(min_length=1, max_length=100)]

# 自定义元数据
from typing import get_type_hints

def process(
    value: Annotated[int, "positive", "even"]
) -> Annotated[int, "result"]:
    return value * 2

# 获取元数据
hints = get_type_hints(process, include_extras=True)
```

### TypeVar（泛型）

```python
from typing import TypeVar, Generic, List

T = TypeVar("T")  # 类型变量

class Stack(Generic[T]):
    """泛型栈"""
    def __init__(self):
        self.items: List[T] = []
    
    def push(self, item: T) -> None:
        self.items.append(item)
    
    def pop(self) -> T:
        return self.items.pop()
    
    def peek(self) -> T:
        return self.items[-1]

# 使用
int_stack = Stack[int]()
int_stack.push(1)
int_stack.push(2)
value = int_stack.pop()  # 类型推断为 int

str_stack = Stack[str]()
str_stack.push("hello")
```

### Callable（可调用类型）

```python
from typing import Callable, List

def apply_func(items: List[int], func: Callable[[int], int]) -> List[int]:
    """对列表每个元素应用函数"""
    return [func(item) for item in items]

# 使用
result = apply_func([1, 2, 3], lambda x: x * 2)  # [2, 4, 6]
```

### TypeAlias（类型别名）

```python
from typing import TypeAlias

# 定义类型别名
UserId: TypeAlias = int
UserData: TypeAlias = dict[str, str | int]
Callback: TypeAlias = Callable[[str], None]

def get_user(user_id: UserId) -> UserData:
    pass

def register_callback(callback: Callback) -> None:
    pass
```

## LangChain 中的类型应用

### Agent 状态定义

```python
from typing import Annotated, TypedDict, Literal
from langgraph.graph.message import add_messages
from langchain_core.messages import BaseMessage

class AgentState(TypedDict):
    """Agent 状态类型定义"""
    # 消息列表，使用 add_messages 注解实现自动追加
    messages: Annotated[list[BaseMessage], add_messages]
    
    # 当前步骤
    current_step: Literal["think", "act", "observe"]
    
    # 迭代计数
    iteration: int
    
    # 可选的中间结果
    intermediate_steps: list[tuple] | None
```

### 工具参数类型

```python
from langchain.tools import tool
from pydantic import BaseModel, Field
from typing import Literal

class SearchInput(BaseModel):
    """搜索工具输入"""
    query: str = Field(description="搜索关键词")
    search_type: Literal["web", "news", "academic"] = Field(
        default="web",
        description="搜索类型"
    )
    max_results: int = Field(
        default=5,
        ge=1,
        le=20,
        description="最大结果数"
    )

@tool(args_schema=SearchInput)
def search(query: str, search_type: str = "web", max_results: int = 5) -> str:
    """执行搜索"""
    return f"搜索 {query}（{search_type}），返回 {max_results} 条"
```
