# 装饰器进阶

> 装饰器是 Python 的核心特性，在 LangChain 中广泛用于回调系统、中间件、工具注册等场景。理解装饰器是掌握 LangChain 内部机制的关键。

## 装饰器本质

装饰器是一个接收函数并返回函数的高阶函数，用 `@` 语法糖简化调用。

```python
import functools

# 装饰器本质
def my_decorator(func):
    @functools.wraps(func)  # 保留原函数元数据
    def wrapper(*args, **kwargs):
        print("调用前")
        result = func(*args, **kwargs)
        print("调用后")
        return result
    return wrapper

@my_decorator
def say_hello(name):
    """打招呼"""
    print(f"Hello, {name}!")

say_hello("Alice")
# 输出：
# 调用前
# Hello, Alice!
# 调用后
```

## 带参数的装饰器

```python
def repeat(times):
    """重复执行装饰器"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for _ in range(times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(3)
def greet(name):
    print(f"Hello, {name}!")

greet("Alice")  # 打印 3 次
```

## 类装饰器

```python
class CountCalls:
    """统计函数调用次数"""
    def __init__(self, func):
        self.func = func
        self.count = 0
        functools.update_wrapper(self, func)
    
    def __call__(self, *args, **kwargs):
        self.count += 1
        print(f"第 {self.count} 次调用 {self.func.__name__}")
        return self.func(*args, **kwargs)

@CountCalls
def say_hello():
    print("Hello!")

say_hello()  # 第 1 次调用 say_hello
say_hello()  # 第 2 次调用 say_hello
```

## LangChain 回调系统

LangChain 的回调系统基于装饰器模式，允许在 LLM 调用的各个阶段插入自定义逻辑。

### CallbackHandler

```python
from langchain_core.callbacks import BaseCallbackHandler
from langchain_openai import ChatOpenAI

class TimingHandler(BaseCallbackHandler):
    """记录每个步骤的耗时"""
    
    def on_llm_start(self, serialized, prompts, **kwargs):
        print(f"LLM 开始: {serialized.get('name')}")
        self.start_time = time.time()
    
    def on_llm_end(self, response, **kwargs):
        elapsed = time.time() - self.start_time
        print(f"LLM 完成，耗时: {elapsed:.2f}s")
    
    def on_tool_start(self, serialized, input_str, **kwargs):
        print(f"工具调用: {serialized.get('name')}")
    
    def on_tool_end(self, output, **kwargs):
        print(f"工具完成: {output[:50]}...")

# 使用回调
llm = ChatOpenAI(model="gpt-4o-mini", callbacks=[TimingHandler()])
result = llm.invoke("你好")
```

### 常用回调方法

| 方法 | 触发时机 |
|------|---------|
| `on_llm_start` | LLM 开始生成 |
| `on_llm_new_token` | 每个新 token（流式） |
| `on_llm_end` | LLM 生成完成 |
| `on_llm_error` | LLM 出错 |
| `on_chain_start` | 链开始执行 |
| `on_chain_end` | 链执行完成 |
| `on_tool_start` | 工具开始调用 |
| `on_tool_end` | 工具调用完成 |
| `on_agent_action` | Agent 执行动作 |
| `on_agent_finish` | Agent 完成 |

## LangChain 中间件（Middleware）

中间件是 LangChain 0.3+ 引入的概念，用于在模型调用前后插入逻辑。

```python
from langchain.agents.middleware import wrap_model_call

# 定义中间件
@wrap_model_call
def add_system_context(request, handler):
    """在系统提示中添加上下文"""
    system_msg = request.messages[0]
    new_content = system_msg.content + "\n当前时间: 2026-07-08"
    
    # 修改请求
    new_messages = [
        SystemMessage(content=new_content),
        *request.messages[1:]
    ]
    
    # 调用处理器（继续执行）
    return handler(request.override(messages=new_messages))

# 应用到 Agent
agent = create_react_agent(
    llm,
    tools,
    middleware=[add_system_context]
)
```

### 内置中间件

```python
from langchain.agents.middleware import (
    wrap_model_call,
    ModelRetryMiddleware,
    TokenCountMiddleware,
)

# 重试中间件
@wrap_model_call
def retry_on_error(request, handler, max_retries=3):
    for i in range(max_retries):
        try:
            return handler(request)
        except Exception as e:
            if i == max_retries - 1:
                raise
            print(f"重试 {i+1}/{max_retries}")

# Token 计数中间件
@wrap_model_call
def count_tokens(request, handler):
    response = handler(request)
    tokens = response.usage_metadata["total_tokens"]
    print(f"消耗 tokens: {tokens}")
    return response
```

## 工具注册装饰器

```python
from langchain.tools import tool

# 基础工具
@tool
def search(query: str) -> str:
    """搜索信息"""
    return f"搜索结果: {query}"

# 带配置的装饰器
def tool_with_retry(max_retries=3):
    def decorator(func):
        @tool(name=func.__name__, description=func.__doc__)
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for i in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if i == max_retries - 1:
                        raise
            return None
        return wrapper
    return decorator

@tool_with_retry(max_retries=3)
def unstable_api(query: str) -> str:
    """调用不稳定的 API"""
    # 可能失败的操作
    pass
```

## 属性装饰器（@property）

```python
class Agent:
    def __init__(self, name):
        self._name = name
        self._call_count = 0
    
    @property
    def name(self):
        """只读属性"""
        return self._name
    
    @property
    def call_count(self):
        """计算属性"""
        return self._call_count
    
    def run(self):
        self._call_count += 1
        return f"{self.name} 运行中"

agent = Agent("Assistant")
agent.name          # "Assistant"（像属性一样访问）
agent.run()
agent.call_count    # 1
```

## 静态方法与类方法

```python
class ToolRegistry:
    _tools = {}
    
    def __init__(self, name):
        self.name = name
    
    @classmethod
    def register(cls, tool_name, tool_func):
        """类方法：注册工具"""
        cls._tools[tool_name][[ = tool_func
    
    @staticmethod
    def validate_tool_name(name: str) -> bool:
        """静态方法：验证工具名"""
        return name.isidentifier()
    
    @classmethod
    def get_tool(cls, name):
        return cls._tools.get(name)

# 使用
ToolRegistry.register("search", search_func)
tool = ToolRegistry.get_tool("search")
```

## functools 常用工具

| 装饰器/函数 | 作用 |
|------------|------|
| `@wraps(func)` | 保留原函数的 `__name__`、`__doc__` 等 |
| `@lru_cache(maxsize)` | 缓存函数结果（LRU 策略） |
| `@cached_property` | 缓存属性值（只计算一次） |
| `partial(func, *args)` | 偏函数，固定部分参数 |
| `total_ordering` | 自动生成比较方法 |

```python
from functools import lru_cache, cached_property, partial

# LRU 缓存
@lru_cache(maxsize=128)
def expensive_computation(x):
    return x ** 2

# 缓存属性
class Config:
    @cached_property
    def db_connection(self):
        return create_connection()  # 只创建一次

# 偏函数
add_10 = partial(lambda x, y: x + y, 10)
add_10(5)  # 15
```
