# Python 文件操作

> 文件操作涵盖文件读写（open/with）、路径操作（pathlib）和数据序列化（JSON/CSV/pickle）。`with` 语句是管理文件资源的标准方式。

## 文件读写

### 基本用法（推荐 with 语句）

```python
# 读取整个文件
with open("data.txt", "r", encoding="utf-8") as f:
    content = f.read()

# 逐行读取（内存友好，推荐大文件）
with open("data.txt", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())              # strip() 去除换行符

# 读取为行列表
with open("data.txt", "r", encoding="utf-8") as f:
    lines = f.readlines()

# 写入文件（覆盖）
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("Hello\n")
    f.write("World\n")

# 追加写入
with open("log.txt", "a", encoding="utf-8") as f:
    f.write("New log entry\n")

# 写入多行
with open("output.txt", "w", encoding="utf-8") as f:
    f.writelines(["line1\n", "line2\n", "line3\n"])
```

### 打开模式

| 模式 | 说明 | 文件不存在时 | 文件存在时 |
|------|------|-------------|-----------|
| `"r"` | 只读（默认） | 报错 | 正常读取 |
| `"w"` | 写入（覆盖） | 创建 | **清空内容** |
| `"a"` | 追加 | 创建 | 追加到末尾 |
| `"x"` | 独占创建 | 创建 | **报错** |
| `"b"` | 二进制模式 | — | — |
| `"r+"` | 读写 | 报错 | 正常读写 |
| `"w+"` | 读写（覆盖） | 创建 | **清空内容** |
| `"a+"` | 追加+读取 | 创建 | 追加到末尾 |

> **始终指定 `encoding="utf-8"`**，避免跨平台编码问题（Windows 默认 GBK）。

### 文件对象方法

| 方法 | 说明 | 返回值 |
|------|------|--------|
| `f.read(size)` | 读取指定字节数，不指定则读全部 | str |
| `f.readline()` | 读取一行 | str |
| `f.readlines()` | 读取所有行为列表 | list[str] |
| `f.write(str)` | 写入字符串 | int（写入字节数） |
| `f.writelines(lines)` | 写入字符串列表 | None |
| `f.seek(offset)` | 移动文件指针到指定位置 | int |
| `f.tell()` | 当前指针位置 | int |
| `f.flush()` | 刷新缓冲区到磁盘 | None |
| `f.close()` | 关闭文件（with 语句自动调用） | None |
| `f.fileno()` | 文件描述符 | int |
| `f.isatty()` | 是否为终端设备 | bool |
| `f.readable()` | 是否可读 | bool |
| `f.writable()` | 是否可写 | bool |
| `f.seekable()` | 是否可 seek | bool |

### 二进制文件操作

```python
# 读取二进制文件（图片、音频等）
with open("image.png", "rb") as f:
    data = f.read()

# 写入二进制文件
with open("output.bin", "wb") as f:
    f.write(b"\x00\x01\x02\x03")

# 复制文件
with open("source.bin", "rb") as src:
    with open("dest.bin", "wb") as dst:
        dst.write(src.read())
```

## 路径操作（pathlib）

`pathlib` 是 Python 3.4+ 推荐的面向对象路径操作库，替代传统的 `os.path`。

### 创建路径对象

```python
from pathlib import Path

# 绝对路径
p = Path("/home/user/data/file.txt")

# 相对路径
p = Path("data/output/file.txt")

# 当前目录
p = Path(".")

# 用户主目录
p = Path.home()           # /home/user 或 C:\Users\user

# 当前工作目录
p = Path.cwd()
```

### 常用属性

```python
p = Path("data/output/file.txt")

p.name          # "file.txt"（文件名含扩展名）
p.stem          # "file"（不含扩展名）
p.suffix        # ".txt"（扩展名）
p.suffixes      # [".tar", ".gz"]（多扩展名）
p.parent        # Path("data/output")（父目录）
p.parts         # ("data", "output", "file.txt")（路径组件）
p.anchor        # "/" 或 "C:\\"（根目录）
```

### 路径拼接

```python
# 使用 / 运算符拼接
p = Path("data") / "output" / "file.txt"

# 使用 joinpath
p = Path("data").joinpath("output", "file.txt")

# 带变量拼接
base = Path("data")
filename = "report.txt"
p = base / filename
```

### 判断与查询

```python
p = Path("data/file.txt")

p.exists()      # 是否存在（文件或目录）
p.is_file()     # 是否为文件
p.is_dir()      # 是否为目录
p.is_absolute() # 是否为绝对路径
p.is_relative_to(Path("data"))  # 是否在指定目录下（3.9+）

# 文件信息
p.stat()                # 文件状态（大小、时间等）
p.stat().st_size        # 文件大小（字节）
p.stat().st_mtime       # 最后修改时间
```

### 目录操作

```python
# 创建目录
Path("data/output").mkdir(parents=True, exist_ok=True)
# parents=True：自动创建父目录
# exist_ok=True：目录已存在不报错

# 遍历目录
for item in Path(".").iterdir():
    print(item)

# 递归遍历
for f in Path(".").glob("**/*.py"):    # 查找所有 .py 文件
    print(f)

# 模式匹配
for f in Path(".").glob("*.txt"):      # 当前目录 .txt
    print(f)

for f in Path(".").rglob("*.txt"):     # 递归查找 .txt（等价于 **/*.txt）
    print(f)
```

### 快捷读写方法

```python
# 文本读写
Path("data.txt").write_text("Hello, World!", encoding="utf-8")
content = Path("data.txt").read_text(encoding="utf-8")

# 二进制读写
Path("data.bin").write_bytes(b"\x00\x01\x02")
data = Path("data.bin").read_bytes()

# 追加文本
Path("log.txt").write_text("line1\n", encoding="utf-8")
# 注意：write_text 会覆盖，追加需用 open("a")
```

### 路径转换

```python
p = Path("data/file.txt")

# 相对路径 → 绝对路径
p.absolute()            # /current/dir/data/file.txt

# 绝对路径 → 相对路径
p.relative_to(Path.cwd())

# 解析符号链接
p.resolve()             # 完整绝对路径（解析符号链接）

# 转换为字符串
str(p)                  # "data/file.txt"
p.as_posix()            # "data/file.txt"（始终用 /）
p.as_uri()              # "file:///path/to/data/file.txt"
```

### 文件操作

```python
p = Path("data/file.txt")

# 重命名
p.rename("data/new_name.txt")

# 移动（等同于重命名）
p.replace("data/moved.txt")

# 删除
p.unlink()              # 删除文件
p.rmdir()               # 删除空目录

# 复制（需要 shutil）
import shutil
shutil.copy2("source.txt", "dest.txt")       # 复制文件
shutil.copytree("src_dir", "dst_dir")        # 复制目录
shutil.move("source.txt", "dest.txt")        # 移动
```

## 数据序列化

### JSON

```python
import json

# 序列化：Python 对象 → JSON 字符串
data = {"name": "Alice", "age": 30, "hobbies": ["reading", "coding"]}
json_str = json.dumps(data, ensure_ascii=False, indent=2)
print(json_str)
# {
#   "name": "Alice",
#   "age": 30,
#   "hobbies": [
#     "reading",
#     "coding"
#   ]
# }

# 反序列化：JSON 字符串 → Python 对象
parsed = json.loads(json_str)

# 写入文件
with open("data.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)

# 读取文件
with open("data.json", "r", encoding="utf-8") as f:
    data = json.load(f)
```

### JSON 类型映射

| Python | JSON |
|--------|------|
| `dict` | `{}` 对象 |
| `list`/`tuple` | `[]` 数组 |
| `str` | `""` 字符串 |
| `int`/`float` | 数字 |
| `True`/`False` | `true`/`false` |
| `None` | `null` |

### CSV

```python
import csv

# 写入 CSV
with open("data.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["name", "age", "city"])       # 表头
    writer.writerow(["Alice", 30, "Beijing"])
    writer.writerows([                              # 批量写入
        ["Bob", 25, "Shanghai"],
        ["Charlie", 35, "Guangzhou"],
    ])

# 读取 CSV
with open("data.csv", "r", encoding="utf-8") as f:
    reader = csv.reader(f)
    header = next(reader)                           # 读取表头
    for row in reader:
        print(row)

# DictReader/DictWriter（以字典方式读写）
with open("data.csv", "r", encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(row["name"], row["age"])              # 按列名访问

# 写入字典格式
with open("data.csv", "w", newline="", encoding="utf-8") as f:
    fieldnames = ["name", "age", "city"]
    writer = csv.DictWriter(f, fieldnames=fieldnames)
    writer.writeheader()
    writer.writerow({"name": "Alice", "age": 30, "city": "Beijing"})
```

### pickle（Python 对象序列化）

```python
import pickle

# 序列化
data = {"key": [1, 2, 3][[, "nested": {"a": True}}
with open("data.pkl", "wb") as f:
    pickle.dump(data, f)

# 反序列化
with open("data.pkl", "rb") as f:
    loaded = pickle.load(f)

# 注意：pickle 仅用于 Python 环境，不可信数据不要用 pickle（安全风险）
```

## 实用模式

### 安全文件写入（原子写入）

```python
import tempfile
import os

def safe_write(filepath, content):
    """原子写入：先写临时文件，再重命名"""
    dir_path = os.path.dirname(filepath)
    with tempfile.NamedTemporaryFile(
        mode="w",
        dir=dir_path,
        delete=False,
        encoding="utf-8"
    ) as f:
        f.write(content)
        temp_path = f.name
    
    try:
        os.replace(temp_path, filepath)  # 原子重命名
    except:
        os.unlink(temp_path)             # 失败时清理
        raise
```

### 文件监控（简单版）

```python
import time
from pathlib import Path

def watch_file(filepath, callback):
    """监控文件变化"""
    last_mtime = 0
    while True:
        try:
            mtime = Path(filepath).stat().st_mtime
            if mtime > last_mtime:
                last_mtime = mtime
                callback(filepath)
        except FileNotFoundError:
            pass
        time.sleep(1)
```
