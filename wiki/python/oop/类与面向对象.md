# Python 类与面向对象

> Python 支持面向对象编程（OOP），通过 `class` 定义类，支持封装、继承、多态。所有事物皆对象——int、str、list 等都是类的实例。

## 类定义

```python
class Dog:
    """一只狗的类定义"""

    # 类变量（所有实例共享）
    species = "Canis familiaris"

    def __init__(self, name: str, age: int):
        """构造方法：创建实例时自动调用"""
        self.name = name          # 实例变量
        self.age = age

    def bark(self) -> str:
        """实例方法：第一个参数必须是 self"""
        return f"{self.name} says Woof!"

    def __str__(self) -> str:
        """print() 时的字符串表示"""
        return f"Dog({self.name}, {self.age})"

    def __repr__(self) -> str:
        """开发者友好的字符串表示"""
        return f"Dog(name='{self.name}', age={self.age})"

# 创建实例
dog = Dog("Buddy", 5)
dog.bark()              # "Buddy says Woof!"
print(dog)              # Dog(Buddy, 5)
```

## 实例方法、类方法、静态方法

```python
class MyClass:
    class_var = 0

    def __init__(self, value):
        self.instance_var = value

    def instance_method(self):
        """实例方法：通过实例调用，可访问 self"""
        return self.instance_var

    @classmethod
    def class_method(cls, arg):
        """类方法：通过类调用，可访问 cls（类本身）"""
        cls.class_var += 1
        return cls(arg)

    @staticmethod
    def static_method(x, y):
        """静态方法：不访问 self 或 cls，纯工具函数"""
        return x + y

# 调用
obj = MyClass(10)
obj.instance_method()          # 10
MyClass.class_method(20)       # 创建新实例
MyClass.static_method(3, 4)    # 7
```

## 继承

```python
class Animal:
    def __init__(self, name):
        self.name = name

    def speak(self):
        raise NotImplementedError("子类必须实现 speak()")

    def __str__(self):
        return self.name

class Dog(Animal):
    def speak(self):
        return f"{self.name}: Woof!"

class Cat(Animal):
    def speak(self):
        return f"{self.name}: Meow!"

# 多态
animals = [Dog("Buddy"), Cat("Whiskers")]
for animal in animals:
    print(animal.speak())       # Buddy: Woof! / Whiskers: Meow!
```

### super() 调用父类

```python
class Puppy(Dog):
    def __init__(self, name, trick):
        super().__init__(name)    # 调用父类 __init__
        self.trick = trick

    def speak(self):
        base = super().speak()    # 调用父类 speak()
        return f"{base} (can do {self.trick})"

pup = Puppy("Max", "roll over")
pup.speak()              # "Max: Woof! (can do roll over)"
```

### 多重继承与 MRO

```python
class A:
    def method(self):
        return "A"

class B(A):
    def method(self):
        return "B"

class C(A):
    def method(self):
        return "C"

class D(B, C):
    pass

d = D()
d.method()               # "B"（按 MRO 顺序：D → B → C → A）
D.mro()                  # 查看方法解析顺序
```

## property（属性装饰器）

```python
class Temperature:
    def __init__(self, celsius):
        self._celsius = celsius     # 内部变量

    @property
    def celsius(self):
        """像属性一样访问，实际调用方法"""
        return self._celsius

    @celsius.setter
    def celsius(self, value):
        if value < -273.15:
            raise ValueError("温度不能低于绝对零度")
        self._celsius = value

    @property
    def fahrenheit(self):
        """只读计算属性"""
        return self._celsius * 9/5 + 32

t = Temperature(100)
t.celsius              # 100（自动调用 getter）
t.celsius = 0          # 自动调用 setter
t.fahrenheit           # 32.0（计算属性）
# t.fahrenheit = 100   # AttributeError: 无 setter
```

## dataclass（数据类，3.7+）

自动生成 `__init__`、`__repr__`、`__eq__` 等方法。

```python
from dataclasses import dataclass, field

@dataclass
class Point:
    x: float
    y: float
    label: str = "default"
    tags: list = field(default_factory=list)

p1 = Point(1.0, 2.0)
p2 = Point(1.0, 2.0)
p1 == p2               # True（自动实现 __eq__）
print(p1)              # Point(x=1.0, y=2.0, label='default', tags=[][[)
```

## 常用魔术方法（Dunder Methods）

> 详细分类讲解（生命周期、字符串、比较、算术、容器、属性访问、上下文管理器等）见 [[python/oop/魔法方法|魔法方法专题页]]。

| 方法 | 触发场景 | 说明 |
|------|---------|------|
| `__init__(self, ...)` | `MyClass(...)` | 构造方法 |
| `__str__(self)` | `print(obj)` / `str(obj)` | 用户友好的字符串 |
| `__repr__(self)` | 交互式显示 / `repr(obj)` | 开发者友好的字符串 |
| `__len__(self)` | `len(obj)` | 长度 |
| `__getitem__(self, key)` | `obj[key]` | 索引访问 |
| `__setitem__(self, key, value)` | `obj[key] = value` | 索引赋值 |
| `__delitem__(self, key)` | `del obj[key][[` | 索引删除 |
| `__contains__(self, item)` | `item in obj` | 成员检测 |
| `__iter__(self)` | `for x in obj` | 迭代 |
| `__next__(self)` | `next(obj)` | 迭代器下一步 |
| `__call__(self, ...)` | `obj(...)` | 像函数一样调用 |
| `__eq__(self, other)` | `obj1 == obj2` | 相等比较 |
| `__lt__(self, other)` | `obj1 < obj2` | 小于比较（排序用） |
| `__hash__(self)` | `hash(obj)` | 哈希（用于 dict/set） |
| `__add__(self, other)` | `obj1 + obj2` | 加法 |
| `__enter__/__exit__` | `with obj:` | 上下文管理器 |
| `__getattr__(self, name)` | `obj.name`（属性不存在时） | 动态属性访问 |
| `__setattr__(self, name, value)` | `obj.name = value` | 属性赋值 |

### 魔术方法示例

```python
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __repr__(self):
        return f"Vector({self.x}, {self.y})"

    def __add__(self, other):
        return Vector(self.x + other.x, self.y + other.y)

    def __mul__(self, scalar):
        return Vector(self.x * scalar, self.y * scalar)

    def __abs__(self):
        return (self.x**2 + self.y**2) ** 0.5

    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

v1 = Vector(1, 2)
v2 = Vector(3, 4)
v1 + v2              # Vector(4, 6)
v1 * 3               # Vector(3, 6)
abs(v2)              # 5.0
v1 == Vector(1, 2)   # True
```

## 上下文管理器

```python
# 方式一：类实现
class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode

    def __enter__(self):
        self.file = open(self.filename, self.mode, encoding="utf-8")
        return self.file

    def __exit__(self, exc_type, exc_val, exc_tb):
        self.file.close()
        return False              # 不吞异常

with FileManager("test.txt", "w") as f:
    f.write("Hello")

# 方式二：contextmanager 装饰器（更简洁）
from contextlib import contextmanager

@contextmanager
def managed_file(filename, mode):
    f = open(filename, mode, encoding="utf-8")
    try:
        yield f
    finally:
        f.close()

with managed_file("test.txt", "w") as f:
    f.write("Hello")
```

## 抽象基类（ABC）

```python
from abc import ABC, abstractmethod

class Shape(ABC):
    @abstractmethod
    def area(self) -> float:
        """子类必须实现"""
        pass

    @abstractmethod
    def perimeter(self) -> float:
        pass

class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius

    def area(self):
        return 3.14159 * self.radius ** 2

    def perimeter(self):
        return 2 * 3.14159 * self.radius

# shape = Shape()   # TypeError: 不能实例化抽象类
circle = Circle(5)
circle.area()        # 78.54
```
